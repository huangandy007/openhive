# 计划 6：只读工具 ＋ MCP 网关 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 AI 以「民警本人」的身份、只看「当前项目」的数据，通过 7 个只读工具访问事实层——不直连数据库、不给模型库凭据、不见跨项目数据。

**Architecture:** 新增一个**独立进程**的 MCP 服务（`remote` / HTTP / loopback），由内核监督器拉起并把 `mcp.openhive-case` 一条写进全局配置（照治理 MCP 的单进程形态）。每次工具调用由内核注入 `_meta{openhive/user, openhive/directory}`，网关据此解析出 `{userId, projectId}`、校验项目成员身份，再用**只读 SQL** 取数。取数复用计划 5 的 DAO（`事实详情` / `引用们`），其余是新增只读查询。

**Tech Stack:** Bun 1.4.3 · TypeScript 5.8.2 · Effect 4.0.0-beta.83 · Hono 4.10.7 · Drizzle ORM 1.0.0-rc.2 · `@modelcontextprotocol/sdk` 1.29.0（已在 `packages/asset-service` 依赖里）· PGlite（测试用生产驱动夹具）· oxlint 1.60.0

**Spec:** `docs/superpowers/specs/2026-11-05-案件知识与情报体系/design.md` §7.1（走工具，不走裸查）/ §7.2（回答格式）/ §4.2（事实层字段）/ §9.1（可信状态）

**为什么目录能反解出项目**：`@opencode-ai/auth/workspace` 的 `projectDirectory(root, userId, projectId)` 就是 `join(root, userId, projectId)`（`packages/auth/src/workspace.ts:108`），全仓**只有这一处**定义。所以 `<沙箱根>/<userId>/<projectId>` 去掉前两段就是 `projectId`。**不要**另写「猜项目」的算法（按名字、按 git 仓库、查 `project_meta`）——那会与中间件改写 `?directory=` 的那份算法漂（`LEARNINGS #005-04`）。

- [ ] **Step 3: 最小实现**

```ts
import { isAbsolute, relative, resolve, sep } from "node:path"
import { assertSafeUserId } from "@opencode-ai/auth/workspace"

/**
 * 一次工具调用的身份：从内核注入的 `_meta` 解出来。
 *
 * 键名与客户端注入点逐字一致（`packages/opencode/src/mcp/openhive-identity.ts`）；
 * 服务端**不信任**客户端传参，只认这两个键。
 */
export const USER_META_KEY = "openhive/user"
export const DIRECTORY_META_KEY = "openhive/directory"

export type 身份错误码 = "identity_required"

export class 身份错误 extends Error {
  readonly code: 身份错误码 = "identity_required"
  constructor(message: string) {
    super(message)
    this.name = "身份错误"
  }
}

export interface 调用身份 {
  readonly userId: string
  readonly projectId: string
}

const 取字符串 = (record: Record<string, unknown>, key: string): string | undefined => {
  const value = record[key]
  return typeof value === "string" && value.trim() !== "" ? value : undefined
}

/**
 * `_meta` → `{userId, projectId}`。
 *
 * 目录必须**正好**是 `<workspaceRoot>/<userId>/<projectId>`：多一段、少一段、指向别人的
 * 沙箱、带 `..` 逃逸，一律拒。拒绝是安全的默认值——猜错的代价是把别人的项目数据发给模型。
 */
export function 解析调用身份(input: { readonly workspaceRoot: string; readonly meta: unknown }): 调用身份 {
  if (input.meta === null || typeof input.meta !== "object" || Array.isArray(input.meta)) {
    throw new 身份错误("缺少调用身份：_meta 未注入 openhive/user")
  }
  const record = input.meta as Record<string, unknown>
  const userId = 取字符串(record, USER_META_KEY)
  if (userId === undefined) throw new 身份错误("缺少调用身份：_meta 未注入 openhive/user")
  try {
    assertSafeUserId(userId, "解析 MCP 调用身份")
  } catch {
    throw new 身份错误(`调用身份不合法：${JSON.stringify(userId)}`)
  }

  const directory = 取字符串(record, DIRECTORY_META_KEY)
  if (directory === undefined) throw new 身份错误("缺少调用目录：_meta 未注入 openhive/directory")

  const 期望根 = resolve(input.workspaceRoot, userId)
  const 相对 = relative(期望根, resolve(directory))
  if (相对 === "" || isAbsolute(相对)) {
    throw new 身份错误(`调用目录必须落在自己的沙箱里：${JSON.stringify(directory)}`)
  }
  const 段 = 相对.split(sep)
  if (段.length !== 1 || 段[0] === ".." || 段[0] === ".") {
    throw new 身份错误(`调用目录必须正好是「沙箱根/本人/项目」这一层：${JSON.stringify(directory)}`)
  }
  return { userId, projectId: 段[0]! }
}
```

```ts
import { sql, type SQL } from "drizzle-orm"
import type { Queryable } from "./case-store"
import type { 调用身份 } from "./case-mcp-identity"

export interface 工具结果 { readonly 数据: unknown; readonly 口径?: string; readonly 缺口?: readonly string[] }
export type 工具错误码 = "not_a_member" | "bad_input" | "not_found"
export class 工具错误 extends Error {
  readonly code: 工具错误码
  constructor(code: 工具错误码, message: string) {
    super(message)
    this.code = code
    this.name = "工具错误"
  }
}
export interface 工具上下文 { readonly db: Queryable; readonly 身份: 调用身份 }
export const 默认限制 = 50
export const 最大限制 = 200

/**
 * 限制归一化。**超上限是截断不是报错**（模型给 999 是常见行为），但必须在 `缺口` 里说清；
 * 非正整数是模型用错了参数，回 `bad_input` 让它改。
 */
export function 归一化限制(值: number | undefined): number {
  if (值 === undefined) return 默认限制
  if (!Number.isInteger(值) || 值 <= 0) throw new 工具错误("bad_input", `限制必须是正整数，收到 ${String(值)}`)
  return Math.min(值, 最大限制)
}

/**
 * `缺口` 里「结果被截过」的两句。**请求值超上限时无条件说**——只看「实得行数 === 限制」
 * 判断不出来：库里只有 5 行时，请求 999 实得 5 行，模型会以为拿全了。
 */
export function 截断缺口(请求: number | undefined, 限制: number, 实得行数: number): string[] {
  const 句: string[] = []
  if (请求 !== undefined && 请求 > 最大限制) 句.push(`请求的 ${请求} 条超过上限，已按 ${最大限制} 条取`)
  if (实得行数 === 限制) 句.push(`已截到前 ${限制} 条，可能还有更多`)
  return 句
}

/** 成员校验放在调用入口一次，不是每个工具各判一次（`LEARNINGS #004-09`：门在副作用之前）。 */
export async function 校验成员(ctx: 工具上下文): Promise<void> {
  const rows = await ctx.db.execute(sql`
    select 1 from auth.project_member
     where project_id = ${ctx.身份.projectId} and user_id = ${ctx.身份.userId}
     limit 1
  `)
  if (rowsOf(rows).length === 0) {
    throw new 工具错误("not_a_member", `你不是项目「${ctx.身份.projectId}」的成员，看不到它的数据`)
  }
}

export async function search_materials(
  ctx: 工具上下文,
  args: { 关键词?: string; 类型?: string; 限制?: number },
): Promise<工具结果> {
  const 限制 = 归一化限制(args.限制)
  const 条件: SQL[] = [sql`project_id = ${ctx.身份.projectId}`]
  if (args.类型 !== undefined) 条件.push(sql`kind = ${args.类型}`)
  const 关键词 = args.关键词?.trim()
  if (关键词 !== undefined && 关键词 !== "") {
    const 模式 = `%${关键词}%`
    条件.push(sql`(file_name ilike ${模式} or coalesce(text_content, '') ilike ${模式})`)
  }
  const rows = await ctx.db.execute(sql`
    select id, file_name, kind, parse_status, coalesce(text_content, '') as text_content,
           created_at
      from auth.case_material
     where ${sql.join(条件, sql` and `)}
     order by created_at desc, id desc
     limit ${限制}
  `)
  const 数据 = rowsOf(rows).map((行) => ({
    id: 行.id as string,
    file_name: 行.file_name as string,
    kind: 行.kind as string,
    parse_status: 行.parse_status as string,
    片段: 片段(行.text_content as string, 关键词),
    命中文件名: 关键词 !== undefined && lower(行.file_name as string).includes(lower(关键词)),
  }))
  return {
    数据,
    口径: "材料检索按文件名与正文关键词（ilike），材料类型仅限本项目；正文命中给片段不给全文。",
    缺口: 截断缺口(args.限制, 限制, 数据.length),
  }
}

export async function query_entities(
  ctx: 工具上下文,
  args: { 类型?: string; 关键词?: string; 含已排除?: boolean; 限制?: number },
): Promise<工具结果> {
  const 限制 = 归一化限制(args.限制)
  const 条件: SQL[] = [sql`project_id = ${ctx.身份.projectId}`]
  if (args.类型 !== undefined) 条件.push(sql`type = ${args.类型}`)
  const 关键词 = args.关键词?.trim()
  if (关键词 !== undefined && 关键词 !== "") {
    const 模式 = `%${关键词}%`
    条件.push(
      sql`(canonical_value ilike ${模式} or exists (select 1 from unnest(aliases) as 别名 where 别名 ilike ${模式}))`,
    )
  }
  if (args.含已排除 !== true) 条件.push(sql`trust_state <> '已排除'`)
  const rows = await ctx.db.execute(sql`
    select id, type, canonical_value, array_to_json(aliases)::text as aliases, attrs::text as attrs,
           trust_state, created_at
      from auth.case_entity
     where ${sql.join(条件, sql` and `)}
     order by type asc, canonical_value asc, id asc
     limit ${限制}
  `)
  const 数据 = rowsOf(rows).map((行) => ({
    id: 行.id as string,
    type: 行.type as string,
    canonical_value: 行.canonical_value as string,
    aliases: 解JSON<string[]>(行.aliases, []),
    attrs: 解JSON<unknown>(行.attrs, {}),
    trust_state: 行.trust_state as string,
    created_at: 行.created_at as number,
  }))
  return {
    数据,
    口径: "实体按类型与规范值/别名关键词过滤；默认不含「已排除」（设计 §9.1）。",
    缺口: 截断缺口(args.限制, 限制, 数据.length),
  }
}
```

```ts
import { rowsOf } from "@opencode-ai/auth/migrate"   // 与 case-fact-read.ts:4 同一个来源，不要另写

/** 命中片段：以第一个命中为中心前后各 40 个字符。**必须在 JS 做**——SQL 的 substring 按字节切会把中文切碎。
 *  不区分大小写：SQL 那边是 `ilike`，这里若用大小写敏感的 `indexOf`，大写命中会返回 null，模型看不出为什么这行被搜到。 */
export function 片段(正文: string, 关键词: string | undefined, 半径 = 40): string | null {
  if (关键词 === undefined || 关键词 === "" || 正文 === "") return null
  const 位置 = 正文.toLowerCase().indexOf(关键词.toLowerCase())
  if (位置 < 0) return null
  return 正文.slice(Math.max(0, 位置 - 半径), Math.min(正文.length, 位置 + 关键词.length + 半径))
}

/**
 * `text[]` / `jsonb` 一律**以 JSON 文本出库**再解析：仓内没有「驱动读回形状」的约定或先例
 * 可依（`case-fact-read.ts:22-25` 记的是同一件事）。查询写 `array_to_json(col)::text` 或
 * `col::text`，出库形状就与驱动无关；裸选后 `as string[]` 则把「程序库给数组还是给
 * `"{a,b}"` 文本」这种不确定行为直接当成契约，测试与生产一旦不同就是绿在测试、错在生产。
 *
 * （不要把风险归因到某个具体驱动：2026-11-06 复审用真 PG、PGlite socket 与
 * `drizzle-orm/pglite` 三处实测，裸选 `text[]` / `jsonb` 都回数组 / 对象。结论是
 * 「没有先例，所以不给它机会」，不是「某个驱动会回字符串」。）
 */
export function 解JSON<T>(值: unknown, 兜底: T): T {
  if (typeof 值 !== "string" || 值 === "") return 兜底
  try {
    return JSON.parse(值) as T
  } catch {
    return 兜底
  }
}
```

```ts
async function 解析实体(ctx: 工具上下文, 值: string, 称呼: string): Promise<{ id: string; value: string; type: string }> {
  const rows = await ctx.db.execute(sql`
    select id, type, canonical_value from auth.case_entity
     where project_id = ${ctx.身份.projectId} and (id = ${值} or canonical_value = ${值})
     order by (id = ${值}) desc, type asc, id asc
     limit 1
  `)
  const 行 = rowsOf(rows)[0]
  if (行 === undefined) throw new 工具错误("not_found", `本项目里找不到${称呼}：${值}`)
  return { id: String(行.id), value: String(行.canonical_value), type: String(行.type) }
```

`aggregate_money` 的汇总：

```ts
select case when from_id = ${账户.id} then '出' else '入' end as 方向,
       count(*)::int as 笔数,
       coalesce(sum(amount), 0)::text as 合计
  from auth.case_relation
 where project_id = ${项目} and type = '转账' and trust_state <> '已排除'
   and (from_id = ${账户.id} or to_id = ${账户.id})
 [and coalesce(last_at, first_at, 0) >= ${起}] [and coalesce(first_at, last_at, 0) <= ${止}]
 group by 方向
 order by 方向
```

⚠️ **返回前在 JS 里按固定「出 → 入」重排**：`order by 方向` 的顺序由库的 collation 决定（2026-11-06 本机 PGlite 实测先回「入」），断言按「出 → 入」写就必须自己排，不能把排序规则当契约。

对手方：

```ts
select 对手.canonical_value as 对手方,
       case when r.from_id = ${账户.id} then '出' else '入' end as 方向,
       count(*)::int as 笔数,
       coalesce(sum(r.amount), 0)::text as 合计
  from auth.case_relation r
  join auth.case_entity 对手
    on 对手.project_id = r.project_id
   and 对手.id = case when r.from_id = ${账户.id} then r.to_id else r.from_id end
 where r.project_id = ${项目} and r.type = '转账' and r.trust_state <> '已排除'
   and 对手.trust_state <> '已排除'   -- §9.1：已排除实体不进 AI 回答（复审 I2）
   and (r.from_id = ${账户.id} or r.to_id = ${账户.id})
 group by 对手.canonical_value, 方向
 order by sum(r.amount) desc nulls last, 对手.canonical_value asc
 limit 10
```

⚠️ `limit 10` 是**静默上限**：必须在 `缺口` 里写「对手方只列金额前 10」「满额时可能还有更多」，与余额 200 行同一口径（复审 I3）。

余额走势：

```ts
select material_id, sheet, row_no, started_at,
       amount::text as amount, balance::text as balance
  from auth.case_fact_row
 where project_id = ${项目}
   and (counterparty = ${账户.value} or caller = ${账户.value} or callee = ${账户.value})
 [and started_at is not null and started_at >= ${起} and started_at <= ${止}]
 order by started_at asc nulls last, material_id asc, sheet asc, row_no asc
 limit 200
```
**`::text` 是必须的**：`numeric` 在两个驱动上读回形状不同（bun-sql 是字符串、PGlite 可能是数字，`case-fact-read.ts` 头注释记过同一件事），而用例断言的是 `"100.00"`。PG 的 `numeric::text` 保留列的小数位，两个驱动都拿到同一串。
取到 0 行时 `余额走势: null`，并在 `缺口` 里写「本项目明细行里没有该账户的流水（按 counterparty/caller/callee 精确匹配）」。取到 > 0 行时：

- **首末**：只从「`started_at` 非 null 且 `balance` 非 null」的行里取，顺序由 `order by started_at asc` 定。**不能直接拿数组首末**——`nulls last` 会把无时间的行排在最后，它没有时间，不代表「最后一条流水」。全部行都没有时间时首末给 `null`。
- **极值**：`balance` 非 null 的行都参与，用 `Number(字符串)` 比大小、返回原始字符串。
- `行数` = 命中的行数（含无时间的行）。

`find_crossings` 的共同时空：

```ts
select id, type, occurred_at, occurred_raw, precision,
       array_to_json(participants)::text as participants, summary
  from auth.case_event
 where project_id = ${项目} and participants @> array[${甲.value}, ${乙.value}]::text[]
   and trust_state <> '已排除'
 [and occurred_at is not null and occurred_at >= ${起} and occurred_at <= ${止}]
 order by occurred_at desc nulls last, id asc
 limit 50
```

⚠️ `limit 50` 同样是**静默上限**：`缺口` 必须写「共同时空最多 50 条」「满额时可能还有更多」（复审 I3）。`participants` 里若含已排除实体，按 §9.1 处理（剔除这些值，或在 `口径`+`缺口` 如实说明未剔除——二选一，别默认过去，复审 I2）。

共同关联：**两侧各查一次**（不要合成一条 `or` 查询再 `limit 500`——那样一方的边可能把配额吃光，另一方的交集被静默截断）：

```ts
// 对 x ∈ [甲.id, 乙.id] 各跑一次：
select case when from_id = ${x} then to_id else from_id end as 另一端
  from auth.case_relation
 where project_id = ${项目} and trust_state <> '已排除'
   and (from_id = ${x} or to_id = ${x})
 order by 另一端 asc
 limit 500
```
JS 里各自收集「另一端 id」成两个集合（两侧各限 500 条，用满时**在 `缺口` 里写明**「每侧最多取 500 条关系，交集可能不全」），取交集；交集为空直接返回 `[]`，不空则回查实体行（`id = any(...)` 用 `${sql.raw}` 不安全，改用 `in` 片段拼：`sql`id in (${sql.join(ids.map((i) => sql`${i}`), sql`, `)})``），**回查结果的行形状逐字为** `{ id, type, value, trust_state }`（`canonical_value as value`）——用例断言的是 `.value`。回查也要带 `project_id` 与 `trust_state <> '已排除'`。

**分发规则（写死）：**

| 步骤 | 失败 ⇒ |
|---|---|
| ① `解析调用身份({ workspaceRoot: deps.workspaceRoot, meta: request.params._meta })` | `identity_required` |
| ② `校验成员(ctx)` | `not_a_member` |
| ③ 按工具名调用对应函数，`args` 原样传入 | `工具错误.code`（`bad_input` / `not_found`） |
| ④ 其余异常 | `tool_failed`。**回给模型的 message 是一句固定中文**（「工具执行失败，已记入服务端日志」），**原文一个字符都不回显**——原文可能带表名 / 列名 / SQL 片段；原文只进 stderr 留痕的 `错误原文` 字段给运维查（用例 `工具抛非业务异常 ⇒ tool_failed` 钉的就是这一条） |

返回体统一：
- 成功 ⇒ `{ content: [{ type: "text", text: JSON.stringify(结果, null, 2) }] }`
- 失败 ⇒ `{ isError: true, content: [{ type: "text", text: JSON.stringify({ error: 错误码, message }) }] }`

**留痕**：每次调用**无论成败**都出一行（`ok` / `错误码` 区分）。默认实现：
```ts
const 默认留痕 = (行: 调用留痕): void => {
  process.stderr.write(`${JSON.stringify({ 事件: "mcp工具调用", ...行 })}\n`)
}
```
> 为什么是 stderr 而不是表：`auth.asset_usage_event` 那套是**事后**从每用户 SQLite 的 part 表抽取的，网关没有 part、也不该为一行日志新开迁移。表级使用记录与「工具授权闸」一起做（见缺口表）。**这一条不是「已覆盖」，是明说的缺口。**

`收参数` 做**类型收口**（四参数：`args`、文本字段、整数字段、布尔字段——上一版把 `含已排除` 塞进整数列表是笔误）：

```ts
/**
 * 模型给的 JSON → 工具入参。**未列出的键 ⇒ bad_input**（fail-closed，不静默忽略——
 * 模型把「起始时间」写成 `起时` 时，静默丢掉等于按全时段返回，答案会悄悄错）。
 */
function 收参数<T>(args: Record<string, unknown>, 文本: string[], 整数: string[], 布尔: string[]): T {
  const 允许 = new Set([...文本, ...整数, ...布尔])
  for (const 键 of Object.keys(args)) {
    if (!允许.has(键)) throw new 工具错误("bad_input", `不认识的参数：${键}；可用参数：${[...允许].join("、")}`)
  }
  const 结果: Record<string, unknown> = {}
  for (const 键 of 文本) {
    const 值 = args[键]
    if (值 === undefined) continue
    if (typeof 值 !== "string") throw new 工具错误("bad_input", `参数「${键}」必须是字符串`)
    结果[键] = 值
  }
  for (const 键 of 整数) {
    const 值 = args[键]
    if (值 === undefined) continue
    if (typeof 值 !== "number" || !Number.isInteger(值)) throw new 工具错误("bad_input", `参数「${键}」必须是整数`)
    结果[键] = 值
  }
  for (const 键 of 布尔) {
    const 值 = args[键]
    if (值 === undefined) continue
    if (typeof 值 !== "boolean") throw new 工具错误("bad_input", `参数「${键}」必须是布尔值`)
    结果[键] = 值
  }
  return 结果 as unknown as T
}
```

`收参数` 的返回类型 `T` 由调用点显式给（分发表里已写成 `收参数<Parameters<typeof 工具名>[1]>(…)`）；返回值走 `as unknown as T` 是两段式断言，**不要**退回 `as any`。

```ts
export const CASE_MCP_SERVER = "openhive-case"
export const COMMAND_ENV = "OPENHIVE_CASE_MCP_CMD"
export const DISABLE_ENV = "OPENHIVE_CASE_MCP"

export function caseCommand(env: Record<string, string | undefined>): CommandPlan {
  if (env[DISABLE_ENV] === "off") return { kind: "off" }
  const explicit = env[COMMAND_ENV]
  if (explicit !== undefined && explicit.trim() !== "") {
    try {
      const parsed: unknown = JSON.parse(explicit)
      if (!Array.isArray(parsed) || parsed.length < 2 || !parsed.every((it) => typeof it === "string")) {
        return { kind: "invalid", reason: `${COMMAND_ENV} 必须是长度 ≥ 2 的字符串数组` }
      }
      return { kind: "command", command: parsed }
    } catch {
      return { kind: "invalid", reason: `${COMMAND_ENV} 不是合法 JSON` }
    }
  }
  // 五连，与 `connect()` 的 REQUIRED 一致（`packages/auth/src/db.ts:3`）。
  if (!env.PG_HOST || !env.PG_PORT || !env.PG_USER || !env.PG_PASSWORD || !env.PG_DATABASE) return { kind: "off" }
  return { kind: "command", command: ["bun", "run", join(import.meta.dir, "case-mcp-main.ts").replaceAll("\\", "/")] }
}

export function readyConfigPatch(port: number) {
  return { mcp: { [CASE_MCP_SERVER]: { type: "remote" as const, url: `http://127.0.0.1:${port}${MCP_PATH}` } } }
```
