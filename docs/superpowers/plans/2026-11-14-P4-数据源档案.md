# P4 学习层 · 数据源档案 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让「同一类来源、列顺序变了」的表格在指纹不命中时，按表头集合套用已确认的档案映射，把 L0 全自动从「同一份格式」扩展到「同一类来源」。

**Architecture:** 新表 `auth.case_source_profile` 按项目存「表头原文 → 字段」映射；解析时先按表头集合匹配档案，命中则不跑四层、不调模型、不写格式缓存，只 `hit_count + 1`；材料解析成功时把各表指纹写进 `case_material.format_fingerprints`，建档接口据此从 `case_table_format` 读回表头与映射，不重读文件。

**Tech Stack:** Bun + TypeScript、Drizzle SQL 模板、Effect HTTP 路由、`bun:test`、真 PG 夹具（`startProductionDb` + `migrate`）。

**Spec:** `docs/superpowers/specs/2026-11-09-事实卡片抽象/2026-11-14-P4-数据源档案-design.md`

`case-store.ts` 末尾（挨着其他材料写函数）加：

```ts
/**
 * 解析成功时记录各表指纹。与状态 / 事实写入**同一事务**：材料标了「已解析」却没有指纹，
 * 建档接口只会说「先重放」，用户看不出是写漏还是真没解析过（`#007-01`）。
 */
export async function setMaterialFormatFingerprints(
  db: Queryable,
  input: {
    projectId: string
    materialId: string
    readonly 指纹们: readonly { 工作表: string; 指纹: string }[]
  },
): Promise<void> {
  await db.execute(sql`
    update auth.case_material
       set format_fingerprints = ${JSON.stringify(input.指纹们)}::text::jsonb
     where project_id = ${input.projectId} and id = ${input.materialId}
  `)
}
```

```ts
export * as OpenhiveCaseSourceProfile from "./case-source-profile"

import { classifyMaterialType } from "./case-mapping"
import { 是识别字段, type 识别字段 } from "./case-field-registry"

/**
 * 数据源档案的命中判定（设计 §4）。纯函数：不碰库、不读时钟。
 *
 * 与格式缓存的区别：缓存键是「表头 + 每列形状」的指纹，列顺序一变就失效；
 * 档案键是**表头集合**，列顺序无关。所以这里的每一步都必须 fail-closed：
 * 缺列、字段失效、类型不符、两档案打平——一律不命中，回落普通识别。
 */
export interface 档案候选 {
  id: string
  name: string
  type: "资金" | "话单" | "其他"
  /** 表头原文（trim 后）→ 字段。 */
  mapping: Record<string, string>
  hitCount: number
  updatedAt: number
}

export interface 档案命中 {
  id: string
  name: string
  /** 还原成**当前**列下标。 */
  columns: Record<number, 识别字段>
}

interface 合格候选 {
  档案: 档案候选
  columns: Record<number, 识别字段>
  有效列数: number
  签名: string
}

/** 映射签名：只看「表头 → 字段」，用于判断两个档案是否真是同一份映射。 */
function 映射签名(mapping: Record<string, string>): string {
  return JSON.stringify(
    Object.entries(mapping)
      .map(([表头, 字段]) => [表头.trim(), 字段] as const)
      .sort(([甲], [乙]) => (甲 < 乙 ? -1 : 甲 > 乙 ? 1 : 0)),
  )
}

export function 匹配数据源档案(输入: {
  表头: readonly (string | null)[]
  档案们: readonly 档案候选[]
}): 档案命中 | null {
  // 表头重名时取第一次出现的列：档案按表头原文建，重名表头在建档时已被拒绝；
  // 这里取第一个只是让「库里有脏数据」时行为确定，不是支持重名。
  const 下标 = new Map<string, number>()
  输入.表头.forEach((原始, i) => {
    const 表头 = 原始?.trim() ?? ""
    if (表头 !== "" && !下标.has(表头)) 下标.set(表头, i)
  })

  const 合格: 合格候选[] = []
  for (const 档案 of 输入.档案们) {
    const columns: Record<number, 识别字段> = {}
    let 不合格 = false

    for (const [原始表头, 字段] of Object.entries(档案.mapping)) {
      const 表头 = 原始表头.trim()
      const 列号 = 下标.get(表头)
      // 规则 1：档案里的每个表头都必须在当前表里；缺一列不套。
      if (列号 === undefined) {
        不合格 = true
        break
      }
      // 规则 2 后半：失效字段 / 已删字段 ⇒ 整条档案不命中（设计 §2）。
      if (!是识别字段(字段)) {
        不合格 = true
        break
      }
      // 两个不同的映射键 trim 后落到同一列：说不清哪一个是真，fail-closed。
      const 已有 = columns[列号]
      if (已有 !== undefined && 已有 !== 字段) {
        不合格 = true
        break
      }
      columns[列号] = 字段
    }
    if (不合格) continue

    const 有效列数 = Object.keys(columns).length
    // 规则 2 前半：有效字段 ≥ 2。
    if (有效列数 < 2) continue
    // 规则 3：还原下标后重新判类型，防「资金档案套话单」。
    if (classifyMaterialType(columns) !== 档案.type) continue

    合格.push({ 档案, columns, 有效列数, 签名: 映射签名(档案.mapping) })
  }

  if (合格.length === 0) return null

  合格.sort(
    (甲, 乙) =>
      乙.有效列数 - 甲.有效列数 ||
      乙.档案.hitCount - 甲.档案.hitCount ||
      乙.档案.updatedAt - 甲.档案.updatedAt ||
      (甲.档案.id < 乙.档案.id ? -1 : 甲.档案.id > 乙.档案.id ? 1 : 0),
  )

  // 规则 5：前两名「列数与 hit_count 都相同、映射不同」⇒ 不猜。
  const 第一 = 合格[0]
  if (第一 === undefined) return null
  const 第二 = 合格[1]
  if (
    第二 !== undefined &&
    第一.有效列数 === 第二.有效列数 &&
    第一.档案.hitCount === 第二.档案.hitCount &&
    第一.签名 !== 第二.签名
  ) {
    return null
  }

  return { id: 第一.档案.id, name: 第一.档案.name, columns: 第一.columns }
}
```

/** jsonb 在 bun-sql 下可能是对象也可能是字符串；坏值一律当空，不抛（`#006-28`：catch 里要有动作）。 */
function 解析json(原始: unknown): unknown {
  if (typeof 原始 !== "string") return 原始
  try {
    return JSON.parse(原始)
  } catch {
    return undefined
  }
}

function 解析映射(原始: unknown): Record<string, string> {
  const 值 = 解析json(原始)
  if (typeof 值 !== "object" || 值 === null || Array.isArray(值)) return {}
  const 干净: Record<string, string> = {}
  for (const [k, v] of Object.entries(值)) if (typeof v === "string") 干净[k] = v
  return 干净
}

function 解析字符串数组(原始: unknown): string[] {
  const 值 = 解析json(原始)
  return Array.isArray(值) ? 值.filter((v): v is string => typeof v === "string") : []
}

export async function 列出数据源档案(

        from auth.case_source_profile
       where project_id = ${input.projectId}
         and (${input.含撤销 === true} or revoked_at is null)
       order by updated_at desc, id asc
    `),
  )
  const 结果: 数据源档案行[] = []
  for (const 行 of 行们) {

    if (type !== "资金" && type !== "话单" && type !== "其他") continue

interface 指纹项 {
  工作表: string
  指纹: string
}

function 解析指纹(原始: unknown): 指纹项[] {
  const 值 = 解析json(原始)
  if (!Array.isArray(值)) return []
  const 结果: 指纹项[] = []
  for (const 项 of 值) {
    if (typeof 项 !== "object" || 项 === null || Array.isArray(项)) continue
    const 表 = 项 as Record<string, unknown>
    if (typeof 表["工作表"] !== "string" || typeof 表["指纹"] !== "string") continue
    结果.push({ 工作表: 表["工作表"], 指纹: 表["指纹"] })
  }
  return 结果
}

/** 表头重合率：`交集 / max(两边)`。分母取大的，宁严勿宽。 */
function 重合率(甲: readonly string[], 乙: readonly string[]): number {
  const 甲集 = new Set(甲)
  const 乙集 = new Set(乙)
  let 交 = 0
  for (const 值 of 甲集) if (乙集.has(值)) 交 += 1
  return 交 / Math.max(甲集.size, 乙集.size)
}

function 同映射(甲: Record<string, string>, 乙: Record<string, string>): boolean {
  const 键 = (m: Record<string, string>) => Object.keys(m).sort()
  const 甲键 = 键(甲)
  const 乙键 = 键(乙)
  if (甲键.length !== 乙键.length) return false
  return 甲键.every((k, i) => k === 乙键[i] && 甲[k] === 乙[k])
}

    }

    const 指定 = input.工作表?.trim()
    if (指定 !== undefined && 指定 !== "" && !指纹们.some((项) => 项.工作表 === 指定)) {
      throw new 档案错误("sheet_not_found", `工作表不在这份材料的解析指纹里：${指定}`)
    }

    let 选中: { 工作表: string; 表头: (string | null)[]; columns: Record<number, 识别字段>; 类型: "资金" | "话单" | "其他" } | undefined
    let 见过其他 = false
    for (const 项 of 指纹们) {
      if (指定 !== undefined && 指定 !== "" && 项.工作表 !== 指定) continue
      const 缓存行 = rowsOf(
        await tx.execute(sql`
          select header, mapping from auth.case_table_format
           where project_id = ${input.projectId} and fingerprint = ${项.指纹}
        `),
      )[0]
      if (缓存行 === undefined) continue
      const columns = 解析缓存映射(解析json(缓存行.mapping))
      if (Object.keys(columns).length === 0) continue
      const 原始表头 = 解析json(缓存行.header)
      if (!Array.isArray(原始表头)) continue
      const 表头 = 原始表头.map((v) => (typeof v === "string" ? v : null))
      const 有效 = Object.values(columns).filter((f) => 是识别字段(f)).length
      const 类型 = classifyMaterialType(columns)
      if (有效 >= 2 && 类型 === "其他") {
        见过其他 = true
        continue
      }
      if (有效 < 2) continue
      选中 = { 工作表: 项.工作表, 表头, columns, 类型 }
      break
    }

    if (选中 === undefined) {
      throw new 档案错误(见过其他 ? "type_other" : "no_usable_sheet", "没有可建档的工作表（至少要 2 个有效字段，且不是「其他」）")
    }
    if (选中.类型 === "其他") throw new 档案错误("type_other", "认不出列的格式不固化成档案")

    const mapping: Record<string, string> = {}
    for (const [k, 字段] of Object.entries(选中.columns)) {
      if (!/^\d+$/.test(k)) continue
      if (!是识别字段(字段)) continue
      const 列号 = Number(k)
      if (!Number.isInteger(列号) || 列号 < 0 || 列号 >= 选中.表头.length) continue
      const 表头 = 选中.表头[列号]?.trim() ?? ""
      if (表头 === "") continue
      if (mapping[表头] !== undefined && mapping[表头] !== 字段) {
        throw new 档案错误("header_duplicate", `表头重名，不能建档案：${表头}`)
      }
      mapping[表头] = 字段
    }
    const headerSet = Object.keys(mapping).sort()
    if (headerSet.length < 2) throw new 档案错误("no_usable_sheet", "有效表头少于 2 列")

    const name = (input.name ?? "").trim() || `${String(材料行.file_name)} · ${选中.工作表}`

    const 活跃 = rowsOf(
      await tx.execute(sql`
        select id, name, header_set, mapping from auth.case_source_profile
         where project_id = ${input.projectId} and type = ${选中.类型} and revoked_at is null
         order by updated_at desc, id asc
      `),
    )
    for (const 行 of 活跃) {
      const 旧映射 = 解析映射(行.mapping)
      const 旧表头 = 解析字符串数组(行.header_set)
      if (同映射(旧映射, mapping) && JSON.stringify([...旧表头].sort()) === JSON.stringify(headerSet)) {
        return { id: String(行.id), created: false }
      }
    }

    let 最像: { id: string; name: string; 重合: number } | undefined
    for (const 行 of 活跃) {
      const 重合 = 重合率(解析字符串数组(行.header_set), headerSet)
      if (重合 >= 0.8 && (最像 === undefined || 重合 > 最像.重合)) {
        最像 = { id: String(行.id), name: String(行.name), 重合 }
      }
    }

    const now = nowSeconds()
    if (最像 !== undefined) {
      if (input.覆盖 !== true) {
        throw new 档案错误("conflict", "同类型且表头高度重合的档案已存在但映射不同；确认要覆盖请传 覆盖=true")
      }
      // 覆盖只换映射；没显式给 name 就保留原名，别把用户改过的名字悄悄重置成默认值。
      const 覆盖后名 = (input.name ?? "").trim() || 最像.name
      await tx.execute(sql`
        update auth.case_source_profile
           set name = ${覆盖后名}, header_set = ${JSON.stringify(headerSet)}::text::jsonb,
               mapping = ${JSON.stringify(mapping)}::text::jsonb, updated_at = ${now}
         where project_id = ${input.projectId} and id = ${最像.id} and revoked_at is null
      `)
      return { id: 最像.id, created: false }
    }

    const id = randomUUID()
    await tx.execute(sql`
      insert into auth.case_source_profile
        (id, project_id, name, type, header_set, mapping, hit_count, created_at, updated_at)
      values (${id}, ${input.projectId}, ${name}, ${选中.类型},
              ${JSON.stringify(headerSet)}::text::jsonb, ${JSON.stringify(mapping)}::text::jsonb,
              0, ${now}, ${now})
    `)
    return { id, created: true }
  })
}

import {
  列出数据源档案,
  建数据源档案,
  撤销数据源档案,

export async function list_source_profiles(
  ctx: 工具上下文,
  args: { 含撤销?: boolean },
): Promise<工具结果> {
  const list = await 列出数据源档案(ctx.db, {
    projectId: ctx.身份.projectId,
    含撤销: args.含撤销 === true,
  })
  return {
    摘要: `本项目共有 ${list.length} 个数据源档案`,
    数据: { list },
    口径: args.含撤销 === true ? "含已撤销" : "只看未撤销",
    缺口: [],
    空原因: list.length === 0 ? "对象未登记" : null,
    出处: [],
    下一步: [],
  }
}

export async function build_source_profile(
  ctx: 工具上下文,
  args: { 材料: string; 工作表?: string; 名称?: string; 覆盖?: boolean },
): Promise<工具结果> {
  try {
    const 回执 = await 建数据源档案(ctx.db, {
      projectId: ctx.身份.projectId,
      materialId: args.材料,
      ...(args.工作表 === undefined ? {} : { 工作表: args.工作表 }),
      ...(args.名称 === undefined ? {} : { name: args.名称 }),
      ...(args.覆盖 === undefined ? {} : { 覆盖: args.覆盖 }),
    })
    return {
      摘要: 回执.created ? "已建立数据源档案" : "已复用或更新数据源档案",
      数据: 回执,
      口径: "从该材料的 format_fingerprints 读回表头与映射；只建本项目档案",
      缺口: [],
      空原因: null,
      出处: [],
      下一步: [],
    }
  } catch (错误) {
    if (错误 instanceof 档案错误) {
      throw new 工具错误(错误.code === "material_not_found" ? "not_found" : "bad_input", 错误.message)
    }
    throw 错误
  }
}

export async function revoke_source_profile(
  ctx: 工具上下文,
  args: { 档案: string },
): Promise<工具结果> {
  const 撤销 = await 撤销数据源档案(ctx.db, { projectId: ctx.身份.projectId, id: args.档案 })
  if (!撤销) throw new 工具错误("not_found", "没有这个档案或已撤销")
  return {
    摘要: "已撤销数据源档案",
    数据: { ok: true },
    口径: "只撤销本项目档案",
    缺口: [],
    空原因: null,
    出处: [],
    下一步: [],
  }
}
