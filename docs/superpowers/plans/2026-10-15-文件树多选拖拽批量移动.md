**Goal:** 在项目管理的左栏「文件」Tab 里，选定多个同级文件/文件夹后拖到一个目标文件夹上松手，即把这一批移过去；失败了用弹窗把「成了的 / 没成的 ＋ 原因」一次说清。

```ts
/**
 * 同树拖拽的落点判据（纯函数：无 DOM、无信号、零依赖）。
 *
 * ## 为什么单独一个文件
 *
 * 它不读组件状态、也不碰 DOM，却要能被单测直接钉住；`file-tree-model.ts` 是 005「空目录那一支」
 * 的专用模块（单导出 `buildProjectFileTreeModel`），把拖拽判据塞进去会污染它的职责
 * （`docs/superpowers/specs/2026-10-15-file-tree-multi-select-move-design.md` §4 勘误）。
 * 新文件对上游**零冲突面**（第一号约束）。
 *
 * ## 路径写法
 *
 * 与仓里其余各处同一套：`/` 分隔、无前后斜杠、**项目根是 `""`**（同 `file-tree.tsx` 的
 * `落点()`）。不另立一套切分规则（`LEARNINGS #006-23`）。
 */

/** 路径它**现在**住在哪个目录里。顶层 ⇒ `""`（项目根）。 */
const 父目录 = (path: string) => {
  const index = path.lastIndexOf("/")
  return index === -1 ? "" : path.slice(0, index)
}

/**
 * 这一批源能不能落到 `目标` 目录上。三条不许：
 *
 * ① 目标是**某个源自己**（拖到自己身上）
 * ② 目标以「某个源 ＋ `/`」开头（拖进**自己的子孙**）
 * ③ **源现在就住在目标里**（`父目录(源) === 目标`）——搬过去等于什么都没发生
 *
 * 多个源里**有一个不合法 ⇒ 整体不行**：一次手势在磁盘上只该发生一件事，
 * 「三个里两个搬进了第三个」几乎不是任何人的本意（spec §9 取舍 5）。
 *
 * ⚠️ ③ 写的是 `父目录(源)` 而**不是** `落点(源)`：`落点()` 对**目录**回它自己（它要回答的是
 * 「往它这儿建东西该建在哪」），于是「一个顶层目录拖到项目根」这种「它本来就在那儿」的情形
 * 会**漏掉**。`父目录` 对文件与目录是同一条规则（都是 dirname），恰是 ③ 要说的事
 * （spec §5.2 写的是 `落点(源) === 目标`，此处按它那句话的**意思**修正常见的读法——spec 里
 * 要的是「它现在就在那儿」，不是「落点等于目标」）。
 */
export function 能落到(源们: readonly string[], 目标: string): boolean {
  if (源们.length === 0) return false
  for (const 源 of 源们) {
    if (目标 === 源) return false
    if (目标.startsWith(`${源}/`)) return false
    if (父目录(源) === 目标) return false
  }
  return true
}
```

```ts
/**
 * 挂一棵树并把深层行**全展开**。
 *
 * ⚠️ 树默认**全收缩**（用户第 5 条，⑦-5）⇒ `资料/a.txt` 这类行压根不在树上，`行按名` 会**抛**
 * （那正是它的用处，`#005-38`）。凡引用深层行的用例都得自己立这一步前提——本文件 `行按名`
 * 上面那条规矩就是这么写的。**不许靠静默兜底蒙过去。**
 *
 * ⚠️ 展开**不改选中集**——本次新增的东西不掺进这一步里。
 */
function 挂全开(paths: readonly string[]) {
  const host = mount(() => <FileTree paths={paths} />)
  动作(host, "expand-all")?.click()
  return host
}

/**
 * 按**路径**取一行（不是按名字）。找不到就**抛**。
 *
 * 这一组用例一律用路径：`data-path` 是契约本身那个值（`file-tree.tsx` 的 `落点()` 与真栈定位器
 * 都用它），而名字在同一棵树里可能重名、同层顺序又随 locale 变（`file-tree-v2-model` 用
 * `localeCompare`）⇒ 按名找会写出「只在我这台机器上对」的断言。
 */
function 行按路径(host: HTMLElement, path: string) {
  const 找到 = 行(host).find((el) => el.getAttribute("data-path") === path)
  if (!找到) throw new Error(`这一行不在树上：找不到 ${path}，此刻可见的是 ${JSON.stringify(路径(host))}`)
  return 找到
}

/**
 * 点树里某一行，带修饰键。
 *
 * ⚠️ 走**原生派发**而不是 `.click()`：`.click()` 交不出 `ctrlKey` / `shiftKey`
 * （`MouseEvent` 的构造参数在 `HTMLElement.click()` 里是写死的全 `false`）。
 */
function 点行(host: HTMLElement, path: string, 修饰: { ctrl?: boolean; shift?: boolean } = {}) {
  行按路径(host, path).dispatchEvent(
    new MouseEvent("click", {
      bubbles: true,
      cancelable: true,
      ctrlKey: 修饰.ctrl === true,
      shiftKey: 修饰.shift === true,
    }),
  )
}

/** 当前可见行的路径（滤掉 `null`，好让断言下标拿到的是 `string`）。 */
const 可见序 = (host: HTMLElement) => 路径(host).filter((p): p is string => p !== null)
```

在 `落点` 函数**之前**（同一层作用域，`return (` 之上）插入五个函数：

```tsx
  /**
   * 单选：焦点行、选中集、锚点三样一起落到它身上。
   *
   * 单击与键盘回车/空格**共用这一份**——同一个动作不许两处各写一份（`LEARNINGS #002-06`）。
   */
  const 单选 = (path: string) => {
    setSelected(path)
    set选中集(new Set([path]))
    set锚点(path)
  }

  /** Ctrl/⌘ 加选与取消：只拨这一行的成员资格。焦点行跟着走，**锚点不动**。 */
  const 拨一位 = (path: string) => {
    const 下一份 = new Set(选中集())
    if (!下一份.delete(path)) 下一份.add(path)
    set选中集(下一份)
  }

  /**
   * `Shift` 连选：锚点到它之间**可见的**行（含两端）。
   *
   * 范围按 `行列表()` 取——**折叠着的子孙不在里面**，正是用户眼睛看到的那一段。
   * 没有锚点时拿焦点行当锚点（并把它**升成**锚点，好让第二次 Shift 还连得上）；连焦点行也没有
   * （刚挂载就被 Shift 点）就退化成单选。
   */
  const 连选到 = (path: string) => {
    const 序 = 行列表().map((row) => row.node.path)
    const 起 = 锚点() ?? selected()
    const 头 = 起 === undefined ? -1 : 序.indexOf(起)
    const 尾 = 序.indexOf(path)
    if (头 === -1 || 尾 === -1) {
      单选(path)
      return
    }
    const [低, 高] = 头 <= 尾 ? [头, 尾] : [尾, 头]
    set选中集(new Set(序.slice(低, 高 + 1)))
    set锚点(起)
  }

  /** 行被点了一下：按修饰键分流。三种手势都先把焦点行落到它身上。 */
  const 点行 = (path: string, event: MouseEvent) => {
    if (event.shiftKey) {
      setSelected(path)
      连选到(path)
      return
    }
    if (event.ctrlKey || event.metaKey) {
      setSelected(path)
      拨一位(path)
      return
    }
    单选(path)
  }

  /**
   * 点在**行以外**的空白上：清空选中集与锚点，焦点行**不动**。
   *
   * 判据用 `指针行`（与右键共用那一份，`LEARNINGS #002-06`）而不是「点到谁身上」：行上的点击
   * 会冒泡到这一层，靠 `指针行` 才分得开「点在行上」与「点在行之间的空白」。
   */
  const 点空白 = (event: MouseEvent) => {
    if (指针行(event) !== undefined) return
    set选中集(new Set())
    set锚点(undefined)
  }
```

```ts
/**
 * 派一次**同树**拖拽（`dragstart` / `dragover` / `drop` / `dragend`）。
 *
 * 与 `拖入` 分开：那条造的是**跨进程**拖拽（`dataTransfer.types` 含 `"Files"`，判据读 `files`），
 * 这条走的是同树那一支——同树判据**不读 payload**（只认组件自己的信号），所以连 `dataTransfer`
 * 都不必挂。产品码对 `dataTransfer` 一律可选链，正是为了这里进得来。
 *
 * 返回造出来的事件：`dragover` 那一条要判 `defaultPrevented`（不 `preventDefault` 就不会有 `drop`）。
 */
function 同树拖(el: HTMLElement | null | undefined, type: "dragstart" | "dragover" | "drop" | "dragend") {
  if (!el) throw new Error("拖拽落点不在——这条用例的前提不成立（`#004-14`：先立前提再判果）")
  const event = new Event(type, { bubbles: true, cancelable: true })
  el.dispatchEvent(event)
  return event
}

/**
 * 挂一棵**全展开**的树（同 Task 2 的 `挂全开`），并记下 `onMoveTo` 收到的每一批。
 *
 * ⚠️ **必须全展开**：树默认全收缩（用户第 5 条）⇒ `资料/a.txt` 这类深层行压根不在树上，
 * `行按路径` 会**抛**。**不许靠静默兜底蒙过去**（`#005-38`）。
 */
function 挂可搬(paths: readonly string[]) {
  const 收到: Array<[readonly string[], string]> = []
  const host = mount(() => <FileTree paths={paths} draggable onMoveTo={(路径们, dir) => 收到.push([路径们, dir])} />)
  动作(host, "expand-all")?.click()
  return { host, 收到 }
}

/** **一次完整手势**：在某行起拖，落到某行（或 region）松手。 */
function 拖到(host: HTMLElement, 起路径: string, 落点路径: string | undefined) {
  同树拖(行按路径(host, 起路径), "dragstart")
  const 落 = 落点路径 === undefined ? 槽(host, "file-tree-region") : 行按路径(host, 落点路径)
  同树拖(落, "dragover")
  同树拖(落, "drop")
  同树拖(行按路径(host, 起路径), "dragend")
}
```

```tsx
  /**
   * 悬停：**不 `preventDefault` 就不会触发 `drop`**（HTML5 拖拽的规矩，不是可选的优化）。
   *
   * ⚠️ **桌面那一支必须判在最前面**：它的判据是 `dataTransfer.types` 含 `"Files"`，与同树分支
   * 互斥，但顺序写反会让「从桌面拖文件进来」被同树分支吞掉（spec §5.2）。
   */
  function 拖过(event: DragEvent) {
    if (从桌面(event)) {
      event.preventDefault()
      return
    }
    const 源 = 拖源()
    if (!源) return
    const 行 = 指针行(event)
    if (!能落到(源, 落点(行))) {
      清悬停()
      return
    }
    event.preventDefault()
    set悬停行(行)
    set悬停根(行 === undefined)
  }

  function 放开(event: DragEvent) {
    if (从桌面(event)) {
      event.preventDefault()
      const files = Array.from(event.dataTransfer?.files ?? [])
      if (files.length === 0) return
      props.onDropFiles?.(files, 落点(指针行(event)))
      return
    }
    const 源 = 拖源()
    if (!源) return
    event.preventDefault()
    // 落点按**松手这一刻**重算（不看悬停时记下的东西）：拖拽期间树可能被重取 / 折叠过。
    const 目标 = 落点(指针行(event))
    set拖源(undefined)
    清悬停()
    if (!能落到(源, 目标)) return
    // 移动之后清空选中集（spec §5.3 / §9 取舍 6）：那些路径多半已经不在原处，不清的话第二次
    // 拖拽会拿一批失效的路径去发请求。⚠️ **在这里清**而不是等结果回来——信号长在本组件里，
    // 接收方拿不到它；而「等结果」要一条反向通道，正是取舍 6 明确拒绝的那条。
    set选中集(new Set())
    props.onMoveTo?.(源, 目标)
  }
```

```tsx
   *
```

```tsx
    if (!id || !projectData) return
    if (搬忙()) return

      return
    }
    set文件话(undefined)
    // 有成了的就得重取（盘上真变了）；全败**不重取**——磁盘一点没变，重取是空跑。
```

```ts
import { expect, test, type BrowserContext, type Page } from "@playwright/test"
import { existsSync, mkdirSync, readdirSync, readFileSync, writeFileSync } from "fs"
import { tmpdir } from "os"
import path from "path"

/**
 * 左栏文件树：多选后拖到目标文件夹松手 ⇒ **磁盘上真的搬过去了**。
 *
 * ## 为什么这条非得在真栈
 *
 * 组件层那份判据是**裸 `Event` 顶的**（happy-dom 没有 `DragEvent`）⇒ 它只能证明「事件到了，
 * 我的代码做了什么」，证明不了**浏览器会为可放目标触发 `drop`**。判据 30 / 31 才是那条链的牙。
 *
 * ## 拖拽走裸指针，不走 `locator.dragTo`
 *
 * 同 `send-to-session-real.spec.ts`（那份用 `page.mouse` 实测到了 `dragstart→dragover→drop`）
 * 与 `center-pane-space-real.spec.ts`（`#005-30`：`dragTo` 会卡在 actionability 上，
 * 量到的是 Playwright 的规矩、不是产品的行为）。
 *
 * ## 落盘判据读**真目录**
 *
 * `readdirSync` / `existsSync`（同 `file-tree-delete-dialog-real.spec.ts`）——界面说搬好了
 * 不算数，盘上真在那儿才算。
 */

const readStack = () =>
  JSON.parse(readFileSync(path.join(tmpdir(), "openhive-real-stack.json"), "utf8")) as {
    kernel: number
    front: number
    token: string
  }

const FRONT = () => `http://127.0.0.1:${readStack().front}`
const KERNEL = () => `http://127.0.0.1:${readStack().kernel}`
const COOKIE = () => `openhive_session=${readStack().token}`

async function kernelJson(url: string, init: RequestInit = {}) {
  const res = await fetch(`${KERNEL()}${url}`, {
    ...init,
    // ⚠️ 身份走 **cookie**，不是 `Authorization: Bearer`（Bearer 拿到的是**没有响应体**的 401）。
    headers: { Cookie: COOKIE(), ...(init.headers ?? {}) },
  })
  return { status: res.status, body: res.ok ? await res.json() : await res.text() }
}

async function 建真项目(名: string): Promise<{ id: string; directory: string }> {
  const 建 = await kernelJson("/openhive/project", {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ name: 名, type: "private" }),
  })
  expect(建.status, "真内核建项目应成功").toBe(200)
  const id = (建.body as { id?: string } | undefined)?.id
  expect(id, "建项目应当回一个服务端给的 id").toBeTruthy()

  const 列 = await kernelJson("/openhive/project")
  const 行 = (列.body as { id: string; directory?: string }[] | undefined)?.find((行) => 行.id === id)
  expect(行?.directory, "列项目应当给出这一行的 directory").toBeTruthy()
  return { id: id!, directory: 行!.directory! }
}

async function 进项目(page: Page, context: BrowserContext, id: string) {
  await context.addCookies([
    { name: "openhive_session", value: readStack().token, url: FRONT() },
    { name: "openhive_project", value: id, url: FRONT() },
  ])
  await page.addInitScript(() => {
    localStorage.setItem(
      "settings.v3",
      JSON.stringify({ general: { newLayoutDesigns: true, shouldDisplayTabsToast: false } }),
    )
  })
  await page.goto("/")
  await page.locator("[data-slot='sidebar-tab'][data-tab='files']").click()
}

/** 树上一行。 */
const 行 = (page: Page, 路径: string) =>
  page.locator(`[data-slot='sandbox-tree'] [data-slot='file-tree-row'][data-path='${路径}']`)

/** 裸指针把一行拖到另一行上松手（`#005-30`：不走 `locator.dragTo`）。 */
async function 拖行到行(page: Page, 起: string, 落: string) {
  const 起盒 = await 行(page, 起).boundingBox()
  const 落盒 = await 行(page, 落).boundingBox()
  if (!起盒 || !落盒) throw new Error("行没有盒——这条用例的前提不成立")
  await page.mouse.move(起盒.x + 起盒.width / 2, 起盒.y + 起盒.height / 2)
  await page.mouse.down()
  // 多步移动：Chromium 要看到一次真正的位移才起拖（同 send-to-session-real.spec.ts 的 30 步）。
  await page.mouse.move(落盒.x + 落盒.width / 2, 落盒.y + 落盒.height / 2, { steps: 30 })
  await page.mouse.up()
  // 落盘是异步的（请求 + 重取清单），给足时间再读盘。
  await page.waitForTimeout(2000)
}

function 种盘(directory: string) {
  mkdirSync(path.join(directory, "资料"), { recursive: true })
  writeFileSync(path.join(directory, "资料", "甲.csv"), "甲")
  writeFileSync(path.join(directory, "资料", "乙.csv"), "乙")
  mkdirSync(path.join(directory, "档案"), { recursive: true })
}

test("判据 30：拖两行到一个文件夹上松手 ⇒ 磁盘上那两个真的在那文件夹里", async ({ page, context }) => {
  const { id, directory } = await 建真项目("多选拖拽-成功")
  // ⚠️ 种盘要在 `进项目` **之前**：树的清单是首屏取的，种晚了这一条就看不出东西。
  种盘(directory)
  await 进项目(page, context, id)

  // 前置：两行都在（证明这一条的前提活着，`#002-02`）
  await expect(行(page, "资料/甲.csv"), "种下去的文件应当在树上").toBeVisible()
  await expect(行(page, "档案"), "目标文件夹应当在树上").toBeVisible()

  await 行(page, "资料/甲.csv").click()
  await 行(page, "资料/乙.csv").click({ modifiers: ["Control"] })
  拖行到行(page, "资料/甲.csv", "档案")

  const 磁盘 = readdirSync(path.join(directory, "档案"))
  console.log(`[多选拖拽] 档案/ = ${JSON.stringify(磁盘)}`)
  expect(磁盘, "甲应当真的搬进档案了").toContain("甲.csv")
  expect(磁盘, "乙应当真的搬进档案了").toContain("乙.csv")
  expect(readdirSync(path.join(directory, "资料")), "原处不该还留着").toEqual([])
  expect(existsSync(path.join(directory, "资料", "甲.csv")), "旧位置必须真的没了").toBe(false)
})

test("判据 31：拖到源自己身上 ⇒ 磁盘上一个都没动", async ({ page, context }) => {
  const { id, directory } = await 建真项目("多选拖拽-非法")
  种盘(directory)
  await 进项目(page, context, id)

  const 盘前 = readdirSync(path.join(directory, "资料"))
  拖行到行(page, "资料", "资料")

  console.log(`[非法落点] 资料/ = ${JSON.stringify(readdirSync(path.join(directory, "资料")))}`)
  expect(readdirSync(path.join(directory, "资料")), "非法落点不该动磁盘").toEqual(盘前)
  expect(existsSync(path.join(directory, "资料", "甲.csv")), "文件必须还在原处").toBe(true)
})
```
