> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让民警在会话里直接看到 MCP 查库工具返回的七字段信封（摘要优先、缺口可展开、原始 JSON 折叠），而不是现在的 `GenericTool`（只有调用名与入参）。

**Architecture:** 三层。① `envelope-parse.ts` 纯函数解析层（可 TDD）；② `envelope-tool.tsx` 渲染层（复用 `BasicTool` 外壳，未知形状一律回退 `GenericTool`）；③ 接线只改 `message-part.tsx` 的 `render` memo **一行**，`openhive-case_` 前缀判断留在我们自己的文件里 ⇒ 以后新增工具自动有卡，不维护第二份工具清单（`#002-06`）。

**Tech Stack:** SolidJS 1.9 + Tailwind v4 类（`bg-surface-warning-weak` / `border-border-warning-base` / `text-text-weak`）+ `BasicTool`（`packages/session-ui/src/components/basic-tool.tsx`）+ i18n（`packages/ui/src/i18n/{en,zh}.ts`）+ storybook（`packages/storybook`，已 glob `session-ui/src/**/*.stories.tsx`）+ 真 Chromium axe 审计（`packages/app/e2e/a11y/storybook-axe-audit.ts`）。

**Spec:** `docs/superpowers/specs/2026-11-09-事实卡片抽象/09-AI探索与MCP工具-设计.md` §3.1–§3.5（七字段与空原因）、§12 #9（工具卡落点）、§5.3（`下一步` 是给人的建议）。

/**
 * 七字段信封的**解析层**（`09` §3.1–§3.5）。
 *
 * 只做「把工具返回的文本认成什么」——不碰 DOM、不依赖 Solid，纯函数才能被单测钉住
 * （session-ui 没有组件测试基建，渲染层的验收走 storybook + 真浏览器，见计划 Task 4）。
 * 认不出来一律 `非信封`：调用方回退 `GenericTool`，未知形状绝不能炸掉会话渲染。
 */

export interface 出处项 {
  readonly 事实: string
  readonly 代表: string | null
  readonly 可下钻: string | null
}

export interface 下一步项 {
  readonly 工具: string
  readonly 参数: Record<string, unknown>
  readonly 为什么: string
}

export interface 信封 {
  readonly 摘要: string
  readonly 数据: unknown
  readonly 口径: string
  readonly 缺口: readonly string[]
  readonly 空原因: string | null
  readonly 出处: readonly 出处项[]
  readonly 下一步: readonly 下一步项[]
}

export type 解析结果 =
  | { readonly 种类: "信封"; readonly 信封: 信封 }
  | { readonly 种类: "工具错误"; readonly 错误码: string; readonly 消息: string }
  | { readonly 种类: "非信封" }

/** 信封工具在 part 里的键前缀：`McpCatalog.toolName("openhive-case", 工具名)`。 */
const 信封工具前缀 = "openhive-case_"

export function 是信封工具键(工具键: string): boolean {
  return 工具键.startsWith(信封工具前缀) && 工具键.length > 信封工具前缀.length
}

const 是记录 = (值: unknown): 值 is Record<string, unknown> =>
  typeof 值 === "object" && 值 !== null && !Array.isArray(值)

const 字符串数组 = (值: unknown): string[] =>
  Array.isArray(值) ? 值.filter((条): 条 is string => typeof 条 === "string") : []

const 出处数组 = (值: unknown): 出处项[] =>
  Array.isArray(值)
    ? 值.filter(是记录).flatMap((条) => {
        if (typeof 条.事实 !== "string") return []
        return [
          {
            事实: 条.事实,
            代表: typeof 条.代表 === "string" ? 条.代表 : null,
            可下钻: typeof 条.可下钻 === "string" ? 条.可下钻 : null,
          },
        ]
      })
    : []

const 下一步数组 = (值: unknown): 下一步项[] =>
  Array.isArray(值)
    ? 值.filter(是记录).flatMap((条) => {
        if (typeof 条.工具 !== "string") return []
        return [
          {
            工具: 条.工具,
            参数: 是记录(条.参数) ? 条.参数 : {},
            为什么: typeof 条.为什么 === "string" ? 条.为什么 : "",
          },
        ]
      })
    : []

export function 解析工具输出(文本: unknown): 解析结果 {
  if (typeof 文本 !== "string" || 文本.trim() === "") return { 种类: "非信封" }
  let 值: unknown
  try {
    值 = JSON.parse(文本)
  } catch {
    return { 种类: "非信封" }
  }
  if (!是记录(值)) return { 种类: "非信封" }
  // 失败结果先认：`{ error, message }`（服务端失败结果，part 状态仍是 completed）。
  if (typeof 值.error === "string")
    return { 种类: "工具错误", 错误码: 值.error, 消息: typeof 值.message === "string" ? 值.message : "" }
  // 摘要 是信封的判别字段：没有它就不是信封（缺其它字段只降级，不整体判否）。
  if (typeof 值.摘要 !== "string") return { 种类: "非信封" }
  return {
    种类: "信封",
    信封: {
      摘要: 值.摘要,
      数据: 值.数据 === undefined ? null : 值.数据,
      口径: typeof 值.口径 === "string" ? 值.口径 : "",
      缺口: 字符串数组(值.缺口),
      空原因: typeof 值.空原因 === "string" ? 值.空原因 : null,
      出处: 出处数组(值.出处),
      下一步: 下一步数组(值.下一步),
    },
  }
}

/** `数据` 是数组时给行数；对象 / null 给 `null`（汇总类工具的对象没有「行」）。 */
export function 数据行数(数据: unknown): number | null {
  return Array.isArray(数据) ? 数据.length : null
}

/** 与 `09` §3.4 同口径：`Buffer.byteLength` 的浏览器等价物是 UTF-8 编码长度。 */
export function 数据字节数(数据: unknown): number {
  if (数据 === undefined || 数据 === null) return 0
  return new TextEncoder().encode(JSON.stringify(数据)).length
}

import { For, Show, createMemo } from "solid-js"
import { useI18n } from "@opencode-ai/ui/context/i18n"
import { BasicTool, GenericTool } from "./basic-tool"
import { 数据字节数, 数据行数, 解析工具输出, 是信封工具键 } from "./envelope-parse"

/** 渲染器收到的 props（`message-part.tsx` 的 `Dynamic` 传参）。 */
export interface 信封卡属性 {
  tool: string
  input?: Record<string, unknown>
  output?: string
  status?: string
  hideDetails?: boolean
  defaultOpen?: boolean
  open?: boolean
  onOpenChange?: (open: boolean) => void
}

/** `ui.envelopeTool.tool.<工具名>` 有译文就用；没有时 i18n 回退返回 key 本身（context/i18n.tsx:53），这时显示原名。 */
const 工具名文本 = (t: (键: string) => string, 工具: string) => {
  const 键 = `ui.envelopeTool.tool.${工具}`
  const 文本 = t(键)
  return 文本 === 键 ? 工具 : 文本
}

/** 触发器副标题：只挑几个民警看得懂的入参，不是整包 JSON。 */
function 入参摘要(input: Record<string, unknown> | undefined): string[] {
  if (!input) return []
  const 键序 = ["材料", "账户", "实体", "名称", "类型", "方向", "金额起", "金额止"]
  return 键序
    .filter((键) => input[键] !== undefined && input[键] !== null && input[键] !== "")
    .slice(0, 3)
    .map((键) => `${键}=${String(input[键])}`)
}

export function EnvelopeTool(props: 信封卡属性) {
  const i18n = useI18n()
  const 解析 = createMemo(() => 解析工具输出(props.output))
  const 信封数据 = () => {
    const 结果 = 解析()
    return 结果.种类 === "信封" ? 结果.信封 : undefined
  }

  // 错误态不在这里：`status:"error"` 的 part 由 `message-part.tsx` 的 Match 先接走（`信封错误卡`），
  // `Dynamic` 只会在非 error 时渲染本组件，此时 `props.output` 一定有值。
  return (
    <Show
      when={信封数据()}
      fallback={
        <GenericTool tool={props.tool} status={props.status} input={props.input} hideDetails={props.hideDetails} />
      }
    >
      {(信封) => (
          <BasicTool
            icon="mcp"
            status={props.status}
            hideDetails={props.hideDetails}
            defaultOpen={props.defaultOpen}
            open={props.open}
            onOpenChange={props.onOpenChange}
            trigger={{
              title: 信封().摘要,
              subtitle: 工具名文本(i18n.t, props.tool),
              args: 入参摘要(props.input),
            }}
          >
            <div data-component="envelope-tool" class="flex flex-col gap-2 px-1 pb-1">
              <Show when={信封().口径}>
                {(口径) => (
                  <div data-slot="envelope-scope" class="text-12-regular text-text-weak">
                    {i18n.t("ui.envelopeTool.scope")}：{口径()}
                  </div>
                )}
              </Show>

              <Show when={信封().空原因}>
                {(空原因) => (
                  <div
                    data-slot="envelope-empty"
                    class="rounded-[6px] border border-border-warning-base bg-surface-warning-weak px-2 py-1.5 text-13-regular text-text-base"
                  >
                    <div>
                      {i18n.t("ui.envelopeTool.emptyReason")}：{空原因()}
                    </div>
                    <Show when={信封().下一步[0]?.为什么}>
                      {(为什么) => <div class="mt-0.5 text-12-regular text-text-weak">{为什么()}</div>}
                    </Show>
                  </div>
                )}
              </Show>

              <Show when={信封().缺口.length > 0}>
                <div
                  data-slot="envelope-gaps"
                  class="rounded-[6px] border border-border-warning-base bg-surface-warning-weak px-2 py-1.5"
                >
                  <div class="text-12-regular text-text-weak">{i18n.t("ui.envelopeTool.gap")}</div>
                  <ul class="mt-0.5 flex list-disc flex-col gap-0.5 pl-4 text-13-regular text-text-base">
                    <For each={信封().缺口}>{(缺口) => <li>{缺口}</li>}</For>
                  </ul>
                </div>
              </Show>

              <div data-slot="envelope-data" class="text-12-regular text-text-weak">
                {数据行数(信封().数据) === null
                  ? i18n.t("ui.envelopeTool.object")
                  : i18n.t("ui.envelopeTool.rows", { count: 数据行数(信封().数据) as number })}
                {" · "}
                {数据字节数(信封().数据)} B
              </div>

              <Show when={信封().出处.length > 0}>
                <div data-slot="envelope-sources">
                  <div class="text-12-regular text-text-weak">{i18n.t("ui.envelopeTool.source")}</div>
                  <ul class="mt-0.5 flex flex-col gap-0.5">
                    <For each={信封().出处}>
                      {(条) => (
                        <li class="text-13-regular text-text-base">
                          {条.事实}
                          <Show when={条.代表}>
                            {(代表) => <span class="ml-1 text-12-regular text-text-weak">{代表()}</span>}
                          </Show>
                        </li>
                      )}
                    </For>
                  </ul>
                </div>
              </Show>

              <Show when={信封().下一步.length > 0}>
                <div data-slot="envelope-next">
                  <div class="text-12-regular text-text-weak">{i18n.t("ui.envelopeTool.next")}</div>
                  <ul class="mt-0.5 flex flex-col gap-0.5">
                    <For each={信封().下一步}>
                      {(条) => (
                        <li class="text-13-regular text-text-base">
                          {工具名文本(i18n.t, 条.工具)}
                          <Show when={条.为什么}>
                            {(为什么) => <span class="ml-1 text-12-regular text-text-weak">{为什么()}</span>}
                          </Show>
                        </li>
                      )}
                    </For>
                  </ul>
                </div>
              </Show>

              {/* 原始 JSON 用原生 details：键盘可展开，不需要额外组件 API，也不与 BasicTool 的 Collapsible 嵌套 */}
              <details data-slot="envelope-raw">
                <summary class="cursor-pointer text-12-regular text-text-weak">{i18n.t("ui.envelopeTool.raw")}</summary>
                <pre class="mt-1 max-h-64 overflow-auto whitespace-pre-wrap break-all text-12-regular text-text-base">
                  {JSON.stringify(信封(), null, 2)}
                </pre>
              </details>
            </div>
          </BasicTool>
        )}
    </Show>
  )
}

export interface 错误卡属性 {
  tool: string
  /** `part().state.error`：服务端失败信封被 `convertTool` 原样 `throw` 成 Error 后的 message。 */
  文本: string
  status?: string
  defaultOpen?: boolean
  open?: boolean
  onOpenChange?: (open: boolean) => void
}

/**
 * 工具失败卡：接住 `status:"error"` 的 part（见计划「已核实的事实」里那条链路）。
 * 文本能解析成 `{error,message}` 就分「错误码 + 人话原因」显示；解析不了（传输层错误等）就原样显示。
 */
export function 信封错误卡(props: 错误卡属性) {
  const i18n = useI18n()
  const 解析 = createMemo(() => 解析工具输出(props.文本))
  const 错误 = () => {
    const 结果 = 解析()
    return 结果.种类 === "工具错误" ? 结果 : undefined
  }
  return (
    <BasicTool
      icon="warning"
      status={props.status}
      defaultOpen={props.defaultOpen}
      open={props.open}
      onOpenChange={props.onOpenChange}
      trigger={{ title: i18n.t("ui.envelopeTool.error"), subtitle: 工具名文本(i18n.t, props.tool) }}
    >
      <div
        data-slot="envelope-error"
        class="rounded-[6px] border border-border-warning-base bg-surface-warning-weak px-2 py-1.5 text-13-regular text-text-base"
      >
        <Show when={错误()}>
          {(值) => (
            <div>
              <span class="text-text-weak">{i18n.t("ui.envelopeTool.errorCode")}：</span>
              <span>{值().错误码}</span>
              <div class="mt-0.5 whitespace-pre-wrap">{值().消息}</div>
            </div>
          )}
        </Show>
        <Show when={!错误()}>
          <div class="whitespace-pre-wrap">{props.文本}</div>
        </Show>
      </div>
    </BasicTool>
  )
}

/** 接线用：`openhive-case_*` 返回信封卡，其余 `undefined`（交给 ToolRegistry / GenericTool）。 */
export function 信封渲染兜底(工具键: string) {
  return 是信封工具键(工具键) ? EnvelopeTool : undefined
}

## 计划的诚实缺口（开工前就该知道）

1. **没有自动化组件测试**：session-ui 无 `@solidjs/testing-library`，本计划不加依赖（会污染 `bun.lock`）。渲染正确性靠 stories + 真浏览器审计，CI 上**不会自动红**——这是一期明确接受的代价。
2. **`下一步` 不可点击**：只展示「工具 + 为什么」。点击需要构造续页调用并注入会话，属另一件事。
3. **字段展示是窄类型**：`envelope-parse.ts` 自己定义信封类型（窄集），不跨包 import `case-envelope.ts`（前后端 workspace 依赖方向反了）。服务端加字段时前端**不会编译报错**，只会显示不出来——`09` §12 #6 的「随各期同批补」是唯一护栏。
4. **`i18n` 只补 en/zh**：其它语言回退英文（`context/i18n.tsx:53`），符合现有惯例。
5. **未上真栈**：本计划只到 storybook（真 Chromium）。卡片在真实会话里的位置 / 滚动 / 虚拟化行为（`virtualizeDiff` / `deferContent`）要等真栈验收，属下一期。
