# 表格字段识别与明细补全 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: 用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务执行。步骤用 `- [ ]` 跟踪。

**Goal:** 把「表头文本 = 字段身份」改成「字段目录驱动的多路召回 + 人工确认的全局别名库」，让资金表真正落出对手账号/对手户名/本方账号/收付方向，并让话单、订餐/物流共用同一套识别。

**Architecture:** 新增三个纯逻辑模块（字段目录、模型表头识别、识别编排），改造 `detectMapping` 为「别名 → 词典精确 → 词典后缀 → 形状」四层并按候选字段集与修饰语约束，识别结果连同待确认清单一起进 `parse_detail.mapping` 与格式缓存；别名库是部署全局的两张元数据表，写别名与 `alias_version` 递增在同一事务，版本进指纹。

**Tech Stack:** Bun 1.4.2 monorepo、TypeScript（tsgo）、Effect 4 beta（`Schema` / `Effect`）、Drizzle raw `sql`、PG（真库测试走 `startProductionDb`）、SolidJS 1.9（app 侧）、oxlint。

**Spec:** `docs/superpowers/specs/2026-11-09-表格字段识别-design.md`

## Global Constraints

- **识别版本从 `"5"` 升到 `"6"`**（`case-mapping.ts:282`）。识别规则整轮重写，必须让旧指纹不可达。
- **指纹公式**：`sha256([识别版本, alias_version, 原始表头(trim), 每列形状])`；`formatFingerprint` 第三参数**必填**，所有调用点同步（`case-ingest.ts:350` 与全部测试）。
- **别名键必须是原始表头**（只 trim）；归一化只作用于写进标准列的值，**不进别名键、不进指纹**（`#011-01`）。
- **别名只提高召回、不提供证据**：命中后仍过单位闸与形状校验；不达标进待确认。
- **写别名与 `alias_version` +1 必须同一事务**；一个解析任务只读一次 `alias_version`，随任务传下。
- **权限只有 `owner` / `member` 两档**（`0005_project_member.sql`），没有「管理员」。
- **不新增 `parse_status` 枚举值**；待确认只进 `parse_detail.mapping`。
- **模型基础设施复用**：只用 `解析启动模型` + `实例上下文 getModel/getLanguage` + `建调用模型`；不新建 provider 解析、客户端、重试策略。

- **模型不能覆盖**规则结果或别名结果；模型字段名必须在候选字段集内，否则丢弃。

const 识别字段集: ReadonlySet<string> = new Set([...标准列字段们, ...extras字段们])

export function 是识别字段(值: unknown): 值 is 识别字段 {
  return typeof 值 === "string" && 识别字段集.has(值)
}

export function 是标准列(字段: 识别字段): 字段 is 标准列字段 {
  return 字段目录[字段].落点 === "列"
}

export function 归一化值(字段: 识别字段, 原值: string): string {
  const 归一化 = 字段目录[字段].归一化
  return 归一化 === undefined ? 原值.trim() : 归一化(原值)
}

function 列(rows: (string | null)[][], i: number): (string | null)[] {
  return rows.map((r) => r[i] ?? null)
}

export function 形状命中率(字段: 识别字段, 值们: (string | null)[]): number {
  const 形状 = 字段目录[字段].形状
  if (形状 === undefined) return 0
  const 非空 = 值们.filter((v): v is string => v !== null && v.trim() !== "")
  if (非空.length === 0) return 0
  return 非空.filter(形状).length / 非空.length
}

export type 表类型 = "资金" | "话单" | "其他"

/** 按「表头词投票优先、列形状兜底」推断表类型；只用于挑候选字段集，不改 `kind`。 */
export function 推断表类型(header: (string | null)[], rows: (string | null)[][]): 表类型 {
  const 宽 = Math.max(header.length, ...rows.map((r) => r.length), 0)
  let 话单分 = 0
  let 交易分 = 0
  let 号码列 = 0
  let 金额列 = 0
  for (let i = 0; i < 宽; i++) {
    const 值 = 列(rows, i)
    const 非空 = 值.filter((v) => v !== null && v.trim() !== "")
    const 名 = (header[i] ?? "").trim()
    if (话单词.test(名)) 话单分++
    if (交易词.test(名)) 交易分++
    // 少于 3 个非空值的列不投票：一两个格子撑不起一个类型判断。
    if (非空.length < 3) continue
    if (形状命中率("callee", 值) >= 0.9) 号码列++
    if (形状命中率("amount", 值) >= 0.9) 金额列++
  }
  if (话单分 > 交易分) return "话单"
  if (交易分 > 话单分) return "资金"
  if (金额列 > 0) return "资金"
  if (号码列 > 0) return "话单"
  return "其他"
}

/** 候选字段集：类型对应的一组；别的类型字段在本表里不参与认领。 */
export function 候选字段集(
  header: (string | null)[],
  rows: (string | null)[][],
): ReadonlySet<识别字段> {
  const 类型 = 推断表类型(header, rows)
  if (类型 === "资金")
    return new Set<识别字段>([
      "startedAt",
      "amount",
      "balance",
      "counterparty",
      "counterpartyName",
      "ownerAccount",
      "direction",
      "ownerName",
    ])
  if (类型 === "话单")
    return new Set<识别字段>(["startedAt", "caller", "callee", "durationSec", "personName"])
  return new Set<识别字段>(["startedAt", ...extras字段们])
}

export * as OpenhiveCaseFieldAlias from "./case-field-alias"

import { sql } from "drizzle-orm"
import { rowsOf } from "@opencode-ai/auth/migrate"
import { nowSeconds } from "@opencode-ai/auth/time"
import { 是识别字段 } from "./case-field-registry"
import { inTransaction, type Queryable } from "./case-store"

/**
 * 部署全局别名库（设计 §4）。
 *
 * 五条安全边界的落点：
 * - 键是**原始表头**（调用方传进来什么样就什么样，本模块只 trim）；
 * - 命中不提供证据，证据由识别层再判；
 * - 优先级由识别层的顺序保证，本模块不参与；
 * - 写入与 `alias_version` +1 **同一事务**；
 * - 字段名由服务端按字段目录校验（`是识别字段`）。
 */

export interface 别名上下文 {
  readonly 别名: ReadonlyMap<string, string>
  readonly 版本: number
}

/** 一个解析任务开始读一次，随任务传下；不要每张 sheet 各读一次（设计 §4.2-4）。 */
export async function 读别名上下文(db: Queryable): Promise<别名上下文> {
  const 行们 = rowsOf(
    await db.execute(sql`
      select header, field from auth.case_field_alias where revoked_at is null
    `),
  )
  const 别名 = new Map<string, string>()
  for (const 行 of 行们) 别名.set(String(行.header), String(行.field))

  const 版本行 = rowsOf(await db.execute(sql`select version from auth.case_field_alias_version where id = 1`))
  return { 别名, 版本: Number(版本行[0]?.version ?? 0) }
}

export interface 确认别名输入 {
  readonly header: string
  readonly field: string
  readonly materialId: string
  readonly projectId: string
  readonly by: string
}

export async function 确认别名(db: Queryable, 输入: 确认别名输入): Promise<void> {
  const header = 输入.header.trim()
  if (header === "" || header.length > 200) throw new Error("表头为空或超过 200 字，不能写别名")
  if (!是识别字段(输入.field)) throw new Error(`字段不在字段目录内：${输入.field}`)

  await inTransaction(db, async (tx) => {
    // 活跃别名存在 ⇒ 0 行 ⇒ 抛「已存在」；已撤销行 ⇒ 复活（R1）。
    const 行们 = rowsOf(
      await tx.execute(sql`
        insert into auth.case_field_alias
          (header, field, material_id, project_id, confirmed_by, confirmed_at, hit_count)
        values (${header}, ${输入.field}, ${输入.materialId}, ${输入.projectId}, ${输入.by}, ${nowSeconds()}, 0)
        on conflict (header) do update
          set field        = excluded.field,
              material_id  = excluded.material_id,
              project_id   = excluded.project_id,
              confirmed_by = excluded.confirmed_by,
              confirmed_at = excluded.confirmed_at,
              revoked_at   = null
          where auth.case_field_alias.revoked_at is not null
        returning header
      `),
    )
    if (行们.length === 0) throw new Error("该表头已有别名，请先撤销")
    await 递增版本(tx)
  })
}

export async function 撤销别名(db: Queryable, 输入: { header: string; by: string }): Promise<boolean> {
  return await inTransaction(db, async (tx) => {
    const 行们 = rowsOf(
      await tx.execute(sql`
        update auth.case_field_alias
           set revoked_at = ${nowSeconds()}
         where header = ${输入.header.trim()} and revoked_at is null
        returning header
      `),
    )
    if (行们.length === 0) return false
    await 递增版本(tx)
    return true
  })
}

/** 一份材料里同一个别名被多列用到也只 +1（设计 §4.3）。 */
export async function 记别名命中(db: Queryable, 表头们: readonly string[]): Promise<void> {
  const 去重 = [...new Set(表头们.map((h) => h.trim()).filter((h) => h !== ""))]
  if (去重.length === 0) return
  await db.execute(sql`
    update auth.case_field_alias
       set hit_count = hit_count + 1
     where revoked_at is null and header in (${sql.join(去重.map((h) => sql`${h}`), sql`, `)})
  `)
}

async function 递增版本(tx: Queryable): Promise<void> {
  await tx.execute(sql`update auth.case_field_alias_version set version = version + 1 where id = 1`)
}

```ts
export type 待确认原因 = "多列争抢" | "形状不符" | "别名跨字段集" | "候选不唯一" | "单位冲突" | "表头重名"

export interface 待确认项 {
  列号: number
  表头: string | null
  建议字段: 识别字段 | null
  原因: 待确认原因
}

export interface DetectOptions {
  readonly 别名?: ReadonlyMap<string, string>
  readonly 候选字段?: ReadonlySet<识别字段>
}

/** 来源联合现在就把 "模型" 写进来：Task 7 只填值、不改类型。 */
export type 识别来源 = "别名库" | "词典" | "特征" | "模型"

export interface MappingResult {
  columns: Record<number, 识别字段>
  method: "词典" | "特征" | "别名"
  来源: Record<number, 识别来源>
  待确认: 待确认项[]
  unknownColumns: number[]
}

const 对手修饰 = /对方|对手|交易对方|收方|付方/
const 本方修饰 = /本方|查询|客户|本人/
const 户名词尾 = /(名称|户名)$/
const 账号词尾 = /(账号|帐号|卡号|账户|帐号)$/

function 形状达标(field: 识别字段, 值们: (string | null)[]): boolean {
  if (字段目录[field].形状 === undefined) return true
  return 形状命中率(field, 值们) >= 0.9
}

/**
 * 四层召回：别名 → 词典精确（全字段先落定）→ 词典后缀（修饰语优先）→ 形状。
 *
 * 两轮制的理由：若边遍历字段边做后缀，「对方账号」可能被 ownerAccount 的通用词「账号」
 * 先抢走。先让所有精确命中落定，再做后缀轮，且后缀轮里带修饰语的表头只许归对应字段。
 */
export function detectMapping(
  header: (string | null)[],
  rows: (string | null)[][],
  选项: DetectOptions = {},
): MappingResult {
  const 候选 = 选项.候选字段 ?? 候选字段集(header, rows)
  const 规范 = header.map(规范化)
  const columns: Record<number, 识别字段> = {}
  const 来源: MappingResult["来源"] = {}
  const 已占 = new Set<number>()
  const 待确认: 待确认项[] = []
  const 列表头 = header.map((h) => (h ?? "").trim())
  const 表头计数 = new Map<string, number>()
  for (const h of 列表头) if (h !== "") 表头计数.set(h, (表头计数.get(h) ?? 0) + 1)

  const 认领 = (i: number, field: 识别字段, 从: 识别来源) => {
    columns[i] = field
    来源[i] = 从
    已占.add(i)
  }
  const 记待确认 = (i: number, 原因: 待确认原因, 建议字段: 识别字段 | null = null) => {
    待确认.push({ 列号: i, 表头: header[i] ?? null, 建议字段, 原因 })
  }

  // ① 别名：原始表头精确命中；不提供证据，单位闸与形状仍要过
  for (let i = 0; i < header.length; i++) {
    const 原 = 列表头[i] ?? ""
    const 目标 = 原 === "" ? undefined : 选项.别名?.get(原)
    if (目标 === undefined) continue
    if (!是识别字段(目标)) continue
    if ((表头计数.get(原) ?? 0) > 1) {
      记待确认(i, "表头重名", 目标)
      continue
    }
    if (!候选.has(目标)) {
      记待确认(i, "别名跨字段集", 目标)
      continue
    }
    if (!单位闸放行(目标, header[i] ?? null)) {
      记待确认(i, "单位冲突", 目标)
      continue
    }
    if (!形状达标(目标, 列(rows, i))) {
      记待确认(i, "形状不符", 目标)
      continue
    }
    认领(i, 目标, "别名库")
  }

  // ② 精确：全字段先落定，争抢同一列的进待确认
  for (let i = 0; i < header.length; i++) {
    if (已占.has(i)) continue
    const h = 规范[i] ?? ""
    if (h === "") continue
    const 命中 = [...候选].filter((f) => 同义词(f).includes(h) && 单位闸放行(f, header[i] ?? null))
    if (命中.length > 1) {
      记待确认(i, "多列争抢", 命中[0] ?? null)
      continue
    }
    if (命中.length === 1) 认领(i, 命中[0], "词典")
  }

  // ③ 后缀：候选按 (词长 desc, 形状命中率 desc) 排；同字段多列差距 < 5% ⇒ 不猜
  const 后缀候选 = new Map<识别字段, { i: number; 词长: number; 率: number }[]>()
  for (let i = 0; i < header.length; i++) {
    if (已占.has(i)) continue
    const h = 规范[i] ?? ""
    if (h === "") continue
    for (const field of 候选) {
      if (!单位闸放行(field, header[i] ?? null)) continue
      const 词们 = 同义词(field).filter((w) => w !== "" && h !== w && h.endsWith(w))
      if (词们.length === 0) continue
      const 词长 = Math.max(...词们.map((w) => w.length))
      // 修饰语优先：带对手/本方修饰的表头只许归对应字段
      if (对手修饰.test(h)) {
        const 允许 = 户名词尾.test(h) ? ["counterpartyName"] : 账号词尾.test(h) ? ["counterparty"] : []
        if (!允许.includes(field)) continue
      } else if (本方修饰.test(h)) {
        if (field !== "ownerAccount") continue
      } else if (field === "counterparty" && !账号词尾.test(h)) {
        continue
      }
      const 组 = 后缀候选.get(field) ?? []
      组.push({ i, 词长, 率: 形状命中率(field, 列(rows, i)) })
      后缀候选.set(field, 组)
    }
  }
  for (const [field, 组] of 后缀候选) {
    if (Object.values(columns).includes(field)) continue
    const 最长 = Math.max(...组.map((x) => x.词长))
    const 顶 = 组.filter((x) => x.词长 === 最长)
    const 排 = [...顶].sort((a, b) => b.率 - a.率 || a.i - b.i)
    const 第一 = 排[0]
    if (第一 === undefined) continue
    const 第二 = 排[1]
    if (第二 !== undefined && 第一.率 - 第二.率 < 0.05) {
      for (const x of 顶) 记待确认(x.i, "候选不唯一", field)
      continue
    }
    认领(第一.i, field, "词典")
  }

  // ④ 形状：只认 startedAt / amount / callee（R3）
  if (!Object.values(columns).includes("startedAt") && 候选.has("startedAt")) {
    const 候选列 = header.flatMap((_, i) =>
      已占.has(i) || 形状命中率("startedAt", 列(rows, i)) < 0.9 ? [] : [i],
    )
    if (候选列.length === 1) 认领(候选列[0], "startedAt", "特征")
    else if (候选列.length > 1) for (const i of 候选列) 记待确认(i, "候选不唯一", "startedAt")
  }
  if (!Object.values(columns).includes("amount") && 候选.has("amount")) {
    const 候选列 = header.flatMap((_, i) =>
      已占.has(i) || !单位闸放行("amount", header[i] ?? null) || 形状命中率("amount", 列(rows, i)) < 0.9 ? [] : [i],
    )
    if (候选列.length === 1) 认领(候选列[0], "amount", "特征")
  }
  if (!Object.values(columns).includes("callee") && 候选.has("callee")) {
    const 候选列 = header.flatMap((_, i) => (已占.has(i) || 形状命中率("callee", 列(rows, i)) < 0.9 ? [] : [i]))
    if (候选列.length === 1) 认领(候选列[0], "callee", "特征")
  }

  const unknownColumns = header.flatMap((_, i) => (columns[i] === undefined ? [i] : []))
  const 用了别名 = Object.values(来源).includes("别名库")
  const method: MappingResult["method"] = 用了别名
    ? "别名"
    : Object.keys(columns).every((k) => 来源[Number(k)] === "词典")
      ? "词典"
      : "特征"
  return { columns, method, 来源, 待确认, unknownColumns }
}

  db: Queryable,

  fingerprint: string,

    `),
  )

  if (!row) return undefined
  const raw: unknown = typeof row.mapping === "string" ? JSON.parse(row.mapping) : row.mapping
  if (raw === null || typeof raw !== "object" || Array.isArray(raw)) return undefined
  const 结果: unknown = typeof row.result === "string" ? JSON.parse(row.result) : row.result

export type 表头识别器 = (输入: 表头识别输入) => Promise<表头识别结果>

字段名只能从用户给出的候选字段里选；认不出的列写 "unknown"，不许编造字段名。`

const 账号脱敏 = (v: string) => {
  const s = v.replace(/[\s-]/g, "")
  return /^\d{8,}$/.test(s) ? `${s.slice(0, 4)}****${s.slice(-4)}` : v
}
const 姓名脱敏 = (v: string) => (/^[一-龥]{2,4}$/.test(v.trim()) ? `${v.trim().slice(0, 1)}**` : v)
const 金额脱敏 = (v: string) => (/^-?\d+\.\d{2}$/.test(v.trim()) ? `*${v.trim().length}位` : v)

function 脱敏样例(样例: (string | null)[][]): (string | null)[][] {
  return 样例.map((行) => 行.map((v) => (v === null ? null : 金额脱敏(姓名脱敏(账号脱敏(v))))))
}

export function 建表头识别器(模型: LanguageModel, 选项: { 脱敏: boolean }): 表头识别器 {

    结构: 表头输出,
    系统后缀: 表头要求,
  })
  return async (输入) => {
    const 样例们 = 选项.脱敏
      ? 输入.工作表.map((s) => ({ ...s, 样例: 脱敏样例(s.样例) }))

/** 关键字段：资金缺 counterparty/amount，话单缺 callee/startedAt（按推断候选集判）。 */
export function 需要模型(
  每表: readonly MappingResult[],
  表头们: readonly (string | null)[][],
  行们: readonly (string | null)[][][],
): boolean {
  let 未认领 = 0
  let 有形状未认领 = 0
  let 总列 = 0
  for (let s = 0; s < 每表.length; s++) {
    const 结果 = 每表[s]
    const 表头 = 表头们[s] ?? []
    const 行 = 行们[s] ?? []
    const 关键 = 结果.columns
    const 字段们 = Object.values(关键)
    // 关键缺失：任一有形状的字段一个都没认出来（金额/时间/号码）
    const 缺关键 =
      (!字段们.includes("amount") && 字段们.includes("counterparty") === false && 有形状列("amount", 表头, 行)) ||
      (!字段们.includes("startedAt") && 有形状列("startedAt", 表头, 行))
    if (缺关键) return true
    if (结果.待确认.some((项) => 项.原因 === "多列争抢" || 项.原因 === "候选不唯一" || 项.原因 === "形状不符")) return true
    for (let i = 0; i < 表头.length; i++) {
      const 名 = (表头[i] ?? "").trim()
      if (名 === "" || /备注|摘要|网点|说明|用途|序号/.test(名)) continue
      总列++
      if (结果.columns[i] !== undefined) continue
      未认领++
      const 像某种形状 = ["amount", "startedAt", "callee", "counterparty", "ownerAccount"].some(
        (f) => 是识别字段(f) && 形状命中率(f, 行.map((r) => r[i] ?? null)) >= 0.9,
      )
      if (像某种形状) 有形状未认领++
    }
  }
  return 总列 > 0 && 有形状未认领 / 总列 > 0.3
}

function 有形状列(field: 识别字段, 表头: (string | null)[], 行: (string | null)[][]): boolean {
  return 表头.some((_, i) => 形状命中率(field, 行.map((r) => r[i] ?? null)) >= 0.9)
}

export function 合并模型结果(
  表头: (string | null)[],
  行: (string | null)[][],
  基础: MappingResult,
  建议: Record<number, string>,
): MappingResult {
  const columns = { ...基础.columns }
  const 来源 = { ...基础.来源 }
  const 新待确认: 待确认项[] = []
  for (const [键, 字段名] of Object.entries(建议)) {
    const i = Number(键)
    if (!Number.isInteger(i) || i < 0 || i >= 表头.length) continue
    if (columns[i] !== undefined) continue // 规则/别名已认领，模型不许覆盖
    if (!是识别字段(字段名)) continue

      // 本期模型只落标准列；extras 字段仍靠词典/别名
      continue

    if (!形状命中(字段名, 行.map((r) => r[i] ?? null))) continue
    columns[i] = 字段名

  return { ...基础, columns, 来源, 待确认: [...基础.待确认, ...新待确认] }
}

interface 待确认项界面 {
  readonly 列号: number
  readonly 表头: string
  readonly 原因: string
  readonly 建议字段: string | null
}

/** 从 parse_detail 里读待确认；形状不对就当没有（不猜）。 */
function 读待确认(详情: unknown): { sheet: string; 项: 待确认项界面 }[] {
  if (typeof 详情 !== "object" || 详情 === null) return []
  const mapping = Reflect.get(详情, "mapping")
  if (typeof mapping !== "object" || mapping === null) return []
  const 工作表 = Reflect.get(mapping, "工作表")
  if (typeof 工作表 !== "object" || 工作表 === null) return []
  const out: { sheet: string; 项: 待确认项界面 }[] = []
  for (const [sheet, 值] of Object.entries(工作表)) {
    if (typeof 值 !== "object" || 值 === null) continue
    const 列表 = Reflect.get(值, "待确认")
    if (!Array.isArray(列表)) continue
    for (const 原始 of 列表) {
      if (typeof 原始 !== "object" || 原始 === null) continue
      const 列号 = Reflect.get(原始, "列号")
      const 表头 = Reflect.get(原始, "表头")
      const 原因 = Reflect.get(原始, "原因")
      if (typeof 列号 !== "number" || typeof 表头 !== "string" || typeof 原因 !== "string") continue
      const 建议 = Reflect.get(原始, "建议字段")
      out.push({ sheet, 项: { 列号, 表头, 原因, 建议字段: typeof 建议 === "string" ? 建议 : null } })
    }
  }
  return out
}

                        data-slot="material-pending-confirm"
                        disabled={确认中() !== undefined || 行.项.建议字段 === null}
                        onClick={() => {
                          const field = 行.项.建议字段
                          if (field === null) return
                          set确认中(`${行.sheet}:${行.项.列号}`)
                          set确认错误(undefined)
                          void props.api
                            .confirmFieldAlias({
                              projectId: props.projectId,
                              header: 行.项.表头,
                              field,
                              materialId: 清单()?.材料.id ?? "",
                            })
                            .then((结果) => {
                              if (结果.kind === "error") set确认错误(结果.message)
                            })
                            .finally(() => set确认中(undefined))
                        }}
                        class="h-6 rounded-[4px] bg-[var(--v2-background-bg-accent-soft)] px-2 font-semibold text-v2-text-text-base disabled:opacity-50"
                      >
                        确认为 {行.项.建议字段 ?? "（需人工指定）"}
                      </button>
                    </div>
                  )}
                </For>
              </section>
            </Show>
```
