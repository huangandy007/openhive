# 账户进出明细两栏（人说的一栏）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 查账户进出时，「人说的」转账（笔录 / 人工认定的 `source_kind='human'` 关系）作为一栏与流水单笔榜并排出现在工具返回与资金页上，且汇总数字一字不变。

**Architecture:** 汇总口径本来已含 human 关系（`aggregate_money` 的 `汇总条件` 不筛 `source_kind`），本轮只新增「明细出口」：在 `aggregate_money` 内部基于**同一份 `汇总条件`** 派生 `人说的笔`（子查询包住不加别名的条件，避免 `column reference is ambiguous`），`money_flow` 透传并把 `目标方向` 下推进内部选项；资金页路由额外要一份与 `money_flow` 共用的 `单笔最大`，因此把单笔榜的条件与取数抽成共享函数。前端 `money.tsx` 加「明细」两段，键缺席 / 形状不对一律显示「本次未返回」。

**Tech Stack:** Bun monorepo、Hono 路由、Drizzle `sql` 模板、PGlite/远端 PG 测试（`startProductionDb`）、SolidJS 1.9 + happy-dom 组件测试。

**Spec:** `docs/superpowers/specs/2026-11-09-事实卡片抽象/2026-11-15-P4-AI接卡片层-第一刀-账户进出明细-design.md`（下称「设计」）

- [ ] **Step 1: 在夹具里加「只有人说的转账」数据集**

在 `packages/opencode/test/fixtures/p3/dataset.ts` 末尾追加（走生产入口，`#006-35`；独立项目，`#005-34`）：

```ts
/**
 * 人说话数据集：22 条只有笔录（`source_kind='human'`）的转账——21 条「出」+ 1 条「入」。
 *
 * 用途：① 人说的笔的上限 20 与「先按方向过滤再截断」；② 入向观测；③ 对手过滤；
 * ④ 「只有人说的笔时 `数据` 不是 null」（本项目没有任何流水）。
 * 时间与金额逐条不同，排序与金额断言才挡得住「排错 / 取错行」。
 * 入账那笔刻意是最新一天（07-22）：若先 `limit 20` 再按方向过滤，出向会少一条。
 */
export interface 人说话数据集 {
  项目: string
  主账号: string
  对手人名们: string[]
}

export async function 种人说话数据集(
  db: Queryable,
  文件: Map<string, Uint8Array>,
): Promise<人说话数据集> {
  const 项目 = "p-人说话"
  const 主账号 = "6222021234567899100"
  const 收款人 = "人说入账人"
  const 对手人名们 = Array.from({ length: 21 }, (_, i) => `人说对手${i + 1}`)
  const 日期 = (i: number) => `2026年7月${String(i + 1).padStart(2, "0")}日`
  // 账号必须出现在每一行的原文里：`回读` 要求实体规范值与关系出处都能在片文本里找到，
  // 写成「主户」会让整份材料判「有问题」、一条关系都落不了库（Task 1 实得取证）。
  const 正文 = [
    ...对手人名们.map((名, i) => `${i + 1}、${日期(i)}，${主账号}向${名}转款${101 + i}元。`),
    `22、${日期(21)}，${收款人}向${主账号}转款500元。`,
    "",
  ].join("\n")
  const 入队 = 造入队器(db, 文件)
  const 跑 = 造跑器(db, 文件)
  const 材料 = await 入队(项目, "P3人说话.txt", new TextEncoder().encode(正文))
  await 跑(材料, async () =>
    建假抽取器(() => ({
      实体: [
        { 类型: "账户", 取值: 主账号 },
        ...对手人名们.map((名) => ({ 类型: "人", 取值: 名 })),
        { 类型: "人", 取值: 收款人 },
      ],
      关系: [
        ...对手人名们.map((名, i) => ({
          类型: "转账",
          从: 主账号,
          到: 名,
          金额原文: `${101 + i}元`,
          时间原文: 日期(i),
          原文: `${i + 1}、${日期(i)}，主户向${名}转款${101 + i}元。`,
        })),
        {
          类型: "转账",
          从: 收款人,
          到: 主账号,
          金额原文: "500元",
          时间原文: 日期(21),
          原文: `22、${日期(21)}，${收款人}向主户转款500元。`,
        },
      ],
      事件: [],
    })),
  )
  return { 项目, 主账号, 对手人名们 }
}
```

  test("笔录转账按关系行列出：方向 / 对手 / 金额 / 首末 / 精度 / 次数", async () => {

        方向: "出",
        对手: "李四",
        金额: "5000.00",
        首: expect.any(Number),
        末: expect.any(Number),
        精度: "到日",
        次数: null,
      },
      {
        id: expect.any(String),
        方向: "出",
        对手: "李四",
        金额: "10000.00",
        首: expect.any(Number),
        末: expect.any(Number),
        精度: "到日",
        次数: null,
      },
    ])
  })

  test("数据关系一条都不进人说的笔（A→B 机器合计 30000 不许出现）", async () => {
    const 数据 = (await aggregate_money(ctx(), { 账户: 数据集.账号.A })).数据 as { 人说的笔: 人说行[] }
    expect(数据.人说的笔.map((行) => 行.金额)).toEqual(["5000.00", "10000.00"])
    expect(数据.人说的笔.map((行) => 行.金额)).not.toContain("30000.00")
  })

  test("内部选项 方向：只作用于人说的笔，汇总与对手方不变", async () => {

    expect(入数据.汇总).toEqual([{ 方向: "入", 笔数: 3, 合计: "35000.00" }])

```ts
interface 流水行 { readonly id: string; readonly 金额: string; readonly 时间: number | null; readonly 摘要: string | null }
interface 人说行 {
  readonly id: string
  readonly 方向: "出" | "入"
  readonly 对手: string
  readonly 金额: string | null
  readonly 首: number | null
  readonly 末: number | null
  readonly 精度: string | null
  readonly 次数: number | null
}
/** 明细的一段：`未返回` = 服务端没给这个键（老版本）或形状不对——**不是**「空」（`#002-02`）。 */
type 明细段<T> = { readonly 状态: "有"; readonly 行们: T[] } | { readonly 状态: "未返回" }

const 可空数 = (值: unknown): number | null | undefined => (值 === null ? null : 是数(值) ? 值 : undefined)
const 可空串 = (值: unknown): string | null | undefined => (值 === null ? null : 是串(值) ? 值 : undefined)

const 解单笔 = (条: unknown): 流水行 | undefined => {
  if (条 === null || typeof 条 !== "object") return undefined
  const 行 = 条 as Record<string, unknown>
  const 时间 = 可空数(行.时间)
  const 摘要 = 可空串(行.摘要)
  if (!是串(行.id) || !是串(行.金额) || 时间 === undefined || 摘要 === undefined) return undefined
  return { id: 行.id, 金额: 行.金额, 时间, 摘要 }
}

const 解人说 = (条: unknown): 人说行 | undefined => {
  if (条 === null || typeof 条 !== "object") return undefined
  const 行 = 条 as Record<string, unknown>
  const 金额 = 可空串(行.金额)
  const 首 = 可空数(行.首)
  const 末 = 可空数(行.末)
  const 精度 = 可空串(行.精度)
  const 次数 = 可空数(行.次数)
  if (!是串(行.id) || !是串(行.对手)) return undefined
  if (行.方向 !== "出" && 行.方向 !== "入") return undefined
  if (金额 === undefined || 首 === undefined || 末 === undefined || 精度 === undefined || 次数 === undefined)
    return undefined
  return { id: 行.id, 方向: 行.方向, 对手: 行.对手, 金额, 首, 末, 精度, 次数 }
}

/** 一行形状不对 ⇒ 整份判错：不静默降级成「只显示认得的那几行」（`#002-02`）。 */
function 解段<T>(原: unknown, 解一: (条: unknown) => T | undefined): 明细段<T> {
  if (!Array.isArray(原)) return { 状态: "未返回" }
  const 行们: T[] = []
  for (const 条 of 原) {
    const 行 = 解一(条)
    if (行 === undefined) return { 状态: "未返回" }
    行们.push(行)
  }
  return { 状态: "有", 行们 }
}
```
