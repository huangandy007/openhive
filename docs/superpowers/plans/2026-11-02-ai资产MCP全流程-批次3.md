> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 MCP 资产能走完「提交 → 业务审核 → 试连 → 技术发布 → 按人/组授权可见」，并把「工作副本」与「已发布快照」分开，保证迭代中改坏不影响线上。

**Tech Stack:** Bun 1.4.2、TypeScript 5.8、Effect 4 beta、Drizzle 原始 SQL、SolidJS 1.9、`@modelcontextprotocol/sdk` 1.29.0、PGlite 测试夹具。

**Spec:** `docs/superpowers/specs/2026-11-02-ai资产MCP全流程-design.md`（批次 3，§3.1.4 / §3.2.2 / §3.2.4 / §5.4 / §6 / §7）

- **不新增依赖**（`@modelcontextprotocol/sdk` 已在 `packages/asset-service/package.json`）；不改根 `package.json` / `bun.lock`（跑完 `git diff --stat bun.lock` 必须为空）。

- **信号命名照抄既有习惯**：`[值, 设值]`。
- **根目录 `bun test` 禁止**（`exit 1`）；每条测试命令套 `timeout 300`。
- **不 push、不 merge、不 `git add -A`**：每个 Task 只 `git add` 本 Task 的文件。
- **报实测**：报告只写跑过的命令与它的实际输出（`#006-29`）。

```ts
import { Client } from "@modelcontextprotocol/sdk/client/index.js"
import { StdioClientTransport } from "@modelcontextprotocol/sdk/client/stdio.js"
import { StreamableHTTPClientTransport } from "@modelcontextprotocol/sdk/client/streamableHttp.js"
import { spawnSync } from "node:child_process"
import type { McpConnectionInput } from "./mcp-connection"

export const PROBE_DEFAULT_TIMEOUT_MS = 5_000
export const PROBE_MAX_TIMEOUT_MS = 30_000

export type McpProbeResult =
  | { readonly ok: true; readonly toolCount: number }
  | { readonly ok: false; readonly error: string }

export type McpProbe = (
  input: McpConnectionInput,
  options?: { readonly timeoutMs?: number },
) => Promise<McpProbeResult>

/** 试连：真连一次、列一次工具、立刻关掉。任何失败都收成 `ok:false`，不抛。 */
export async function probeMcpConnection(
  input: McpConnectionInput,
  options: { readonly timeoutMs?: number } = {},
): Promise<McpProbeResult> {
  const 超时 = Math.min(options.timeoutMs ?? input.timeoutMs ?? PROBE_DEFAULT_TIMEOUT_MS, PROBE_MAX_TIMEOUT_MS)
  if (!Number.isInteger(超时) || 超时 <= 0) return { ok: false, error: "试连超时必须为正整数（毫秒）" }

  const transport = input.transport === "remote" ? 远程传输(input) : 本地传输(input)
  if (typeof transport === "string") return { ok: false, error: transport }

  const client = new Client({ name: "openhive-asset-probe", version: "1.0.0" }, { capabilities: {} })
  let 计时: ReturnType<typeof setTimeout> | undefined
  try {
    const listed = await Promise.race([
      (async () => {
        await client.connect(transport)
        return client.listTools()
      })(),
      new Promise<never>((_, reject) => {
        计时 = setTimeout(() => reject(new Error(`试连超时（${超时}ms）`)), 超时)
      }),
    ])
    return { ok: true, toolCount: Array.isArray(listed.tools) ? listed.tools.length : 0 }
  } catch (error) {
    return { ok: false, error: 脱敏(错误文本(error), input) }
  } finally {
    if (计时 !== undefined) clearTimeout(计时)
    const pid = transport instanceof StdioClientTransport ? transport.pid : undefined
    try { await client.close() } catch { /* 关不掉不影响结论 */ }
    if (typeof pid === "number") 杀进程树(pid)
  }
}

function 远程传输(input: McpConnectionInput) {
  if (input.url === undefined || input.url === "") return "remote 缺少服务地址"
  let url: URL
  try { url = new URL(input.url) } catch { return "remote 的服务地址不是合法 URL" }
  if (url.protocol !== "http:" && url.protocol !== "https:") return "remote 只支持 http / https"
  return new StreamableHTTPClientTransport(url, {
    requestInit: input.headers === undefined ? undefined : { headers: input.headers },
  })
}

function 本地传输(input: McpConnectionInput) {
  if (input.command === undefined || input.command.length === 0) return "local 缺少命令"
  const [cmd, ...args] = input.command
  return new StdioClientTransport({
    command: cmd!,
    args,
    stderr: "pipe",
    ...(input.cwd === undefined ? {} : { cwd: input.cwd }),
    env: { ...process.env, ...input.environment },
  })
}

/** 超时后连子进程一起杀：Windows 用 taskkill /T；其它平台杀子进程（无进程组）。 */
function 杀进程树(pid: number) {
  try {
    if (process.platform === "win32") {
      spawnSync("taskkill", ["/pid", String(pid), "/T", "/F"], { stdio: "ignore" })
    } else {
      process.kill(pid, "SIGKILL")
    }
  } catch { /* 已经退出 */ }
}

function 错误文本(error: unknown): string {
  return error instanceof Error ? error.message : String(error)
}

/** 错误消息里不得带凭据原文：命中就把出现处换成 `***`，再截断。 */
function 脱敏(text: string, input: McpConnectionInput): string {
  const 秘密 = [...Object.values(input.headers ?? {}), ...Object.values(input.environment ?? {})]
  let out = text
  for (const value of 秘密) {
    if (value !== "") out = out.split(value).join("***")
  }
  return out.slice(0, 500)
}
```

async function prepareMcpPublish(db: AssetTarget, assetId: string, probe: McpProbe, now: number) {

    const 允许 = yield* GovernanceAuthz.进(deps, user.value.id, BOARD, "admin")
    if (!允许) return fail(FORBIDDEN, "需要资产治理管理员权限")
    const params = new URL(request.url, "http://localhost").searchParams
    const q = (params.get("q") ?? "").trim()
    const limit = Math.min(Math.max(Number(params.get("limit") ?? 20) || 20, 1), 50)
    const rows = rowsOf(await deps.pg().execute(sql`

       order by name, id limit ${limit}
