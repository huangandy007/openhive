## Global Constraints

- **不新增依赖**；不改根 `package.json` / `bun.lock`（跑完 `git diff --stat bun.lock` 必须为空）。
- **迁移只加不改**：新增 `0029_asset_mcp_connection.sql` 与其 `.down.sql`，不碰 `0001`–`0028`。本批**不对远端 PG 执行迁移**，只由 PGlite 夹具跑 DDL。
- **jsonb 读取必须两种驱动同形**：`db.execute` 是原始 SQL，PGlite 与 bun-sql 对 jsonb 的返回形状不同。本计划**一律 `::text` 取出再 `JSON.parse`**，不写「字符串 / 对象」双分支。
- **jsonb 写入直接绑对象，不要 `JSON.stringify(...)::jsonb`**（Task 1 实测订正）：Bun SQL 会对字符串参数再编码一层，落库变成 jsonb **字符串**（`"{}"`），`fields_check` 当场拒。绑对象得到 `jsonb_typeof = 'object'`。
- **`text[]` 写入必须显式 `array[...]::text[]`**：绝不能把 JS 数组直接塞进 drizzle 模板（会被展开成行构造器，`42804`）。
- **凭据不出读接口**：`headers` / `environment` 的值只以 `MASKED_VALUE` 回给前端；任何日志与错误消息也不带原文。
- **前端只 `import type` 服务端类型**：`packages/app` 里从 `@opencode-ai/asset-service/mcp-connection` 拿的东西一律 `import type`（运行时导入会把 drizzle / migrate 拖进浏览器 bundle）。需要常量就写本地字面量。
- **信号命名照抄既有习惯**：`[值, 设值]`（`设筛选` / `设提示` / `设加载`），不写 `setXxx`。
- **根目录 `bun test` 禁止**（`exit 1`）；每条测试命令都套 `timeout 300`。
- **授权先于副作用**（`#004-09`）：MCP 创建、连接读写的拒绝都必须发生在写库之前。
- **不 push、不 merge、不 `git add -A`**：每个 Task 只 `git add` 本 Task 的文件。
- **报实测**：报告只写跑过的命令与它的实际输出（`#006-29`）。
- **保留定制注释**：所有改上游文件的段落前加 `// 【保留的定制 · 同步上游时不要丢】`。

export * as McpConnection from "./mcp-connection"

import { sql, type SQL } from "drizzle-orm"
import { rowsOf } from "@opencode-ai/auth/migrate"
import { AssetMetadataError, type AssetTarget } from "./asset"
import type { MetadataIssue } from "./metadata"

/**
 * MCP 连接规格的**唯一实现**（2026-11-02 设计 §3.1.2）。
 *
 * ## 为什么读 jsonb 要 `::text` 再 parse
 *
 * `db.execute` 走原始 SQL，drizzle 的列映射不生效；而 PGlite 与 bun-sql 对 jsonb 的
 * 返回形状不同（一个给对象、一个给 JSON 文本）。`::text` + `JSON.parse` 把差异收在
 * 一处，不写双分支（写了就总有一条没人跑）。
 *
 * ## 为什么值要遮罩两次也不嫌多
 *
 * 遮罩只发生在**出口投影**（`遮罩连接`）。库里仍是明文——同步器与试连要真值。
 * 把遮罩做进 `getMcpConnection` 会让内部读也拿不到凭据，同步器就得再写一份读路径。
 */

export const MCP_TRANSPORTS = ["local", "remote"] as const
export type McpTransport = (typeof MCP_TRANSPORTS)[number]

export const MASKED_VALUE = "***(已设置)"

export interface McpConnectionInput {
  readonly transport: McpTransport
  readonly url?: string
  readonly command?: readonly string[]
  readonly cwd?: string
  readonly headers?: Readonly<Record<string, string>>
  readonly environment?: Readonly<Record<string, string>>
  readonly enabled?: boolean
  readonly timeoutMs?: number
}

export interface McpConnectionRecord {
  readonly assetId: string
  readonly transport: McpTransport
  readonly url: string | null
  readonly command: readonly string[] | null
  readonly cwd: string | null
  readonly headers: Readonly<Record<string, string>>
  readonly environment: Readonly<Record<string, string>>
  readonly enabled: boolean
  readonly timeoutMs: number | null
  readonly publishedConfig: Readonly<Record<string, unknown>> | null
  readonly checkedAt: number | null
  readonly checkOk: boolean | null
  readonly checkError: string
  readonly toolCount: number | null
  readonly createdAt: number
  readonly updatedAt: number
}

/** 读接口的形状：**没有** `publishedConfig`（那是同步器的输入，不是浏览数据）。 */
export type McpConnectionView = Omit<McpConnectionRecord, "publishedConfig">

function 是对象(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value)
}

function 读字符串表(
  value: unknown,
  field: string,
  issues: MetadataIssue[],
): Record<string, string> | undefined {
  if (value === undefined) return undefined
  if (!是对象(value)) {
    issues.push({ field, message: "必须是对象" })
    return undefined
  }
  const out: Record<string, string> = {}
  for (const [key, item] of Object.entries(value)) {
    if (key.trim() === "") {
      issues.push({ field, message: "键名不能为空" })
      return undefined
    }
    if (typeof item !== "string") {
      issues.push({ field: `${field}.${key}`, message: "值必须是字符串" })
      return undefined
    }
    out[key] = item
  }
  return out
}

function 读命令(value: unknown, issues: MetadataIssue[]): readonly string[] | undefined {
  if (value === undefined) return undefined
  if (!Array.isArray(value) || value.length === 0) {
    issues.push({ field: "connection.command", message: "local 必须给非空的命令数组" })
    return undefined
  }
  if (!value.every((part): part is string => typeof part === "string")) {
    issues.push({ field: "connection.command", message: "命令的每一项必须是字符串" })
    return undefined
  }
  if (value[0]!.trim() === "") {
    issues.push({ field: "connection.command", message: "命令的第一项不能为空" })
    return undefined
  }
  return value
}

/**
 * 把外部输入收成闭集内的连接规格；**一次列全部问题**（与 `AssetMetadataError` 同一条纪律）。
 *
 * 表格形状的约束（remote 不能有 command / environment、local 不能有 url / headers）
 * 这里先判一遍，库里的 CHECK 是最后一道；两层都留，是因为出口要能回带字段名的 400，
 * 而库只回约束名。
 */
export function 规范化连接(value: unknown): McpConnectionInput {
  if (!是对象(value)) {
    throw new AssetMetadataError([{ field: "connection", message: "连接规格必须是对象" }])
  }
  const transport = value["transport"]
  if (transport !== "local" && transport !== "remote") {
    throw new AssetMetadataError([{ field: "connection.transport", message: "只支持 local / remote" }])
  }

  const issues: MetadataIssue[] = []
  const url = value["url"]
  const command = 读命令(value["command"], issues)
  const cwd = value["cwd"]
  const headers = 读字符串表(value["headers"], "connection.headers", issues)
  const environment = 读字符串表(value["environment"], "connection.environment", issues)
  const enabled = value["enabled"]
  const timeoutMs = value["timeoutMs"]

  if (transport === "remote") {
    if (typeof url !== "string" || url.trim() === "") {
      issues.push({ field: "connection.url", message: "remote 必须给服务地址" })
    }
    if (value["command"] !== undefined) {
      issues.push({ field: "connection.command", message: "remote 不能带命令" })
    }
    if (environment !== undefined) {
      issues.push({ field: "connection.environment", message: "remote 不能带环境变量" })
    }
    if (cwd !== undefined) {
      issues.push({ field: "connection.cwd", message: "remote 不能带工作目录" })
    }
  } else {
    if (value["command"] === undefined) {
      issues.push({ field: "connection.command", message: "local 必须给命令数组" })
    }
    if (url !== undefined) {
      issues.push({ field: "connection.url", message: "local 不能带服务地址" })
    }
    if (headers !== undefined) {
      issues.push({ field: "connection.headers", message: "local 不能带请求头" })
    }
  }

  if (cwd !== undefined && (typeof cwd !== "string" || cwd.trim() === "")) {
    issues.push({ field: "connection.cwd", message: "工作目录必须是非空字符串" })
  }
  if (enabled !== undefined && typeof enabled !== "boolean") {
    issues.push({ field: "connection.enabled", message: "enabled 必须是布尔值" })
  }
  if (
    timeoutMs !== undefined &&
    (typeof timeoutMs !== "number" || !Number.isInteger(timeoutMs) || timeoutMs <= 0)
  ) {
    issues.push({ field: "connection.timeoutMs", message: "超时必须是正整数（毫秒）" })
  }
  if (issues.length > 0) throw new AssetMetadataError(issues)

  return {
    transport,
    ...(transport === "remote"
      ? {
          url: (url as string).trim(),
          ...(headers === undefined ? {} : { headers }),
        }
      : {
          command: command!,
          ...(cwd === undefined ? {} : { cwd }),
          ...(environment === undefined ? {} : { environment }),
        }),
    ...(enabled === undefined ? {} : { enabled }),
    ...(timeoutMs === undefined ? {} : { timeoutMs }),
  }
}

/**
 * 把命令数组编成 PG 数组表达式。
 *
 * ⚠️ 与 `asset.ts` 的 `tagsLiteral` 同一条硬约束：JS 数组直接进模板会被展开成行构造器
 * `($7, $8)`，`text[]` 列回 `42804`。每个元素仍走独立占位符，不拼字符串。
 */
function 命令数组(command: readonly string[] | undefined): SQL {
  if (command === undefined) return sql`null`
  if (command.length === 0) return sql`'{}'::text[]`
  return sql`array[${sql.join(
    command.map((part) => sql`${part}`),
    sql`, `,
  )}]::text[]`
}

const COLUMNS = sql`
  asset_id, transport, url, command, cwd,
  headers::text as headers, environment::text as environment,
  enabled, timeout_ms, published_config::text as published_config,
  checked_at, check_ok, check_error, tool_count, created_at, updated_at
`

function 解析对象文本(value: unknown, field: string): Record<string, unknown> {
  if (typeof value !== "string") throw new Error(`${field} 期望 JSON 文本，实得 ${typeof value}`)
  const parsed: unknown = JSON.parse(value)
  if (!是对象(parsed)) throw new Error(`${field} 不是对象`)
  return parsed
}

function 解析字符串表(value: unknown, field: string): Record<string, string> {
  const parsed = 解析对象文本(value, field)
  const out: Record<string, string> = {}
  for (const [key, item] of Object.entries(parsed)) {
    if (typeof item !== "string") throw new Error(`${field}.${key} 不是字符串`)
    out[key] = item
  }
  return out
}

function 可空数(value: unknown, field: string): number | null {
  if (value === null) return null
  if (typeof value !== "number" && typeof value !== "string") throw new Error(`${field} 不是数字`)
  const parsed = Number(value)
  if (!Number.isFinite(parsed)) throw new Error(`${field} 不是有限数字`)
  return parsed
}

function toRecord(row: Record<string, unknown>): McpConnectionRecord {
  if (row.transport !== "local" && row.transport !== "remote") {
    throw new Error(`无法解析 transport：${String(row.transport)}`)
  }
  if (!Array.isArray(row.command) && row.command !== null) {
    throw new Error(`无法解析 command：期望数组或 null，实得 ${typeof row.command}`)
  }
  if (typeof row.enabled !== "boolean") throw new Error(`无法解析 enabled：${typeof row.enabled}`)
  if (row.check_ok !== null && typeof row.check_ok !== "boolean") {
    throw new Error(`无法解析 check_ok：${typeof row.check_ok}`)
  }
  const published = row.published_config === null ? null : 解析对象文本(row.published_config, "published_config")
  return {
    assetId: String(row.asset_id),
    transport: row.transport,
    url: row.url === null ? null : String(row.url),
    command: row.command === null ? null : (row.command as unknown[]).map(String),
    cwd: row.cwd === null ? null : String(row.cwd),
    headers: 解析字符串表(row.headers, "headers"),
    environment: 解析字符串表(row.environment, "environment"),
    enabled: row.enabled,
    timeoutMs: 可空数(row.timeout_ms, "timeout_ms"),
    publishedConfig: published,
    checkedAt: 可空数(row.checked_at, "checked_at"),
    checkOk: row.check_ok,
    checkError: String(row.check_error),
    toolCount: 可空数(row.tool_count, "tool_count"),
    createdAt: Number(row.created_at),
    updatedAt: Number(row.updated_at),
  }
}

/** 写一条连接。调用方先 `规范化连接`；主键冲突 / 外键失败必须抛出去。 */
export async function insertMcpConnection(
  db: AssetTarget,
  assetId: string,
  input: McpConnectionInput,
  now: number,
): Promise<void> {
  await db.execute(sql`
    insert into auth.asset_mcp_connection (
      asset_id, transport, url, command, cwd, headers, environment, enabled, timeout_ms,
      published_config, checked_at, check_ok, check_error, tool_count, created_at, updated_at
    ) values (
      ${assetId}, ${input.transport}, ${input.url ?? null}, ${命令数组(input.command)}, ${input.cwd ?? null},
      ${JSON.stringify(input.headers ?? {})}::jsonb, ${JSON.stringify(input.environment ?? {})}::jsonb,
      ${input.enabled ?? true}, ${input.timeoutMs ?? null},
      null, null, null, '', null, ${now}, ${now}
    )
  `)
}

/** 按资产 id 读连接；没有返回 `undefined`（不是半填对象）。 */
export async function getMcpConnection(
  db: AssetTarget,
  assetId: string,
): Promise<McpConnectionRecord | undefined> {
  const rows = rowsOf(
    await db.execute(sql`select ${COLUMNS} from auth.asset_mcp_connection where asset_id = ${assetId}`),
  )
  const [row] = rows
  return row === undefined ? undefined : toRecord(row)
}

/**
 * 改一条已存在的连接。**不插入**：没有行说明资产不是从 `createMcpDraft` 来的
 * （本批的唯一建法），出口按 404 处理比偷偷补一行更好排障。
 */
export async function updateMcpConnection(
  db: AssetTarget,
  assetId: string,
  value: unknown,
  now: number,
): Promise<McpConnectionRecord | undefined> {
  const input = 规范化连接(value)
  const rows = rowsOf(
    await db.execute(sql`
      update auth.asset_mcp_connection set
        transport = ${input.transport},
        url = ${input.url ?? null},
        command = ${命令数组(input.command)},
        cwd = ${input.cwd ?? null},
        headers = ${JSON.stringify(input.headers ?? {})}::jsonb,
        environment = ${JSON.stringify(input.environment ?? {})}::jsonb,
        enabled = ${input.enabled ?? true},
        timeout_ms = ${input.timeoutMs ?? null},
        updated_at = ${now}
      where asset_id = ${assetId}
      returning asset_id
    `),
  )
  return rows.length === 0 ? undefined : getMcpConnection(db, assetId)
}

function 遮罩(map: Readonly<Record<string, string>>): Record<string, string> {
  return Object.fromEntries(Object.keys(map).map((key) => [key, MASKED_VALUE]))
}

/**
 * 出口投影：去掉 `publishedConfig`，把凭据值换成 `MASKED_VALUE`。
 *
 * 逐字段构造而不是解构删键：多一个字段时这里会编译报错，提醒「新字段要不要出读接口」。
 */
export function 遮罩连接(record: McpConnectionRecord): McpConnectionView {
  return {
    assetId: record.assetId,
    transport: record.transport,
    url: record.url,
    command: record.command,
    cwd: record.cwd,
    headers: 遮罩(record.headers),
    environment: 遮罩(record.environment),
    enabled: record.enabled,
    timeoutMs: record.timeoutMs,
    checkedAt: record.checkedAt,
    checkOk: record.checkOk,
    checkError: record.checkError,
    toolCount: record.toolCount,
    createdAt: record.createdAt,
    updatedAt: record.updatedAt,
  }
}

  readonly category: string
  readonly tags?: readonly string[]

/**
 * 【保留的定制 · 同步上游时不要丢】建一份 MCP 草稿（2026-11-02 设计 §3.1.2）。
 *
 * 与 `createDraftAsset` 的差异只有两处：**没有内容目录**（连接就是内容），以及
 * **连接与元数据没有共同事务**。后者的补偿方向与 skill 相反：
 * 先写元数据、再写连接；连接写失败就删掉刚写的元数据（级联删连接），
 * 因为「资产行存在但连不上」比「没有这个资产」更难排障。
 *
 * 校验在**任何写入之前**：分类闭集读库、连接规格 `规范化连接`。
 */
export async function createMcpDraft(input: CreateMcpDraftInput): Promise<AssetMetadata> {
  if (!(await isEnabledCategory(input.db, input.category))) {
    throw new AssetMetadataError([{ field: "category", message: `未知业务分类：${input.category}` }])
  }
  const 连接 = 规范化连接(input.connection)

  const now = input.now ?? Math.floor(Date.now() / 1000)
  const metadata: AssetMetadata = {
    id: input.id ?? `asset_${randomUUID()}`,
    name: input.name,
    type: "mcp",

/** 资产治理板块 id：与 `governance-assets.ts` 的 `BOARD` 逐字相同（刻意不共享内部常量）。 */
const ASSET_GOVERNANCE_BOARD = "asset-governance"

/**
 * 【保留的定制 · 同步上游时不要丢】MCP 草稿分支（2026-11-02 设计 §5.3）。
 *
 * 次序：身份（调用方已取）→ 请求体是对象（调用方已判）→ **治理 admin 授权** →
 * 字段校验 → 写库。授权排在字段校验之前：非管理员不该从 400 的措辞里读到字段规则；
 * 拒绝也要留审计（`GovernanceAuthz.进` 的唯一落点）。
 */
function handleCreateMcpDraft(
  body: Record<string, unknown>,
  deps: Deps,
  userId: string,
  userName: string,
) {
  return Effect.gen(function* () {
    const 允许 = yield* GovernanceAuthz.进(deps, userId, ASSET_GOVERNANCE_BOARD, "admin")
    if (!允许) return fail(FORBIDDEN, "只有资产治理管理员可以新建 MCP 资产")

    const 字段 = yield* 校验资产表单(deps, body)
    if (typeof 字段 === "string") return fail(BAD_REQUEST, 字段)

    const 部门 = yield* 取部门(deps, userId)
    const 结果 = yield* Effect.result(
      Effect.tryPromise({
        try: () =>
          createMcpDraft({
            db: deps.pg(),
            owner: { id: userId, name: userName, department: 部门 },
            name: 字段.名称,
            description: 字段.描述,
            category: 字段.分类,
            visibility: 字段.可见,
            connection: body["connection"],
          }),
        catch: (error) => error,
      }),
    )
    if (Result.isFailure(结果)) {
      const error = 结果.failure
      if (error instanceof AssetMetadataError) return fail(BAD_REQUEST, error.message)
      throw error
    }
    const asset = 结果.success
    return ok({ assetId: asset.id, name: asset.name, status: asset.status, visibility: asset.visibility })
  })
}

/**
 * 连接规格的**读**权：作者 / 显式 read（含空间可见与授权行）/ 治理管理员。
 *
 * 判据只有 `canReadAsset` 一份（`#002-06`）；治理管理员是后补的一条来源——
 * 治理台要能排障，但不能再写一套可见性规则。
 */
function 能读连接(deps: Deps, userId: string, asset: AssetMetadata) {
  return Effect.gen(function* () {
    if (asset.ownerId === userId) return true
    const 部门 = yield* 取部门(deps, userId)
    const grants = yield* Effect.promise(() => grantsFor(deps.pg(), userId))
    if (canReadAsset({ userId, department: 部门 }, asset, grants)) return true
    return yield* GovernanceAuthz.进(deps, userId, ASSET_GOVERNANCE_BOARD, "admin")
  })
}

/**
 * 连接规格的**写**权：作者 / 资产上的 admin 授权 / 治理管理员。
 *
 * `write` 不在此列：连接里的凭据与命令是运行期高危输入，与「改描述」不是同一档
 * （设计 §5.3）。`admin` 授权仍然算——那是资产域的既有管理员来源，不因本功能另造一套。
 */
function 能写连接(deps: Deps, userId: string, asset: AssetMetadata) {
  return Effect.gen(function* () {
    if (asset.ownerId === userId) return true
    const 部门 = yield* 取部门(deps, userId)
    const grants = yield* Effect.promise(() => grantsFor(deps.pg(), userId))
    if (effectivePermissions({ userId, department: 部门 }, asset, grants).has("admin")) return true
    return yield* GovernanceAuthz.进(deps, userId, ASSET_GOVERNANCE_BOARD, "admin")
  })
}

const CONNECTION_WRITABLE_STATUSES: ReadonlySet<AssetMetadata["status"]> = new Set(["draft", "iterating"])

/**
 * `GET /openhive/assets/:id/connection`（2026-11-02 设计 §3.1.2）。
 *
 * 不返回任何凭据值：`遮罩连接` 把 `headers` / `environment` 的值统一换成
 * `MASKED_VALUE`，`publishedConfig` 整个不出视图。
 */
function handleConnectionRead(deps: Deps) {
  return Effect.gen(function* () {
    const user = yield* 身份
    if (Option.isNone(user)) return unauthorized()
    const userId = user.value.id
    const 参数 = yield* HttpRouter.params
    const assetId = 参数.id ?? ""

    const asset = yield* Effect.promise(() => assetById(deps.pg(), assetId))
    if (asset === undefined) return fail(NOT_FOUND, "资产不存在")
    // 授权先于类型判断：否则无关用户能靠 400/403 的差别探出「这个 id 是什么类型」。
    if (!(yield* 能读连接(deps, userId, asset))) return fail(FORBIDDEN, "无权查看这台 MCP 的连接")
    if (asset.type !== "mcp") return fail(BAD_REQUEST, "只有 MCP 资产有连接规格")

    const row = yield* Effect.promise(() => getMcpConnection(deps.pg(), asset.id))
    if (row === undefined) return fail(NOT_FOUND, "这台资产还没有连接规格")
    return ok(遮罩连接(row))
  })
}

/**
 * `PATCH /openhive/assets/:id/connection`（2026-11-02 设计 §3.1.4 的编辑窗口）。
 *
 * 只有 `draft` / `iterating` 能改：审核中 / 待发布期改连接，等于绕开审核换内容。
 * 次序：身份 → 存在 → 授权 → 类型 → **状态** → 请求体 → 校验 / 写。
 */
function handleConnectionWrite(request: HttpServerRequest.HttpServerRequest, deps: Deps) {
  return Effect.gen(function* () {
    const user = yield* 身份
    if (Option.isNone(user)) return unauthorized()
    const userId = user.value.id
    const 参数 = yield* HttpRouter.params
    const assetId = 参数.id ?? ""

    const asset = yield* Effect.promise(() => assetById(deps.pg(), assetId))
    if (asset === undefined) return fail(NOT_FOUND, "资产不存在")
    if (!(yield* 能写连接(deps, userId, asset))) return fail(FORBIDDEN, "无权修改这台 MCP 的连接")
    if (asset.type !== "mcp") return fail(BAD_REQUEST, "只有 MCP 资产有连接规格")
    if (!CONNECTION_WRITABLE_STATUSES.has(asset.status)) {
      return fail(CONFLICT, "只有草稿或迭代中的资产可以改连接")
    }

    const body = yield* 读取体(request)
    if (!isRecord(body)) return fail(BAD_REQUEST, "请求体不是 JSON 对象")

    const 结果 = yield* Effect.result(
      Effect.tryPromise({
        try: () => updateMcpConnection(deps.pg(), asset.id, body["connection"], nowSeconds()),
        catch: (error) => error,
      }),
    )
    if (Result.isFailure(结果)) {
      const error = 结果.failure
      if (error instanceof AssetMetadataError) return fail(BAD_REQUEST, error.message)
      throw error
    }
    if (结果.success === undefined) return fail(NOT_FOUND, "这台资产还没有连接规格")
    return ok(遮罩连接(结果.success))
  })
}

export function 解析命令(text: string): readonly string[] {
  return text
    .split(/\r?\n/)
    .filter((line) => line.trim() !== "")
}

export function McpCreateForm(props: McpCreateFormProps): JSX.Element {
  const [名称, 设名称] = createSignal("")
  const [描述, 设描述] = createSignal("")
  const [分类, 设分类] = createSignal("")
  const [可见, 设可见] = createSignal<CreateMcpDraftInput["visibility"]>("targeted")
  const [传输, 设传输] = createSignal<"local" | "remote">("remote")
  const [地址, 设地址] = createSignal("")
  const [命令, 设命令] = createSignal("")
  const [目录, 设目录] = createSignal("")
  const [失败话, 设失败话] = createSignal<string>()
  const [忙, 设忙] = createSignal(false)

  const 分类已读 = () => props.categoriesLoaded ?? true
  const 分类读不到 = () => props.categories.length === 0 && props.categoriesError !== undefined
  const 分类加载中 = () => props.categories.length === 0 && !分类读不到() && !分类已读()
  const 没有分类 = () => props.categories.length === 0 && !分类读不到() && !分类加载中()

  const 清错误 = () => 设失败话(undefined)

  /** 返回连接规格或一句错误——两种传输方式共用一个出口，避免提交时两处判断漂。 */
  function 连接规格(): McpConnectionInput | string {
    if (传输() === "remote") {
      const url = 地址().trim()
      if (url === "") return "请填写服务地址"
      return { transport: "remote", url }
    }
    const command = 解析命令(命令())
    if (command.length === 0) return "请填写启动命令"
    const cwd = 目录().trim()
    return { transport: "local", command, ...(cwd === "" ? {} : { cwd }) }
  }

  const 能交 = () =>
    名称().trim() !== "" &&
    描述().trim() !== "" &&
    分类() !== "" &&
    !忙() &&
    !分类加载中() &&
    props.categories.length > 0

  async function 提交() {
    if (!能交()) return
    const 连接 = 连接规格()
    if (typeof 连接 === "string") {
      设失败话(连接)
      return
    }
    设忙(true)
    设失败话(undefined)
    let 回话: string | undefined
    try {
      回话 = await props.onCreate({
        name: 名称().trim(),
        description: 描述().trim(),
        category: 分类(),
        visibility: 可见(),
        connection: 连接,
      })
    } finally {
      设忙(false)
    }
    if (typeof 回话 === "string") 设失败话(回话)
  }

  return (
    <form
      data-slot="mcp-create-form"
      class="flex flex-col gap-3"
      onSubmit={(event) => {
        event.preventDefault()
        void 提交()
      }}
    >
      <label class="flex flex-col gap-1">
        <span class="text-[12px] text-v2-text-text-muted">名称</span>
        <TextInputV2
          autofocus
          appearance="large"
          class="!w-full"
          name="mcp-create-name"
          aria-label="MCP 名称"
          value={名称()}
          onInput={(event) => {
            设名称(event.currentTarget.value)
            清错误()
          }}
        />
      </label>

      <label class="flex flex-col gap-1">
        <span class="text-[12px] text-v2-text-text-muted">功能描述（必填）</span>
        <TextareaV2
          class="!w-full"
          name="mcp-create-description"
          aria-label="功能描述"
          rows={2}
          value={描述()}
          onInput={(event) => {
            设描述(event.currentTarget.value)
            清错误()
          }}
        />
      </label>

      <label class="flex flex-col gap-1">
        <span class="text-[12px] text-v2-text-text-muted">分类</span>
        <select
          data-slot="mcp-create-category"
          name="mcp-create-category"
          aria-label="分类"
          disabled={忙()}
          value={分类()}
          onChange={(event) => {
            设分类(event.currentTarget.value)
            清错误()
          }}
          class="h-8 rounded-[6px] border border-v2-border-border-muted bg-v2-background-bg-layer-02 px-2 text-[13px] text-v2-text-text-base"
        >
          <option value="">请选择分类</option>
          <For each={props.categories}>{(行) => <option value={行.id}>{行.name}</option>}</For>
        </select>
        <Show when={分类读不到()}>
          <p data-slot="mcp-create-category-error" class="text-[11px] text-v2-state-fg-warning">
            暂时读不到分类：{props.categoriesError}
          </p>
        </Show>
        <Show when={分类加载中()}>
          <p data-slot="mcp-create-category-loading" class="text-[11px] text-v2-text-text-muted">
            正在读取分类…
          </p>
        </Show>
        <Show when={没有分类()}>
          <p data-slot="mcp-create-no-category" class="text-[11px] text-v2-text-text-muted">
            还没有可选的分类；请先在分类管理里启用至少一个分类。
          </p>
        </Show>
      </label>

      <label class="flex flex-col gap-1">
        <span class="text-[12px] text-v2-text-text-muted">可见范围</span>

          data-slot="mcp-create-visibility"

/**
 * MCP 连接规格的只读客户端（2026-11-02 设计 §3.1.2 / §5.2）。
 *
 * 与服务端类型同形：`McpConnectionView` 从 `asset-service` **只做类型导入**（不把内核
 * 运行时代码带进前端）。严格解析每一列——服务端契约变了要报「数据格式不对」，
 * 不能把缺字段静默当成空值（那会让「没试连」与「字段丢了」长得一样）。
 */
import type { McpConnectionView } from "@opencode-ai/asset-service/mcp-connection"

export type McpConnectionOutcome<T> =
  | { readonly kind: "ok"; readonly value: T }
  | { readonly kind: "rejected"; readonly message: string }
  | { readonly kind: "failed"; readonly message: string }

export type McpConnectionFetch = (input: string, init?: RequestInit) => Promise<Response>

export const TIMEOUT_MS = 10_000

export const PATH = {
  connection: (assetId: string) => `/openhive/assets/${encodeURIComponent(assetId)}/connection`,
} as const

async function 默认发送(input: string, init?: RequestInit): Promise<Response> {
  const 控制器 = new AbortController()
  const 定时器 = setTimeout(() => 控制器.abort(), TIMEOUT_MS)
  try {
    return await fetch(input, { ...init, signal: 控制器.signal })
  } finally {
    clearTimeout(定时器)
  }
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value)
}

function 数(value: unknown): value is number {
  return typeof value === "number" && Number.isFinite(value)
}

function 可空数(value: unknown): number | null | undefined {
  if (value === null) return null
  return 数(value) ? value : undefined
}

/** `checkOk` 只能是 `boolean | null`；`undefined`（缺字段）也判坏。 */
function 可空布尔(value: unknown): value is boolean | null {
  return value === null || typeof value === "boolean"
}

function 字符串表(value: unknown): Record<string, string> | undefined {
  if (!isRecord(value)) return undefined
  const out: Record<string, string> = {}
  for (const [key, item] of Object.entries(value)) {
    if (typeof item !== "string") return undefined
    out[key] = item
  }
  return out
}

function 连接(value: unknown): McpConnectionView | undefined {
  if (!isRecord(value)) return undefined
  const { assetId, transport, url, command, cwd, enabled, checkError } = value
  if (typeof assetId !== "string" || (transport !== "local" && transport !== "remote")) return undefined
  if (url !== null && typeof url !== "string") return undefined
  if (command !== null && (!Array.isArray(command) || !command.every((part) => typeof part === "string"))) {
    return undefined
  }
  if (cwd !== null && typeof cwd !== "string") return undefined
  const headers = 字符串表(value["headers"])
  const environment = 字符串表(value["environment"])
  if (headers === undefined || environment === undefined) return undefined
  if (typeof enabled !== "boolean") return undefined
  if (typeof checkError !== "string") return undefined
  if (!可空布尔(value["checkOk"])) return undefined
  const timeoutMs = 可空数(value["timeoutMs"])
  const checkedAt = 可空数(value["checkedAt"])
  const toolCount = 可空数(value["toolCount"])
  if (timeoutMs === undefined || checkedAt === undefined || toolCount === undefined) return undefined
  const createdAt = value["createdAt"]
  const updatedAt = value["updatedAt"]
  if (!数(createdAt) || !数(updatedAt)) return undefined
  return {
    assetId,
    transport,
    url,
    command: command === null ? null : (command as string[]),
    cwd,
    headers,
    environment,
    enabled,
    timeoutMs,
    checkedAt,
    checkOk: value["checkOk"] as boolean | null,
    checkError,
    toolCount,
    createdAt,
    updatedAt,
  }
}

async function 发送并解析(
  send: McpConnectionFetch,
  input: string,
): Promise<McpConnectionOutcome<McpConnectionView>> {
  let response: Response
  try {
    response = await send(input)
  } catch {
    return { kind: "failed", message: "连接服务失败，请检查网络后重试" }
  }

  let body: unknown
  try {
    body = await response.json()
  } catch {
    body = undefined
  }
  const message =
    isRecord(body) && typeof body["error"] === "string" && body["error"].length > 0
      ? body["error"]
      : `操作失败（HTTP ${response.status}）`
  if (response.status === 401) return { kind: "rejected", message: "登录状态已失效，请重新登录" }
  if (response.status >= 500) return { kind: "failed", message }
  if (!response.ok) return { kind: "rejected", message }
  const parsed = 连接(body)
  if (parsed === undefined) return { kind: "failed", message: "服务返回的数据格式不对" }
  return { kind: "ok", value: parsed }
}

export interface McpConnectionApi {
  get(assetId: string): Promise<McpConnectionOutcome<McpConnectionView>>
}

export function createMcpConnectionApi(send: McpConnectionFetch = 默认发送): McpConnectionApi {
  return {
    get: (assetId) => 发送并解析(send, PATH.connection(assetId)),
  }
}
