# 中栏文件浏览与分屏 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让左栏文件树能打开文件到中栏（右键「打开」/ 双击 ⇒ 一张文件一个 tab，读前问大小、超阈值弹窗提醒），并让中栏能手动分成左右 / 上下两栏、每栏各有自己的一套 tab 与内容。

**Architecture:** 三段各自独立、层层可测：① **后端**加一个只读元信息的 `GET /openhive/file/stat`（第 8 个出口，与 `download` 同一串守卫）；② **客户端取数**在 `project/openhive-files.ts` 加 `statProjectFile` / `readProjectFile`，再用一个模块级常量的 loader 注入 `layout-new.tsx`（该文件只加 1 行 import ＋ 1 个 prop）；③ **闸门**是纯模块 `center/open-file.ts`（依赖全注入 ⇒ 裸单测），弹窗是它的一个实现；④ **分屏**改的是本项目自有的 `center/*` 数据模型（`panes` 两栏），UI 用上游的 `ResizeHandle`。

**Tech Stack:** Bun monorepo · SolidJS 1.9.10（**不是 React**）· Effect 4（后端路由）· happy-dom + `bun test`（组件/单测）· 上游 `ResizeHandle`（`@opencode-ai/ui/resize-handle`）· v2 `DialogV2` / `ButtonV2` / `TooltipV2`。

**Spec:** `docs/superpowers/specs/2026-10-12-中栏文件浏览与分屏-design.md`

```ts
 * 第八个出口 `stat`（2026-10-12）：**只回大小，不读内容**。
 *
 * 这一组要钉的是**它不比 `download` 松**：两者是同一条链上的两个出口，`stat` 松一档
 * 就等于给「读之前先问一句」这道闸门开了个旁路（`LEARNINGS #004-01`：数出口要数
 * 「同一个判断散在几处」，而不是数「我已经写了几处」）。故下面逐条对着 `download`
 * 那一组来，同样的输入要得同样的 400。
 *
 * ⚠️ 文案常量在这里**重新写字面量**、不从源码 import（`LEARNINGS #003-05`）：
 * 它们是跨层的契约，测试里写死才能发现「生产把文案改了」。
 */
describe("openhive 文件出口 · stat（读前问大小）", () => {
  const 项目头 = (project: string) => ({ [PROJECT_HEADER]: project })
  const 问大小 = (project: string, path: string) =>
    `${出口.stat}?${new URLSearchParams({ path })}`

  // 与 `handleStat` / `handleDownload` 的文案逐字相同；测试里是**独立的一份**（见上）。
  const NO_PROJECT = "这次文件操作没有指明项目（缺少 x-openhive-project 头）"
  const GIT_LOCKED = "项目里的 .git 是版本库的元数据，不许查看或改动"
  const LINK_ESCAPE = "这个位置经过链接指向了项目外面（链接不算项目内的路径）"
  const 没这个文件 = "文件不存在"
  const 越界 = "路径越出项目目录"

  const Size = Schema.Struct({ size: Schema.Number })

  it("普通文件 ⇒ 200 且回的是**真实字节数**", () =>
    Effect.gen(function* () {
      yield* 就绪()
      const response = yield* as(ALICE, 问大小(ALPHA, "a.txt"), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(200)
      const body = yield* json(Size, response)
      // 与**盘上现在这一份**比，而不是与一个写死的数字比：夹具内容是活的。
      expect(body.size).toBe(statSync(文件.甲).size)
      expect(body.size).toBeGreaterThan(0)
    }))

  it("目录 ⇒ 400「文件不存在」（闸门只对文件说话）", () =>
    Effect.gen(function* () {
      yield* 就绪()
      const response = yield* as(ALICE, 问大小(ALPHA, "sub"), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(400)
      expect((yield* json(Rejected, response)).error).toBe(没这个文件)
    }))

  it("不存在的路径 ⇒ 400「文件不存在」", () =>
    Effect.gen(function* () {
      yield* 就绪()
      const response = yield* as(ALICE, 问大小(ALPHA, "没有这个.txt"), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(400)
      expect((yield* json(Rejected, response)).error).toBe(没这个文件)
    }))

  it("越界（兄弟项目里真实存在的那份）⇒ 400「路径越出项目目录」", () =>
    Effect.gen(function* () {
      yield* 就绪()
      // 目标**真实存在**：否则这条用例在「文件本来就没有」上假绿（同 download 越界那条的口径）。
      expect(existsSync(文件.密)).toBe(true)
      const response = yield* as(ALICE, 问大小(ALPHA, 文件.密相对), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(400)
      expect((yield* json(Rejected, response)).error).toBe(越界)
    }))

  it("最后一段是文件链接 ⇒ 400「文件不存在」（`lstat` 不跟链接看目标）", () =>
    Effect.gen(function* () {
      yield* 就绪()
      yield* 造链接()
      const response = yield* as(ALICE, 问大小(ALPHA, "链到密件.txt"), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(400)
      expect((yield* json(Rejected, response)).error).toBe(没这个文件)
    }))

  it("中间一层是目录链接 ⇒ 400 LINK_ESCAPE 那句", () =>
    Effect.gen(function* () {
      yield* 就绪()
      yield* 造链接()
      const response = yield* as(ALICE, 问大小(ALPHA, "link/密件.txt"), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(400)
      expect((yield* json(Rejected, response)).error).toBe(LINK_ESCAPE)
    }))

  it(".git 里的文件 ⇒ 400 GIT_LOCKED 那句", () =>
    Effect.gen(function* () {
      yield* 就绪()
      // ⚠️ 前提：项目根里得**真有** `.git`，`inGit()` 才拦得住。夹具的 `重置文件()` 造没造它
      // 是本条唯一的未知数 ⇒ 动手时先看 `download` 那一组里 GIT_LOCKED 那条是怎么造前提的，
      // **照它做**（`#004-12`：同族 harness 整段抄）。若那一组也没有这条用例，就在 `重置文件()`
      // 里补 `mkdirSync(path.join(项目根, ".git"), { recursive: true })` 并把这个前提**写进
      // 本用例第一行显式造出来**——不要靠「别处顺手造过」（`#005-36`：借来的前提会静默过期）。
      yield* Effect.promise(() => mkdirSync(path.join(项目根, ".git"), { recursive: true }))
      const response = yield* as(ALICE, 问大小(ALPHA, ".git/config"), { headers: 项目头(ALPHA) })
      expect(response.status).toBe(400)
      // 注：`.git` 目录里没有 `config` 这个文件，但 `handleStat` 在 `inGit` 那一步就返回了
      // —— 比「文件不存在」更早 ⇒ 得到的是 GIT_LOCKED 而不是「文件不存在」。这条判据的射程
      // 正是**守卫的先后顺序**（`#004-09`：「被拒」与「门在副作用之前」是两条独立判据）。
      expect((yield* json(Rejected, response)).error).toBe(GIT_LOCKED)
    }))

  it("不带头 ⇒ 400，且一个字节都没回（够不着任何项目）", () =>
    Effect.gen(function* () {
      yield* 就绪()
      const response = yield* as(ALICE, 问大小(ALPHA, "a.txt"))
      expect(response.status).toBe(400)
      expect((yield* json(Rejected, response)).error).toBe(NO_PROJECT)
      // 「没回大小」这一条也要钉：回体里不许混进一个 size 字段。
      expect(JSON.stringify(yield* json(Schema.Unknown, response))).not.toContain("size")
    }))
})
```

```ts
 *
 * `gate` → 取 `path` → `inside` → `inGit` → `lstat`（不是 `stat`）→ `realInside`。
 * 两处判据必须同样严：这两条是同一个项目目录上的两个出口，松一档的那个就是别人的旁路
 * （`LEARNINGS #004-09`：「被拒」与「门在副作用之前」是两条独立判据，这里连「拒绝」
 * 本身都要两条出口同口径；`#004-01`：同一件事散在 N 个出口上时，逐个数、别只数写的那个）。
 *
 * ⚠️ 回应里**只有大小**，不带内容、不带类型、不带头像时间：本出口的用途单一，
 * 多回一样就多一个「以后有人拿它当 list 用」的口子。
 */
function handleStat(request: HttpServerRequest.HttpServerRequest, root: string) {
  return Effect.gen(function* () {
    const project = yield* gate(request, root)
    if (typeof project !== "string") return project

    const path = new URL(request.url, "http://localhost").searchParams.get("path")
    if (path === null) return badRequest("缺少 path")

    const target = inside(project, path)
    if (target === undefined) return badRequest("路径越出项目目录")
    if (inGit(project, target)) return badRequest(GIT_LOCKED)

    // `lstat` 而非 `stat`：最后一段是链接时 `stat` 会跟着看目标（同 `handleDownload` 那条注释
    // 实测过的同一件事）⇒ 跟着看就等于把项目外那个文件的**大小**漏出去。
    const info = yield* Effect.promise(() => lstat(target).catch(() => undefined))
    if (info === undefined || !info.isFile()) return badRequest("文件不存在")
    if (!(yield* realInside(project, dirname(target)))) return badRequest(LINK_ESCAPE)

    return HttpServerResponse.jsonUnsafe({ size: info.size })
  })
}
```

```ts
/**
 * 打开文件前的**闸门**（2026-10-12 / FR-005）：从「点了打开」到「真的开一张 tab」之间，
 * 问一句大小；太大了先问用户一声。
 *
 * ## 为什么是纯模块，而不是写在 `workspace-entry.tsx` 里
 *
 * 四件事（已经在受焦栏里开着、问大小、问用户、开 tab）各有各的失败方式，而**它们的分支顺序
 * 就是这条需求本身**：先看开没开、再问大小、问不到就放行、太大了才问用户。写成纯模块
 * ⇒ 11 条判据一个组件、一次外呼都不用起（`open-file.test.ts`）。依赖全注入，与
 * `tab-store.ts` 是同一个范式。
 *
 * ## 闸门是「提醒」，不是「关卡」（设计文档 §4.2 ③）
 *
 * `stat` 问不到（没身份 / 网络错 / 服务端那边出了别的岔子）时**放行**：闸门挡的是
 * 「文件太大、界面可能卡」这一件**我们有把握的事**；对没把握的事拦人，等于把网络抖动
 * 变成「打不开文件」。放行之后真卡了，用户还有浏览器自己那条出路。
 *
 * ## 阈值为什么分两档
 *
 * 表格那一档低（5 MB）是因为 SheetJS 要把**整个工作簿模型**在内存里铺开，同样的字节数
 * 比 PDF 贵得多；其余格式（PDF / 图片 / 文本）5 MB 与 10 MB 之间没有这条额外开销。
 * 两个数都是**用户 2026-10-12 裁定的**。
 */

import { contentTabKey, type ContentTab } from "./tab-store"
import { 表格扩展名 } from "./views"
import { extensionOf } from "./view-registry"

/** 闸门的两档阈值（字节）。用户 2026-10-12 裁定：**表格 5 MB，其余 10 MB**。 */
export const 大小闸门 = {
  表格: 5 * 1024 * 1024,
  其余: 10 * 1024 * 1024,
} as const

/** 弹窗要说的两件事：给谁看的名字，和**真实的大小**（不说「很大」这种形容词）。 */
export interface 打开提醒 {
  /** 文件名（含扩展名）。 */
  文件: string
  /** 字节数。 */
  字节: number
}

/** 问用户一句。`true` ＝ 仍然打开。 */
export type 问用户 = (提醒: 打开提醒) => Promise<boolean>

/** 一栏的读写口。`已有` 是**那一刻的快照**，不是 getter——判定与动作必须看同一份。 */
export interface 栏操作 {
  /** 这一栏里已经开着的 tab key（`contentTabKey` 形态）。 */
  已有: readonly string[]
  切到: (key: string) => void
  开: (tab: ContentTab) => void
}

export interface 打开文件的依赖 {
  /** 新 tab 的**来源模块**（`center.module()`），FR-005 的着色与 tab 身份依据。 */
  module: () => string
  /**
   * **新 tab 要落进哪一栏**（受焦栏）。
   *
   * ⚠️ 一次调用里**只读一次**（`createOpenFile` 开头那个 `栏` 常量）：判「已经在栏里开着」、
   * 切过去、开一张，三件事必须落在**同一栏**。分成三个函数各读一次 `activePane()`，
   * 读到两次就可能落在两栏——而那种缺陷不报错、不变红（`LEARNINGS #002-06`）。
   */
  受焦栏: () => 栏操作
  /** 问服务端这个文件多大；`undefined` ＝ **问不到** ⇒ 放行。 */
  stat: (path: string) => Promise<number | undefined>
  /** 超阈值时问用户。 */
  dialog: 问用户
}

/** 某个路径走哪一档阈值。判据与 `view-registry` 认视图时**同一个** `extensionOf`（不另写一份）。 */
function 阈值(path: string): number {
  const 扩展名 = extensionOf(path)
  // 大小写已经在 `extensionOf` 里归一（它回 `.csv` 这种小写形态）。
  return 扩展名 !== undefined && 表格扩展名.includes(扩展名) ? 大小闸门.表格 : 大小闸门.其余
}

/**
 * 造一个「打开文件」的动作。返回值是异步的：问大小与问用户都要等。
 */
export function createOpenFile(依赖: 打开文件的依赖) {
  return async function 打开文件(文件: { path: string; 标题: string }): Promise<void> {
    const 栏 = 依赖.受焦栏()
    const tab: ContentTab = { module: 依赖.module(), title: 文件.标题, path: 文件.path }
    const key = contentTabKey(tab)

    // ① 已经在**这一栏**开着 ⇒ 切过去就完事：内容早就读过，既不问大小也不弹窗。
    if (栏.已有.includes(key)) {
      栏.切到(key)
      return
    }

    // ② 问大小。问不到 ⇒ 放行（`undefined` 与「很小」在这里走同一条路，这是刻意的）。
    const 字节 = await 依赖.stat(文件.path)
    if (字节 !== undefined && 字节 > 阈值(文件.path)) {
      const 要开 = await 依赖.dialog({ 文件: 文件.标题, 字节 })
      // ③ 用户说不了（取消 / 关掉弹窗）⇒ 一张 tab 都不开。**取消就是取消**。
      if (!要开) return
    }

    栏.开(tab)
  }
}
```

```tsx
import { onCleanup, type JSX } from "solid-js"
import type { 问用户 } from "./open-file"

/** 「87 MB」——四舍五入到整数兆。 */
export function 多少兆(字节: number): string {
  return `${Math.round(字节 / (1024 * 1024))} MB`
}

export interface BigFileDialogProps {
  /** 文件名（含扩展名）——弹窗一盖，树上那一行就不在眼前了。 */
  文件: string
  /** 字节数（真实值）。 */
  字节: number
  /** 交出去的答案。**恰好调一次**：两颗钮各一次，或关闭时兜底一次。 */
  on答: (仍然打开: boolean) => void
}

export function BigFileDialog(props: BigFileDialogProps): JSX.Element {
  const dialog = useDialog()
  let 已答 = false

  const 答 = (仍然打开: boolean) => {
    if (已答) return
    已答 = true
    // 先收弹窗再喊回调：回调那侧会开一张 tab / 继续走闸门，弹窗不该再多留一会儿。
    dialog.close()
    props.on答(仍然打开)
  }

  // 关掉（Escape / 遮罩 / ×）＝ **取消**。理由见文件头。
  onCleanup(() => {
    if (!已答) props.on答(false)
  })

  return (
    <DialogV2 class="w-[min(26rem,calc(100vw-3rem))]">
      <div data-slot="big-file-dialog" class="contents">
        <DialogHeader>
          <DialogTitle>这个文件较大</DialogTitle>
        </DialogHeader>
        <DialogBody class="flex w-full flex-col gap-2 px-4 pt-4 pb-2">
          <p data-slot="big-file-dialog-text" class="min-w-0 text-[13px] break-all text-v2-text-text-base">
            「{props.文件}」有 <span class="font-semibold">{多少兆(props.字节)}</span>
            ，直接打开可能要等一会儿。仍然打开吗？
          </p>
        </DialogBody>
        <DialogFooter>
          {/* `autofocus` 落在**取消**上（同 `file-delete-dialog.tsx` 的取向）：默认答案是不开。
              `DialogV2` 的 `onOpenAutoFocus` 按 `[autofocus]` 找。 */}

            取消
          </ButtonV2>
          <ButtonV2 data-slot="big-file-open" type="button" variant="contrast" onClick={() => 答(true)}>
            仍然打开
          </ButtonV2>
        </DialogFooter>
      </div>
    </DialogV2>
  )
}
```

```ts
  /** 本栏当前激活 tab 的 key（`contentTabKey` 形态）。 */
  active?: string
}

/** 中栏共享 tab 状态（001 的单栏 ＋ 2026-10-12 的分屏）。 */
export interface CenterTabState {
  /** 恒两栏。未分屏时右栏为空且不露（`split` 说了算，不看它有没有 tab）。 */
  panes: [PaneState, PaneState]
  /**
   * 右栏露不露。
   *
   * **显式存**、不从 `panes` 推：**收起**只把右栏藏起来，它那些 tab 留着
   * （设计文档 §3.1 决定 2）——「有没有 tab」与「露不露」是两件事。
   */
  split: boolean
  /**
   * 排列方向：`"左右"` ＝ 并排（竖着一条隔板）、`"上下"` ＝ 叠放（横着一条隔板）。
   *
   * ⚠️ 这两个词**只在交付给 `ResizeHandle` 那一行**翻译成 `horizontal` / `vertical`
   * （`center-split.tsx`），别在本层预先翻（spec §6）。
   */
  排列: "左右" | "上下"
  /** 新开的 tab 落哪一栏 ＝ 哪一栏受焦。未分屏时恒 `0`。 */
  activePane: 0 | 1
  /** 当前**停留**的模块（左栏所在的那个）——切它 MUST NOT 动 tabs（FR-006）。 */
  module?: string
}

/**
 * 空状态。**函数而不是常量**：它是每份状态的起点，共用一个对象等于让所有起点共享同一对
 * `panes` 数组（今天没有一处改它，但那是运气，不是保证）。
 */
export function 空中栏(): CenterTabState {
  return { panes: [{ tabs: [] }, { tabs: [] }], split: false, 排列: "左右", activePane: 0 }
}
```

```ts
/** 换掉某一栏。**只动那一栏**——另一栏必须是**原对象**（`===` 不变，视图才不重挂）。 */
function 换栏(current: CenterTabState, pane: 0 | 1, 下一栏: PaneState): CenterTabState {
  const panes: [PaneState, PaneState] = pane === 0 ? [下一栏, current.panes[1]] : [current.panes[0], 下一栏]
  return { ...current, panes }
}

/**
 * `!split` ⇒ 受焦栏归 0。
 *
 * 不收敛的后果是**静默**的：右栏没露，而新文件开进了它 ⇒ 用户点了「打开」，
 * 屏幕上什么都不发生（`LEARNINGS #006-27` 那类「认不出来就不发」的温床）。
 * 每一处会改动 `activePane` 的函数都要过这一道。
 */
function 收敛(current: CenterTabState): CenterTabState {
  return current.split || current.activePane === 0 ? current : { ...current, activePane: 0 }
}
```

三个既有 reducer（语义一字不改，只多一层「在哪一栏」）：

```ts
export function activateContentTab(current: CenterTabState, key: string, pane: 0 | 1 = current.activePane): CenterTabState {
  return 收敛({ ...换栏(current, pane, { ...current.panes[pane], active: key }), activePane: pane })
}
```

> ⚠️ **`activate` / `close` 里那句 `activePane: pane` 要不要留？** 留。理由：这两个动作**只能**由在那一栏里的点击/按键触发（tab 栏长在那一栏里），所以「动了哪一栏 ⇒ 哪一栏受焦」是**同一件事实的另一种说法**，不是第二份判断。而 `open` 不同（它由**左栏文件树**触发，那不在任何一栏里）⇒ `open` **不改**受焦，落进哪栏由 `activePane` 决定。（若日后出现「键盘切栏」这类非点击路径，`focusPane` 就是它要调的入口。）

```ts
export function openContentTab(current: CenterTabState, tab: ContentTab, pane: 0 | 1 = current.activePane): CenterTabState {
  const key = contentTabKey(tab)
  const 栏 = current.panes[pane]
  // 已经开着就把它切过去（认的是 `contentTabKey`）；否则追加到末尾。无论哪种，打开都意味着切过去。
  const opened: PaneState = 栏.tabs.some((item) => contentTabKey(item) === key) ? 栏 : { ...栏, tabs: [...栏.tabs, tab] }
  return 收敛({ ...换栏(current, pane, { ...opened, active: key }), activePane: pane })
}

export function closeContentTab(current: CenterTabState, key: string, pane: 0 | 1 = current.activePane): CenterTabState {
  const 栏 = current.panes[pane]
  const index = 栏.tabs.findIndex((item) => contentTabKey(item) === key)
  const tabs = 栏.tabs.filter((item) => contentTabKey(item) !== key)
  if (栏.active !== key) return 收敛({ ...换栏(current, pane, { ...栏, tabs }), activePane: pane })
  const heir = tabs[index - 1] ?? tabs[index]
  return 收敛({ ...换栏(current, pane, { ...栏, tabs, active: heir ? contentTabKey(heir) : undefined }), activePane: pane })
}
```

分屏四个新动作：

```ts
/** 分屏：**不动任何 tab**（谁在哪栏、谁激活，一字不改），只把右栏露出来。 */
export function splitPane(current: CenterTabState, 排列: "左右" | "上下"): CenterTabState {
  return { ...current, split: true, 排列 }
}

/** 收起：右栏**藏起来但节点不摘**（`center-split.tsx` 用 `display:none`），它那些 tab 留着。 */
export function unsplitPane(current: CenterTabState): CenterTabState {
  return { ...current, split: false, activePane: 0 }
}

/** 受焦栏改到这一栏（两栏各自的 `onMouseDown` 喊它）。未分屏时收敛回 0。 */
export function focusPane(current: CenterTabState, pane: 0 | 1): CenterTabState {
  return 收敛({ ...current, activePane: pane })
}

/**
 * 清空两栏（**换项目**时喊，2026-10-12 追加）。
 *
 * 为什么必须有：左栏那棵树是**当前项目**的文件，而中栏一张 tab 绑的是一个**路径**。
 * 换了项目之后，旧 tab 的路径在新项目里指向另一样东西（或者什么都不指向）——那看起来像
 * **串项目**，与 `workspace-entry.tsx` 里文件清单那条 effect 的注释（「留着上一个项目的
 * 文件更糟」）是同一个判断。
 *
 * `排列` 留着（下次分屏还是用户上次选的那个方向）；`split` 归 `false`——换完项目回到单栏，
 * 让两栏空着（其中一栏还看不见）是没道理的状态。
 */
export function clearContentTabs(current: CenterTabState): CenterTabState {
  return { ...current, panes: [{ tabs: [] }, { tabs: [] }], split: false, activePane: 0 }
}
```

```tsx
/**
 * 中栏的两栏（2026-10-12 / 中栏分屏）：左边/上边一栏、右边/下边一栏，各带自己的一套
 * tab 栏与内容区，中间一条可拖的隔板。
 *
 * ## 四条决定（每条都是踩过的坑换来的，别在这里重新论证）
 *
 * **① 包裹的边界 ＝ 那一栏的边界。** 每栏外面一层 `relative` 盒子，手柄放在**那一栏自己的
 * 盒子里**；盒子**不能** `overflow-hidden` —— 上游手柄是 `width:8px` ＋ `translateX(±50%)`，
 * 一半探到栏外去盖住那条缝；盒子一旦裁切，缝上就什么都没有了（`three-pane.tsx:134-148`
 * 记着这条是怎么量出来的）。
 *
 * **② 右栏第一次分屏时才挂，挂上就永不摘。** 收起时用 `display:none` 藏起来、**不摘节点**：
 * 摘了再挂会重跑那一栏的 `CenterContent` ⇒ 文件重读、PDF 回到第一页（001 T013 的教训）。
 * 而**首次挂载推迟到第一次分屏**，换来的是「从未分屏过时 DOM 与本组件引入前逐字相同」
 * ——既有测试与真栈布局零风险。（设计文档 §3.1 决定 2 说的是「`!split` 时右栏 keep mounted」；
 * 这里收窄成「挂过一次就不摘」，目的相同、既有面更小。）
 *
 * **③ 方向那两个词只在这一行翻。** 状态里叫 `排列: "左右" | "上下"`，`ResizeHandle` 认的是
 * `horizontal` / `vertical`。翻译**只出现一次**（spec §6）——多一处就多一处会各翻各的地方。
 *
 * **④ 受焦靠手势给。** 两栏各自 `onMouseDown` ⇒ `center.focusPane(p)`。这是 `activePane`
 * **唯一**的写入口（`tab-store.ts` 的 `focusPane`）：新开的文件落进受焦栏，而左栏文件树的
 * 点击**不在任何一栏里**，所以「我看着哪一栏」只能由栏内的手势说了算。
 */

import { ResizeHandle } from "@opencode-ai/ui/resize-handle"
import { createEffect, createSignal, Show, type JSX } from "solid-js"
import { CenterContent } from "./center-content"
import type { LoadFileContent } from "./file-content"
import { TabBar } from "./tab-bar"
import { useCenterTabs } from "./tab-context"
import type { ViewRegistry } from "./view-registry"

/** 分屏后左/上那栏的**初始**尺寸（px）。真栈上用户第一件事就是拖到自己要的位置。 */
const 左右默认 = 480
const 上下默认 = 360
/** 两栏共同的下限：再小就看不清一行字了。 */
const 栏最小 = 200

/**
 * 上限按视口算（同 `three-pane.tsx` 的 `leftMax` / `rightMax`）——**拖到只剩一条缝**是
 * 用户做得出来的一件事，而它做出来之后没有任何办法恢复。
 * `Math.max(栏最小, …)` 兜住「视口量不到」（happy-dom 的 `innerWidth` 可能是 0）：
 * 让 `max < min` 递给 `ResizeHandle` 会得到「一拖就归零」。
 */
const 左右最大 = () => Math.max(栏最小, Math.round(window.innerWidth * 0.8))
const 上下最大 = () => Math.max(栏最小, Math.round(window.innerHeight * 0.8))

export interface CenterSplitProps {
  registry: ViewRegistry
  /** 中栏视图的取数出口（生产走 `LOAD_PROJECT_FILE`）。 */
  load?: LoadFileContent
  /** 调用方页面此刻可不可以露（同 `CenterContent`，只有左/上那一栏承载它）。 */
  pageVisible?: boolean
  /** 「页面不露、也没有激活 tab」时的替身（只有左/上那一栏用）。 */
  empty?: JSX.Element
  /** 上游路由页面（**只在左/上那一栏**）。 */
  children?: JSX.Element
}

export function CenterSplit(props: CenterSplitProps): JSX.Element {
  const center = useCenterTabs()
  const [左右宽, set左右宽] = createSignal(左右默认)
  const [上下高, set上下高] = createSignal(上下默认)
  /** 右栏挂没挂过（一旦为真就是永真，见文件头 ②）。 */
  const [右栏挂过, set右栏挂过] = createSignal(false)
  createEffect(() => {
    if (center.split()) set右栏挂过(true)
  })

  const 是左右 = () => center.排列() === "左右"
  const 容器 = () =>
    `flex w-full min-h-0 min-w-0 flex-1 items-stretch ${是左右() ? "flex-row" : "flex-col"}`
  /** 左/上那一栏：尺寸由信号给，故 `shrink-0`。 */
  const 头栏 = () => `relative flex min-h-0 min-w-0 shrink-0 flex-col items-start ${是左右() ? "h-full" : "w-full"}`
  /** 右/下那一栏：吃剩下那块。 */
  const 尾栏 = () => `flex min-h-0 min-w-0 flex-1 flex-col items-start ${是左右() ? "h-full" : "w-full"}`

  return (
    <div data-slot="center-split" class={容器()}>
      <div
        data-slot="center-pane"
        data-pane="0"
        class={头栏()}
        style={是左右() ? { width: `${左右宽()}px` } : { height: `${上下高()}px` }}
        onMouseDown={() => center.focusPane(0)}
      >
        <TabBar
          tabs={center.pane(0).tabs}
          active={center.pane(0).active}
          onActivate={(key) => center.activate(key, 0)}
          onClose={(key) => center.close(key, 0)}
          focused={center.split() && center.activePane() === 0}
          onSplit={(排列) => center.splitPane(排列)}
        />
        <CenterContent
          pane={0}
          registry={props.registry}
          load={props.load}
          pageVisible={props.pageVisible}
          empty={props.empty}
        >
          {props.children}
        </CenterContent>
        <Show when={center.split()}>
          {/* ⚠️ 方向那两个词的**唯一**翻译处（文件头 ③）。`edge="end"`：
              左右分屏时手柄在下栏的左边缘、上下分屏时在上栏的下边缘（`resize-handle.css`）。
              颜色从本侧给：上游 `::after` 没有 `background`，不传就 hover 也全透明
              （同 `three-pane.tsx` 的 `手柄悬停色`）。 */}
          <ResizeHandle
            direction={是左右() ? "horizontal" : "vertical"}
            edge="end"
            size={是左右() ? 左右宽() : 上下高()}
            min={栏最小}
            max={是左右() ? 左右最大() : 上下最大()}
            onResize={是左右() ? set左右宽 : set上下高}
            class="after:bg-v2-border-border-strong"
          />
        </Show>
      </div>
      <Show when={右栏挂过()}>
        <div
          data-slot="center-pane"
          data-pane="1"
          class={尾栏()}
          // `display:none` 而不是不渲染（文件头 ②）。
          style={{ display: center.split() ? undefined : "none" }}
          onMouseDown={() => center.focusPane(1)}
        >
          <TabBar
            tabs={center.pane(1).tabs}
            active={center.pane(1).active}
            onActivate={(key) => center.activate(key, 1)}
            onClose={(key) => center.close(key, 1)}
            focused={center.split() && center.activePane() === 1}
            onUnsplit={() => center.unsplitPane()}
          />
          {/* ⚠️ 右栏**不承载** `children`（上游路由页面）：页面是「当前这条路」的那一页，
              画两份就是同一页在屏幕上出现两遍（同 `CenterContent` 里 `pageVisible` 那条注释
              防的是同一件事）。也不给 `empty`：那一栏空着就是空着，不必两处都说同一句话。 */}
          <CenterContent pane={1} registry={props.registry} load={props.load} empty={undefined} />
        </div>
      </Show>
    </div>
  )
}
```
