# 卡片模板编辑器·第一刀（内核）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把「卡片模板」从一张只有内置行的表，做成可创建 / 编辑 / 预览 / 停用 / 转正的通用翻译引擎：用户建模板 → 解析时自动或指定选模板 → 生成卡片 → 未验证模板产出的卡片默认对 AI 隐藏。

**Architecture:** 三份受控词表（字段 / 角色 / 实体类型）与模板定义校验放在既有 `case-card-template.ts`；模板行的读写另起 `case-card-template-store.ts`（DAO）；通用翻译引擎另起 `case-template-translate.ts`（纯函数，行 → 卡片草稿），复用 `case-translate.ts` 的资金通路判据（空行 / 汇总行 / 时间 / 归一化）；`case-ingest.ts` 按表选模板并合并翻译结果；HTTP 六口挂在 `case-ui-routes.ts`；AI 出口用一条共享 SQL 片段 `模板可见条件` 在三个 MCP 工具与时间线口统一过滤。

**Tech Stack:** Bun 1.4.x monorepo、TypeScript、`bun:test`、oxlint（typeAware）、Drizzle ORM（`db.execute(sql\`…\`)`）、Effect（HTTP 路由）、PGlite / 真 PG（测试夹具 `startProductionDb`）。

**Spec:** `docs/superpowers/specs/2026-11-09-事实卡片抽象/2026-11-14-P4-卡片模板-内核-design.md`

export interface 槽条目 {

export function 槽条目们(槽位: 模板槽位, 槽名: 槽): 槽条目[] {
  return 拆填法(槽名, 槽位[槽名]).条目.flatMap((填) => {
    if (!是识别字段(填.字段)) return []
    const 一条: 槽条目 = { 字段: 填.字段 }
    if (typeof 填.角色 === "string") 一条.角色 = 填.角色
    if (typeof 填.实体类型 === "string") 一条.实体类型 = 填.实体类型
    return [一条]
  })
}

/** 模板 JSON（库 / HTTP）→ 合法形状；畸形条目丢掉，不抛。写入口已先校验，这里只是收窄。 */
export function 归一槽位(槽位: unknown): 模板槽位 {
  const 出: { [K in 槽]?: 槽填法[] } = {}
  if (!是记录(槽位)) return 出
  for (const 槽名 of 九格们) {
    const 填法们: 槽填法[] = []
    for (const 填 of 拆填法(槽名, 槽位[槽名]).条目) {
      if (!是识别字段(填.字段)) continue
      const 一条: 槽填法 = { 字段: 填.字段 }
      if (typeof 填.角色 === "string") 一条.角色 = 填.角色
      if (typeof 填.实体类型 === "string") 一条.实体类型 = 填.实体类型
      填法们.push(一条)
    }
    if (填法们.length > 0) 出[槽名] = 填法们
  }
  return 出
}

function 是槽(值: unknown): 值 is 槽 {
  return typeof 值 === "string" && 槽集.has(值)

function 规范化必填(required: unknown): { 值: 槽[]; 错: string[] } {
  if (!Array.isArray(required)) return { 值: [], 错: ["required 必须是槽名数组"] }
  const 错: string[] = []
  const 集 = new Set<槽>(["谁"])
  // 2026-11-14 评审回写：§6.3 #9 —— required 不得含「其他」、不得重复。
  const 见过 = new Set<槽>()
  for (const 项 of required) {
    if (!是槽(项)) {
      错.push(`required 里的「${String(项)}」不是合法槽名；可用：${九格们.join("、")}`)
      continue
    }
    if (项 === "其他") {
      错.push("「其他」不能作为必填槽")
      continue
    }
    if (见过.has(项)) {
      错.push(`required 里的槽「${项}」重复了`)
      continue
    }
    见过.add(项)
    集.add(项)
  }
  // 谁 恒必填；按九格顺序输出，保证同样的定义永远得到同样的 required（可比对、可去重）。
  return { 值: 九格们.filter((槽名) => 集.has(槽名)), 错 }
}

/** 角色 / 实体类型这两条（§6.3 第 7、8 条）。2026-11-14 评审回写：角色判定复用导出的 `是角色`。 */
function 校验角色与实体(槽名: 槽, 填: Record<string, unknown>): string[] {
  const 角色 = 填.角色
  const 实体类型 = 填.实体类型
  if (槽名 === "其他") {
    return 角色 !== undefined || 实体类型 !== undefined ? ["「其他」不收角色与实体类型"] : []
  }
  const 错: string[] = []
  if (角色 !== undefined) {
    if (!是角色(角色)) {
      错.push(`「${槽名}」的角色 ${JSON.stringify(角色)} 不在角色词表里`)
    } else if ((槽名 === "谁" || 槽名 === "和谁") && !人称角色集.has(角色)) {
      错.push(`「${槽名}」的角色只能用：${人称角色们.join("、")}`)
    } else if (槽名 === "何地" && !地点角色集.has(角色)) {
      错.push(`「何地」的角色只能用：${地点角色们.join("、")}`)
    } else if (槽名 !== "谁" && 槽名 !== "和谁" && 槽名 !== "何地") {
      错.push(`「${槽名}」不收角色（只有 谁 / 和谁 / 何地 能定角色）`)
    }
  }
  if (实体类型 !== undefined) {
    if (!(typeof 实体类型 === "string" && 实体类型集.has(实体类型))) {
      错.push(`「${槽名}」的实体类型 ${JSON.stringify(实体类型)} 不在实体类型词表里`)
    } else if (槽名 === "何地" && 实体类型 !== "地点") {
      错.push("「何地」的实体类型必须是「地点」")
    }
  }
  return 错
}

const 单值槽: ReadonlySet<string> = new Set(["谁", "何时", "何时止", "多少钱"])
const 模板填法上限 = 100

/**
 * 模板定义的完整校验（§6.3 十条）。返回错误清单（空 = 通过）+ 规范化结果。
 *
 * `校验模板定义` 是本函数的薄包装（HTTP 只关心错误）；DAO 用 `值` 落库 ⇒ 校验与收窄**同一份**
 * （`#002-06`：两处各写一遍就是下一处漂移）。
 */
export function 检查模板定义(定义: 模板定义输入): { 错: string[]; 值?: 模板定义 } {
  const 错: string[] = []
  const name = 定义.name.trim()
  if (name === "" || name.length > 200) 错.push("模板名不能为空且不超过 200 字")

  const eventType = 定义.eventType.trim()
  if (!是事件类型(eventType)) 错.push(`未知的事件类型：${定义.eventType}`)
  else if (eventType === 资金交易 || eventType === "住宿") {
    错.push("资金交易由方向算法负责、住宿由区间合并负责，不能建用户模板")
  }

  const slots = 定义.slots
  错.push(...校验槽位种类(slots))
  if (是记录(slots)) {
    for (const 键 of Object.keys(slots)) {
      if (!槽集.has(键)) 错.push(`未知槽名「${键}」；可用：${九格们.join("、")}`)
    }
  }

  const 字段计数 = new Map<string, number>()
  let 总条数 = 0
  for (const 槽名 of 九格们) {
    const 条目 = 是记录(slots) ? 拆填法(槽名, slots[槽名]).条目 : []
    总条数 += 条目.length
    if (单值槽.has(槽名) && 条目.length > 1) 错.push(`「${槽名}」最多 1 条填法`)
    for (const 填 of 条目) {
      if (typeof 填.字段 === "string") 字段计数.set(填.字段, (字段计数.get(填.字段) ?? 0) + 1)
      错.push(...校验角色与实体(槽名, 填))
    }
  }
  for (const [字段, 数] of 字段计数) {
    if (数 > 1) 错.push(`字段 ${字段} 在模板里出现了 ${数} 次；一个字段只能落一个槽`)
  }
  if (总条数 > 模板填法上限) 错.push(`模板填法共 ${总条数} 条，超过上限 ${模板填法上限}`)

  const 必填 = 规范化必填(定义.required)
  错.push(...必填.错)
  if (必填.错.length === 0) {
    for (const 槽名 of 必填.值) {
      const 条目 = 是记录(slots) ? 拆填法(槽名, slots[槽名]).条目 : []
      if (条目.length === 0) 错.push(`必填槽「${槽名}」还没有填法`)
    }
  }

  if (错.length > 0) return { 错 }
  // 上面 `!是事件类型` 已 push 过错误 ⇒ 走到这里 eventType 必定合法；这一行只为类型收窄。
  if (!是事件类型(eventType)) return { 错: [`未知的事件类型：${定义.eventType}`] }
  return { 错: [], 值: { name, event_type: eventType, slots: 归一槽位(slots), required: 必填.值 } }
}

/** HTTP 层只看错误（§6.3）。 */
export function 校验模板定义(定义: 模板定义输入): string[] {
  return 检查模板定义(定义).错
}

export * as OpenhiveCaseCardTemplateStore from "./case-card-template-store"

import { sql } from "drizzle-orm"
import { pgErrorCode } from "@opencode-ai/auth/pg-errors"
import { rowsOf } from "@opencode-ai/auth/migrate"
import { nowSeconds } from "@opencode-ai/auth/time"
import { 文本数组 } from "./case-fact-store"
import { 九格们, 归一槽位, 是用户模板id, 读模板, type 槽, type 模板行, type 模板定义 } from "./case-card-template"
import { 是事件类型 } from "./case-event-types"
import type { Queryable } from "./case-store"

/**
 * 模板 DAO（§7）。判定（能不能改、能不能转正）放在这里，路由只翻错误码。
 *
 * 三个不变量：
 * 1. **内置行不可变**：`内置:` 前缀一律 `immutable`，代码规则不归用户改；
 * 2. **停用保留行**：`disabled_at` 只写一次，历史卡片照旧可追溯；
 * 3. **改内容才升版本**：`slots` / `required` / `event_type` 任一变化 ⇒ `version+1` 且
 *    `verified_at` 清空——验证只管「这一版」，改过的模板必须重新预览转正。
 */

export const 模板错误码 = ["bad_request", "conflict", "immutable", "not_found", "state_changed"] as const
export type 模板错误码 = (typeof 模板错误码)[number]

export class 模板错误 extends Error {
  readonly code: 模板错误码
  constructor(code: 模板错误码, message: string) {
    super(message)
    this.name = "模板错误"
    this.code = code
  }
}

/** jsonb / text[] 的读回形状随驱动而变（`#002-01`）：字符串就解析，对象直接用。 */
function 读json(值: unknown): unknown {
  if (typeof 值 !== "string") return 值
  try {
    const 解: unknown = JSON.parse(值)
    return 解
  } catch {
    return undefined
  }
}

function 解析必填(值: unknown): 槽[] {
  if (Array.isArray(值)) return 九格们.filter((槽名) => 值.includes(槽名))
  if (typeof 值 === "string") {
    // PG text[] 的文本形态 `{谁,何时}`（某些驱动/日志路径会给这种）。
    const 项们 = 值.replace(/^\{|\}$/g, "").split(",")
    return 九格们.filter((槽名) => 项们.includes(槽名))
  }
  return ["谁"]
}

function 映射模板行(行: Record<string, unknown>): 模板行 {
  const eventType = 行.event_type
  if (!是事件类型(eventType)) throw new Error(`模板的事件类型不在词表里：${String(eventType)}`)
  return {
    id: String(行.id),
    name: String(行.name),
    event_type: eventType,
    slots: 归一槽位(读json(行.slots)),
    required: 解析必填(行.required),
    version: Number(行.version),
    verified_at: 行.verified_at === null || 行.verified_at === undefined ? null : Number(行.verified_at),
    hit_count: Number(行.hit_count),
    disabled_at: 行.disabled_at === null || 行.disabled_at === undefined ? null : Number(行.disabled_at),
  }
}

const 列 = sql`id, name, event_type, slots, required, version, verified_at, hit_count, disabled_at`

/** 非用户前缀：只有 `内置:` 算「代码规则维护」，其余 id 一律不存在。 */
const 内置模板前缀 = "内置:"

/**
 * 列表：内置恒在最前（前端把它当「不可编辑的参照」读），其后按创建时间。
 * 停用行默认不列——「停用」在民警那边就是「没有了」（历史卡片不受影响）。
 */
export async function 列出模板(
  db: Queryable,
  input: { projectId: string; 含停用?: boolean },
): Promise<模板行[]> {
  const 行们 = rowsOf(
    await db.execute(sql`
      select ${列}
        from auth.case_card_template
       where project_id = ${input.projectId}
         and (${input.含停用 === true} or disabled_at is null)
       order by case when id like '内置:%' then 0 else 1 end, created_at asc, id asc
    `),
  )
  return 行们.map(映射模板行)
}

/** 单行（编辑器回显 / 解析选模板 / 转正前读 version 共用）。 */
export async function 读模板行(db: Queryable, projectId: string, id: string): Promise<模板行 | undefined> {
  const 行 = await 读模板(db, projectId, id)
  return 行 === undefined ? undefined : 映射模板行(行)
}

export async function 建模板(db: Queryable, input: { projectId: string; 定义: 模板定义 }): Promise<模板行> {
  const id = `用户:${crypto.randomUUID()}`
  const 结果 = await db
    .execute(sql`
      insert into auth.case_card_template
        (id, project_id, name, event_type, slots, required, version, verified_at, hit_count, created_at, disabled_at)
      values (${id}, ${input.projectId}, ${input.定义.name}, ${input.定义.event_type},
              ${JSON.stringify(input.定义.slots)}::text::jsonb, ${文本数组(input.定义.required)},
              1, null, 0, ${nowSeconds()}, null)
      returning ${列}
    `)
    .catch((错误: unknown) => {
      // 部分唯一索引撞车（23505）：同类型已有一个启用中的用户模板。
      // 预检 SELECT 挡不住并发，唯一索引才是真闸；这里把 DB 错误翻成人能懂的话。
      if (pgErrorCode(错误) === "23505") {
        throw new 模板错误("conflict", "同类型下已有启用中的用户模板；请先停用旧模板再新建")
      }
      throw 错误
    })
  const 行 = rowsOf(结果)[0]
  if (行 === undefined) throw new Error("建模板后没有返回行")
  return 映射模板行(行)
}

/**
 * 改模板。**`verified_at` 只在内容真变时清空**（§7.3）：
 * 只改名字不重置验证状态——名字不影响翻译结果，重置只会让民警白跑一遍预览。
 */
export async function 改模板(
  db: Queryable,
  input: { projectId: string; id: string; 定义: 模板定义 },
): Promise<模板行> {
  要求用户模板(input.id, "修改")
  const 现有 = await 读模板行(db, input.projectId, input.id)
  if (现有 === undefined || 现有.disabled_at !== null) throw new 模板错误("not_found", "没有这个模板")

  const slots = sql`${JSON.stringify(input.定义.slots)}::text::jsonb`
  const required = 文本数组(input.定义.required)
  const eventType = sql`${input.定义.event_type}`
  const 变了 = sql`(slots is distinct from ${slots} or required is distinct from ${required} or event_type is distinct from ${eventType})`

  const 行们 = rowsOf(
    await db.execute(sql`
      update auth.case_card_template
         set name = ${input.定义.name},
             slots = ${slots},
             required = ${required},
             event_type = ${eventType},
             version = version + case when ${变了} then 1 else 0 end,
             verified_at = case when ${变了} then null else verified_at end
       where project_id = ${input.projectId} and id = ${input.id} and disabled_at is null
       returning ${列}
    `),
  )
  const 行 = 行们[0]
  if (行 === undefined) throw new 模板错误("not_found", "没有这个模板")
  return 映射模板行(行)
}

/** 停用：保留行。返回是否真的停了（已停用 / 不存在 ⇒ false）。 */
export async function 停用模板(db: Queryable, input: { projectId: string; id: string }): Promise<boolean> {
  要求用户模板(input.id, "停用")
  const 现有 = await 读模板行(db, input.projectId, input.id)
  if (现有 === undefined) throw new 模板错误("not_found", "没有这个模板")
  if (现有.disabled_at !== null) return false
  const 行们 = rowsOf(
    await db.execute(sql`
      update auth.case_card_template
         set disabled_at = ${nowSeconds()}
       where project_id = ${input.projectId} and id = ${input.id} and disabled_at is null
       returning id
    `),
  )
  return 行们.length > 0
}

/**
 * 转正（§10.4）：`expectedVersion` 是预览时读到的版本 —— 预览与转正之间模板被改过 ⇒
 * `state_changed`，让用户重新预览。已转正的返回原时间（幂等，路由另有 `already` 标记）。
 */
export async function 转正模板(
  db: Queryable,
  input: { projectId: string; id: string; expectedVersion: number },
): Promise<{ verified_at: number; version: number }> {
  要求用户模板(input.id, "转正")
  const 现有 = await 读模板行(db, input.projectId, input.id)
  if (现有 === undefined || 现有.disabled_at !== null) throw new 模板错误("not_found", "没有这个模板")
  if (现有.verified_at !== null) return { verified_at: 现有.verified_at, version: 现有.version }

  const 行们 = rowsOf(
    await db.execute(sql`
      update auth.case_card_template
         set verified_at = ${nowSeconds()}
       where project_id = ${input.projectId} and id = ${input.id}
         and verified_at is null and disabled_at is null and version = ${input.expectedVersion}
       returning verified_at, version
    `),
  )
  const 行 = 行们[0]
  if (行 === undefined) throw new 模板错误("state_changed", "模板在预览期间被修改，请重新预览")
  return { verified_at: Number(行.verified_at), version: Number(行.version) }
}

/** 解析命中计数：只认**用户**模板（内置行不参与编辑器热度），去重、跳过不存在。 */
export async function 记模板命中(db: Queryable, input: { projectId: string; ids: readonly string[] }): Promise<void> {
  const 用户id们 = [...new Set(input.ids.filter(是用户模板id))]
  if (用户id们.length === 0) return
  await db.execute(sql`
    update auth.case_card_template
       set hit_count = hit_count + 1
     where project_id = ${input.projectId} and id = any(${文本数组(用户id们)})
  `)
}

/** 可编辑性闸：内置 ⇒ immutable；非用户前缀且非内置 ⇒ not_found。 */
function 要求用户模板(id: string, 动作: string): void {
  if (是用户模板id(id)) return
  if (id.startsWith(内置模板前缀)) throw new 模板错误("immutable", `内置模板由代码规则维护，不能${动作}`)
  throw new 模板错误("not_found", "没有这个模板")
}

/** 未保存草稿的伪模板行：卡片 id 的槽位用固定字面量「预览」，永不落库（§10.3）。 */
export const 预览模板id = "预览"

export function 草稿模板行(定义: 模板定义): 模板行 {
  return {
    id: 预览模板id,
    name: 定义.name,
    event_type: 定义.event_type,
    slots: 定义.slots,
    required: 定义.required,
    version: 0,
    verified_at: null,
    hit_count: 0,
    disabled_at: null,
  }
}

import { 默认角色, 是角色, 槽条目们, type 槽条目, type 角色, type 模板行 } from "./case-card-template"

import {
  卡片id,
  汇总行原因,
  是空行,
  文本值,
  type 跳过原因,
  type 卡片草稿,
  type 卡片参与者,

import type { FactRowInput } from "./case-store"

/**
 * 通用翻译引擎（§8）：模板 + 明细行 → 卡片草稿。**纯函数**，不碰库、不读时钟、不调模型。
 *
 * 资金通路（`翻译资金`）继续走自己的算法：方向、户名兜底、备注事实都是代码规则，
 * 模板只负责「表格 + 用户配置」这一类（§4 偏差 3）。
 */

export interface 翻译模板输入 {
  projectId: string
  materialId: string
  template: 模板行
  rows: readonly FactRowInput[]
}

const 摘要上限 = 200
const 实体类型集: ReadonlySet<string> = new Set(实体类型们)

function 是实体类型(值: string | undefined): 值 is 实体类型 {
  return 值 !== undefined && 实体类型集.has(值)
}

/** 单元格原文。标准列取已解析/归一化值；`startedAt`/账号走原文旁路（`01:800`）。 */
function 原值(row: FactRowInput, 字段: 识别字段): string {
  if (字段 === "startedAt") return row.startedRaw ?? ""
  if (字段 === "ownerAccount") return row.ownerAccountRaw ?? row.ownerAccount ?? ""
  if (字段 === "counterparty") return row.counterpartyRaw ?? row.counterparty ?? ""
  if (是标准列(字段)) {
    const 值 = row[字段]
    return 值 === null || 值 === undefined ? "" : String(值)
  }
  const 值 = row.extras?.[字段]
  return typeof 值 === "string" ? 值 : ""
}

function 有值(row: FactRowInput, 字段: 识别字段): boolean {
  return 原值(row, 字段).trim() !== ""
}

interface 取时结果 {
  值: number | null
  原文: string
  精度: "到日" | "到分" | "到秒" | null
}

/** 时间槽取值：`startedAt` 直接用已解析结果（含精度），其他字段走 `parseTime`。 */
function 取时间(row: FactRowInput, 条目们: readonly 槽条目[]): 取时结果 {
  const 填 = 条目们.find((条目) => 有值(row, 条目.字段))
  if (填 === undefined) return { 值: null, 原文: "", 精度: null }
  if (填.字段 === "startedAt") {
    return { 值: row.startedAt ?? null, 原文: row.startedRaw ?? "", 精度: row.precision ?? null }
  }
  const 原文 = 原值(row, 填.字段)
  const 解析 = parseTime(原文)
  return { 值: 解析?.seconds ?? null, 原文, 精度: 解析?.precision ?? null }
}

/** 字段种类 → 实体类型（§8.2 的表）。`其他` 字段（地址/商户/姓名…）按此表推导。 */
function 按种类推导(字段: 识别字段): 实体类型 {
  switch (字段目录[字段].种类) {
    case "账号":
      return "账户"
    case "号码":
      return "号码"
    case "商户":
      return "组织"
    case "地点":
      return "地点"
    default:
      return "人"
  }
}

/** 槽位没写角色（或写了词表外的东西）⇒ 用默认角色。`是角色` 是唯一判定（`#002-06`）。 */
function 取角色(给: string | undefined, 缺省: 角色): 角色 {
  return 是角色(给) ? 给 : 缺省
}

function 造参与者(row: FactRowInput, 填: 槽条目, ordinal: number, 角色: 角色, 强制类型?: 实体类型): 卡片参与者 {
  const 原文 = 原值(row, 填.字段)
  const 规范值 = 归一化值(填.字段, 原文)
  return {
    规范值,
    角色,
    提及: 原文,
    ordinal,
    实体: 规范值 !== "",
    实体类型: 强制类型 ?? (是实体类型(填.实体类型) ? 填.实体类型 : 按种类推导(填.字段)),
  }
}

function 玩家(参与者: 卡片参与者): string {
  return 参与者.提及.trim() === "" ? 参与者.规范值 : 参与者.提及
}

function 拼摘要(input: {
  type: string
  主体: 卡片参与者
  和谁们: readonly 卡片参与者[]
  何时: 取时结果
  何时止: 取时结果
  何地们: readonly 卡片参与者[]
  金额: string | null
  币种: string
  attrs: Record<string, unknown>
}): string {
  let 摘要 = `${input.type}：${玩家(input.主体)}`
  if (input.和谁们.length > 0) 摘要 += ` 与 ${input.和谁们.map(玩家).join("、")}`
  if (input.何时.原文 !== "") 摘要 += `，${input.何时.原文}`
  if (input.何时止.原文 !== "") 摘要 += ` 至 ${input.何时止.原文}`
  if (input.何地们.length > 0) 摘要 += `，${input.何地们.map(玩家).join("、")}`
  if (input.金额 !== null) 摘要 += `，${input.金额} ${input.币种}`
  for (const [键, 值] of Object.entries(input.attrs)) 摘要 += `，${键} ${String(值)}`
  if (摘要.length > 摘要上限) 摘要 = `${摘要.slice(0, 摘要上限)}…`
  return 摘要
}

export function 翻译模板(input: 翻译模板输入): 翻译结果 {
  const 卡片们: 卡片草稿[] = []
  const 跳过们: { sheet: string; rowNo: number; 原因: 跳过原因 }[] = []
  for (const row of input.rows) {
    const 原因 = 逐行(input, row, 卡片们)
    if (原因 !== undefined) 跳过们.push({ sheet: row.sheet, rowNo: row.rowNo, 原因 })
  }
  // 模板卡只产卡片：别名（户名挂账户）与实体属性属于资金通路，本引擎不产（§4 偏差 4）。
  return { 卡片们, 别名们: [], 属性们: [], 跳过们 }
}

function 逐行(input: 翻译模板输入, row: FactRowInput, 卡片们: 卡片草稿[]): 跳过原因 | undefined {
  if (是空行(row)) return "空行"
  const 汇总 = 汇总行原因(row)
  if (汇总 !== null) return 汇总

  const { slots, required, event_type: type } = input.template
  for (const 槽名 of required) {
    if (!槽条目们(slots, 槽名).some((条目) => 有值(row, 条目.字段))) return "必填槽为空"
  }

  const 何时 = 取时间(row, 槽条目们(slots, "何时"))
  if (何时.原文 !== "" && 何时.值 === null) return "时间无法解析"
  const 何时止 = 取时间(row, 槽条目们(slots, "何时止"))
  if (何时止.原文 !== "" && 何时止.值 === null) return "时间无法解析"
  if (何时.值 !== null && 何时止.值 !== null && 何时止.值 < 何时.值) return "时间区间倒置"

  const 主体填 = 槽条目们(slots, "谁").find((条目) => 有值(row, 条目.字段))
  if (主体填 === undefined) return "必填槽为空" // 谁 恒必填（校验层保证），这里只是兜底
  const 主体 = 造参与者(row, 主体填, 0, 取角色(主体填.角色, 默认角色.谁))

  const 和谁们: 卡片参与者[] = []
  for (const 填 of 槽条目们(slots, "和谁")) {
    if (!有值(row, 填.字段)) continue
    和谁们.push(造参与者(row, 填, 1 + 和谁们.length, 取角色(填.角色, 默认角色.和谁)))
  }
  const 何地们: 卡片参与者[] = []
  for (const 填 of 槽条目们(slots, "何地")) {
    if (!有值(row, 填.字段)) continue
    何地们.push(造参与者(row, 填, 1 + 和谁们.length + 何地们.length, 取角色(填.角色, 默认角色.何地), "地点"))
  }
  const 参与者们: 卡片参与者[] = [主体, ...和谁们, ...何地们]

  const 金额填 = 槽条目们(slots, "多少钱").find((条目) => 有值(row, 条目.字段))
  const 金额 = 金额填 === undefined ? null : 原值(row, 金额填.字段).trim().replace(/^-/, "")

  const attrs: Record<string, unknown> = {}
  for (const 填 of 槽条目们(slots, "其他")) {
    const 原文 = 原值(row, 填.字段)
    if (原文.trim() === "") continue
    attrs[字段显示名[填.字段]] = 填.字段 === "durationSec" ? (row.durationSec ?? 原文) : 原文
  }

  const 币种 = row.currency ?? "CNY"
  卡片们.push({
    id: 卡片id(input.projectId, input.materialId, row.sheet, row.rowNo, input.template.id),
    type,
    occurredAt: 何时.值,
    occurredRaw: 何时.原文 === "" ? null : 何时.原文,
    precision: 何时.精度,
    amount: 金额,
    currency: 币种,
    attrs,
    participants: [主体, ...和谁们, ...何地们],
    摘要: 拼摘要({ type, 主体, 和谁们, 何时, 何时止, 何地们, 金额, 币种, attrs }),
    sheet: row.sheet,
    rowNo: row.rowNo,
    用途: 判定用途({ 备注: null, 交易类型: null, 方向: null }),
    模板id: input.template.id,
    occurredEnd: 何时止.值,
  })
  return undefined
}

export interface 模板建议 {
  id: string
  name: string
  event_type: 事件类型
  命中槽数: number

const 建议最低命中 = 2

/**
 * 按表自动建议模板（§7.4）。只认启用中的**用户**模板：
 * 内置模板由材料类型直通、不参与打分（否则资金模板永远赢）。
 *
 * 三条判据缺一不可：①必填槽的字段全在映射里；②命中槽数 ≥ 2；③最高分唯一。
 * 并列就返回 `undefined`——猜错模板比不猜更糟（卡片会整批出现在错误的事件类型下）。
 */
export function 建议模板(
  模板们: readonly 模板行[],
  映射字段集: ReadonlySet<识别字段>,
): 模板建议 | undefined {
  const 候选: 模板建议[] = []
  for (const 模板 of 模板们) {
    if (模板.disabled_at !== null || !是用户模板id(模板.id)) continue
    const 命中必填 = 模板.required.every((槽名) =>
      槽条目们(模板.slots, 槽名).some((条目) => 映射字段集.has(条目.字段)),
    )
    if (!命中必填) continue
    const 命中槽数 = 九格们
      .flatMap((槽名) => 槽条目们(模板.slots, 槽名))
      .filter((条目) => 映射字段集.has(条目.字段)).length
    候选.push({ id: 模板.id, name: 模板.name, event_type: 模板.event_type, 命中槽数 })
  }
  if (候选.length === 0) return undefined
  候选.sort((甲, 乙) => 乙.命中槽数 - 甲.命中槽数)
  const 最高 = 候选[0]
  if (最高 === undefined || 最高.命中槽数 < 建议最低命中) return undefined
  if (候选.length > 1 && 候选[1]?.命中槽数 === 最高.命中槽数) return undefined
  return 最高
}

import * as XLSX from "xlsx"
import { claimNextJob, enqueueMaterial, runJob } from "../../src/server/openhive/case-ingest"
import { getMaterial } from "../../src/server/openhive/case-store"
import { 建模板 } from "../../src/server/openhive/case-card-template-store"
import { 检查模板定义 } from "../../src/server/openhive/case-card-template"

function 造多表工作簿(表们: [string, (string | number | null)[][]][]): Uint8Array {
  const wb = XLSX.utils.book_new()
  for (const [名, rows] of 表们) XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(rows), 名)
  const 输出: unknown = XLSX.write(wb, { type: "array", bookType: "xlsx" })
  if (!(输出 instanceof ArrayBuffer)) throw new Error("xlsx 没有返回 ArrayBuffer")
  return new Uint8Array(输出)
}

/** 一份「话单」：模板（谁=主叫号码、和谁=被叫号码、何时=通话时间、其他=通话时长）。 */
const 话单表 = [
  ["主叫号码", "被叫号码", "通话时间", "通话时长", "基站"],
  ["13800000001", "13900000002", "2023-11-14 22:13:20", "59", "杭州西湖"],
  ["13800000003", "13900000004", "2023-11-14 22:20:00", "12", "杭州滨江"],
  ["合计", "", "", "", ""],
]

function 话单定义() {
  const 检查 = 检查模板定义({
    name: "话单",
    eventType: "通话",
    slots: {
      谁: { 字段: "caller", 角色: "主叫" },
      和谁: [{ 字段: "callee", 角色: "被叫" }],
      何时: { 字段: "startedAt" },
      其他: [{ 字段: "durationSec" }],
    },
    required: ["何时"],
  })
  if (检查.值 === undefined) throw new Error(`夹具不合法：${检查.错.join("；")}`)
  return 检查.值
}

function 跑字节(项目: string, 名: string, 字节: Uint8Array) {
  const 文件 = new Map([[名, 字节]])
  const deps = {
    readFile: async (路径: string) => {
      const 内容 = 文件.get(路径)
      if (内容 === undefined) throw new Error(`读不到：${路径}`)
      return 内容
    },
    pg: () => db(),
    database: undefined as never,
  }
  return { deps, 字节 }
}

async function 跑一份(项目: string, 文件名: string, 字节: Uint8Array, templateId?: string) {
  const { deps } = 跑字节(项目, 文件名, 字节)
  await enqueueMaterial(db(), {
    projectId: 项目,
    path: 文件名,
    sourcePath: 文件名,
    fileName: 文件名,
    fingerprint: `fp-${项目}-${文件名}-${templateId ?? "auto"}`,
    ...(templateId === undefined ? {} : { templateId }),
  })
  const job = await claimNextJob(db())
  if (job === undefined) throw new Error("没有认领到作业")
  await runJob(job, deps as never)
  return job
}

/** 请求体 → 模板定义入参；形状不对 ⇒ undefined（400）。 */
function 读定义(body: Record<string, unknown>): { name: string; eventType: string; slots: unknown; required: unknown } | undefined {
  const name = body["name"]
  const eventType = body["eventType"]
  const slots = body["slots"]
  const required = body["required"]
  if (typeof name !== "string" || typeof eventType !== "string") return undefined
  if (typeof slots !== "object" || slots === null || Array.isArray(slots)) return undefined
  if (!Array.isArray(required) || required.some((项) => typeof 项 !== "string")) return undefined
  return { name, eventType, slots, required }
}

function 翻模板错误(错误: 模板错误) {
  if (错误.code === "not_found") return fail(404, 错误.message, "not_found")
  if (错误.code === "state_changed") return fail(409, 错误.message, "state_changed")
  return fail(错误.code === "bad_request" ? 400 : 409, 错误.message, 错误.code)
}

```ts
/** 预览输出（§10.3）：`卡片们` 只给编辑器要画的字段。 */
interface 预览输出 {
  卡片数: number
  跳过: Record<string, number>
  卡片们: ReturnType<typeof 预览卡>[]
  工作表们: { sheet: string; 卡片数: number; 跳过: Record<string, number> }[]
}

/** 可判别联合：成功 / 失败各带自己的字段，调用处 `if (!结果.ok)` 一次分流（不用 `as`）。 */
type 预览结果 = { ok: true; 预览: 预览输出 } | { ok: false; 状态: 404 | 409; code: string; error: string }

async function 跑预览(
  db: Queryable,
  input: { projectId: string; materialId: string; 模板: 模板行 },
): Promise<预览结果> {
  const 材料 = await getMaterial(db, input.projectId, input.materialId)
  if (材料 === undefined) return { ok: false, 状态: 404, code: "material_not_found", error: "找不到这份材料" }
  const 表们 = 预览表们(材料.parseDetail)
  if (材料.parseStatus !== "已解析" || 表们.length === 0 || 表们.some((表) => !表.有列)) {
    return {
      ok: false,
      状态: 409,
      code: "material_not_ready",
      error: "这份材料还没有可预览的解析结果（旧材料需要重新解析）",
    }
  }
  const 卡片们: ReturnType<typeof 预览卡>[] = []
  const 跳过合计: Record<string, number> = {}
  const 表回执们: { sheet: string; 卡片数: number; 跳过: Record<string, number> }[] = []
  for (const 表 of 表们) {
    const 行们 = await 读材料事实行(db, input.projectId, input.materialId, 表.sheet, 20)
    const 翻译 = 翻译模板({ projectId: input.projectId, materialId: input.materialId, template: input.模板, rows: 行们 })
    const 跳过: Record<string, number> = {}
    for (const 条 of 翻译.跳过们) 跳过[条.原因] = (跳过[条.原因] ?? 0) + 1
    for (const [原因, 数] of Object.entries(跳过)) 跳过合计[原因] = (跳过合计[原因] ?? 0) + 数
    表回执们.push({ sheet: 表.sheet, 卡片数: 翻译.卡片们.length, 跳过 })
    for (const 卡 of 翻译.卡片们) 卡片们.push(预览卡(卡))
  }
  return { ok: true, 预览: { 卡片数: 卡片们.length, 跳过: 跳过合计, 卡片们, 工作表们: 表回执们 } }
}

function 是记录值(值: unknown): 值 is Record<string, unknown> {
  return typeof 值 === "object" && 值 !== null && !Array.isArray(值)
}

/** parse_detail.mapping.工作表 的条目；`列` 是「这份材料可预览」的判据（§10.3）。 */
function 预览表们(detail: unknown): { sheet: string; 有列: boolean }[] {
  if (!是记录值(detail)) return []
  const mapping = detail["mapping"]
  if (!是记录值(mapping)) return []
  const 表 = mapping["工作表"]
  if (!是记录值(表)) return []
  return Object.entries(表).flatMap(([sheet, 值]) => (是记录值(值) ? [{ sheet, 有列: 是记录值(值["列"]) }] : []))
}

function handleCardTemplateVerify(request: HttpServerRequest.HttpServerRequest, deps: Deps) {
  return Effect.gen(function* () {
    const 用户 = yield* 身份

    if (typeof materialId !== "string" || materialId === "") return fail(400, "materialId 必填", "bad_request")
    const 期望版本 = body["expectedVersion"]
    if (期望版本 !== undefined && (typeof 期望版本 !== "number" || !Number.isInteger(期望版本))) {

    }

    const 门 = yield* 过门(deps, projectId, {
      要属主: true,

    if (!预览.ok) return fail(预览.状态, 预览.error, 预览.code)
    const { 卡片数, 跳过 } = 预览.预览
    if (模板.verified_at !== null) {
      return ok({ verified_at: 模板.verified_at, 卡片数, 跳过, already: true })
    }
    if (卡片数 === 0) return fail(400, "预览没有产出任何卡片：请检查必填槽与数据", "preview_failed")
    // 没给 expectedVersion 就用刚才读到的版本：「有卡可看」这件事已经发生过，版本只在

    const 结果 = yield* 跑模板(() =>

        projectId,
        id,

      }),
    )

    return ok({ verified_at: 结果.值.verified_at, 卡片数, 跳过 })
  })

`case-card-template.ts` 末尾追加（`sql` 已 import）：

```ts
/**
 * AI 出口的模板可见条件（§11.1）。
 *
 * 「未验证模板产出的卡片」默认对 AI 隐藏：模板是用户配的，预览没过就不该被当成事实回答。
 * **只允许在 `from auth.case_event`（未起别名）的查询里用**——片段裸引 `case_event`。
 *
 * 返回 `undefined` 表示不设条件。调用方**必须判空再 push**：把 `sql```（空片段）拼进
 * `where … and ${x}` 会产出 `and and` 语法错（`#002-06` 的反例）。
 *
 * `模板_id is null`（不存在这个模板行）⇒ `not exists` 为真 ⇒ 可见：历史卡片与内置卡照旧。
 */
export function 模板可见条件(含未验证: boolean | undefined): SQL | undefined {
  if (含未验证 === true) return undefined
  return sql`not exists (
    select 1 from auth.case_card_template t
     where t.project_id = case_event.project_id
       and t.id = case_event.template_id
       and t.verified_at is null)`
}
```

（`SQL` 类型从 `drizzle-orm` import：`import { sql, type SQL } from "drizzle-orm"`。）
