# 项目能力面板 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在项目左栏加第三个 tab「能力 N」，回答「我在这个项目里现在能用什么」——三组纵向排列（技能 / 知识库 / MCP），收起的组压成横条贴底，点哪条哪条上移展开。

**Architecture:** 后端在既有的 `GET /openhive/project/assets` 里，跑完对账后顺带读内核 `Skill.Service.available()`，把清单并进同一个回包（`skills` 字段）。前端拿这一份数据做三态合并（正常 / 没生效 / 本项目自带），渲染成三档手风琴。前端不碰沙箱路径，也不发第二个请求。

**Tech Stack:** Bun · TypeScript · Effect（后端路由）· SolidJS + happy-dom（前端组件）· Tailwind v4（样式类）

**Spec:** `docs/superpowers/specs/2026-10-27-项目能力面板-design.md`

## Global Constraints

- **不动上游文件**：本次改到的 `packages/opencode/src/server/openhive/project-assets.ts`、`packages/app/src/{project,ai-assets,workspace}` 下的文件全部是 openhive 自建；上游文件一个字不改。
- **不新增依赖**。
- **`bun test` 必须带 `timeout 300`**；退出码 124 不自动重跑。根目录禁止 `bun test`（其 script 直接 `exit 1`），一律在受影响 package 内跑。
- **不得 `git add -A` / `git add .`**，每次只 add 本任务列出的文件。
- **`undefined`（不知道）≠ 空数组 / `0`（明确没有）**：任何「读不到」都不得渲染成「没有」。
- **测试断言不用 `expect(节点).toBeNull()`**——被 Solid 渲染过的节点会让断言序列化时炸成挂死（`#009-01` / `#005-01`）。用布尔断言。

/**
 * 内核视角的一条技能。
 *
 * **不含 `content`**：`Skill.Info.content` 是 SKILL.md 全文，进 HTTP 回包既浪费又泄露。
 * `assetId` / `assetName` 只在它确实是库里的一份资产时才有——「本项目自带」的那条两者皆无，
 * 前端据此分第三态（不能给它画「停用」）。
 */
export interface 内核技能 {
  readonly name: string
  readonly description?: string
  readonly location: string
  readonly assetId?: string
  readonly assetName?: string
}

interface 状态基 {
  readonly enabled: readonly string[]
  readonly enabledCount: number
  readonly limit: number
}

/**
 * 把内核清单并进返回体。
 *
 * `内核 === undefined` ⇒ **整个 `skills` 键缺席**，不是空数组。空数组会被前端读成
 * 「内核一条都没看见」⇒ 全部技能标成「没生效」，而事实只是「这次没读到」（`#002-02`）。
 */
export function 带内核清单<T extends 状态基>(
  base: T,
  内核: readonly 内核技能[] | undefined,
): T & { skills?: readonly 内核技能[] } {
  if (内核 === undefined) return base
  return { ...base, skills: 内核 }
}
```

/** 从内核技能的 location 里取最后一级目录名（= assetId 或用户自己的目录名）。 */
function 目录名(location: string): string | undefined {
  const 段 = location.split(/[\\/]/).filter((s) => s.length > 0)
  // <…>/.opencode/(skill|skills)/<名>/SKILL.md —— 少一段就不认，宁可算「自带」
  if (段.length < 2) return undefined
  return 段[段.length - 2]
}

/**
 * 在算好的项目目录上读内核技能清单（`Skill.Service.available()`）。
 *
 * `store` 缺席（只挂路由子集的测试）⇒ 返回 `undefined`：**不知道**，不谎报为空。
 * 读失败同理——告警并返回 `undefined`，让前端说「暂时读不到」，而不是把每条技能标成没生效。
 */
function 读内核清单(deps: Deps, directory: string, 标题: ReadonlyMap<string, string>) {
  const store = deps.store
  if (Option.isNone(store)) return Effect.succeed<readonly 内核技能[] | undefined>(undefined)
  return store.value
    .provide(
      { directory },
      Effect.gen(function* () {
        const skill = yield* Skill.Service
        return yield* skill.available()
      }),
    )
    .pipe(
      Effect.map((list) =>
        list.map((info): 内核技能 => {
          const id = 目录名(info.location)
          const 名 = id === undefined ? undefined : 标题.get(id)
          return {
            name: info.name,
            ...(info.description === undefined ? {} : { description: info.description }),
            location: info.location,
            ...(id === undefined ? {} : { assetId: id }),
            ...(名 === undefined ? {} : { assetName: 名 }),
          }
        }),
      ),
      Effect.catchCause((cause) =>
        Effect.logWarning("读内核技能清单失败，本次不回报（不谎报为空）", { directory, cause }).pipe(
          Effect.as(undefined),
        ),
      ),
    )
}
```

import type { 内核技能 } from "@/ai-assets/enable-api"

/**
 * 「这个项目现在能用什么」的三态合并（设计 §4）。
 *
 * ## 三态是什么
 *
 * - `library` + `loaded: true` —— 库里有记录、内核也加载到了（正常）
 * - `library` + `loaded: false` —— 库里有、内核没看见（**调用前已对过账**，所以这是真问题）
 * - `local` —— 内核看得见、库里没有（用户自己在项目文件夹里写的）
 *
 * ## 为什么没有内核数据就返回 `undefined`
 *
 * 只有库侧数据时无法区分「没生效」与「只是没读到」。返回空数组会把这些技能全画成
 * 「没装东西」，那是假话（`#002-02`）。
 *
 * ## 顺序
 *
 * 库内条目按 `enabled` 的插入顺序（服务端的顺序），自带条目按内核返回顺序排在后面。
 */
export interface 库内条目 {
  readonly kind: "library"
  readonly assetId: string
  readonly name: string
  readonly description?: string
  readonly loaded: boolean
}

export interface 自带条目 {
  readonly kind: "local"
  readonly name: string
  readonly description?: string
}

export type 能力条目 = 库内条目 | 自带条目

export interface 合并输入 {
  readonly enabled: ReadonlySet<string>
  readonly skills: readonly 内核技能[] | undefined
}

export function 合并能力(输入: 合并输入): 能力条目[] | undefined {
  const 清单 = 输入.skills
  if (清单 === undefined) return undefined

  const 内核在某资产 = new Map<string, 内核技能>()
  const 自带: 自带条目[] = []
  for (const 条 of 清单) {
    if (条.assetId !== undefined && 输入.enabled.has(条.assetId)) 内核在某资产.set(条.assetId, 条)
    else 自带.push({ kind: "local", name: 条.name, ...(条.description === undefined ? {} : { description: 条.description }) })
  }

  const 库内: 库内条目[] = []
  for (const assetId of 输入.enabled) {
    const 命中 = 内核在某资产.get(assetId)
    库内.push({
      kind: "library",
      assetId,
      // 中文名优先取库里标题；「自带」那类没有 assetName，退回内核名。
      name: 命中?.assetName ?? 命中?.name ?? assetId,
      ...(命中?.description === undefined ? {} : { description: 命中.description }),
      loaded: 命中 !== undefined,
    })
  }

  return [...库内, ...自带]
}

/** tab 上的数字 = 库内条目数（含没生效的）。自带的不在账上，不数。 */
export function 计数(条目: readonly 能力条目[]): number {
  return 条目.filter((条) => 条.kind === "library").length
}

import { createSignal, For, Show, type JSX } from "solid-js"
import type { 内核技能 } from "@/ai-assets/enable-api"
import { 合并能力, 计数, type 能力条目 } from "./capability-source"

/**
 * 项目「能力」面板（`2026-10-27-项目能力面板-design.md`）。
 *
 * ## 三档手风琴
 *
 * 永远恰好一组展开。展开组**之前**的组收在顶部，**之后**的组滑到底部贴住。
 * 点已展开的组头 ⇒ 收起自己、回到「技能」展开（这条是兜底：否则会出现三组全收起的空面板）。
 *
 * ## 组头形态借自 MinIO 条（`minio-bar.tsx:130-154`）
 *
 * 13px、`muted → base` 的 hover、右侧箭头说明「再点一下会怎样」。**不借**它的默认收起：
 * 用户点这个 tab 就是来看内容的。
 *
 * ## 知识库永远不写数字
 *
 * 它不是「0 个」，是这个功能还没做——内核连它的工具映射都没有（`rbac.ts:117`）。
 * 写 `0` 等于替后端回答一个它回答不了的问题（`#002-02`）。
 */
export type 能力组 = "skill" | "knowledge" | "mcp"

const 组序: readonly 能力组[] = ["skill", "knowledge", "mcp"]
const 组名: Record<能力组, string> = { skill: "技能", knowledge: "知识库", mcp: "MCP" }

export interface CapabilityPanelProps {
  readonly hasProject: () => boolean
  readonly enabled: () => ReadonlySet<string>
  readonly skills: () => readonly 内核技能[] | undefined
  readonly error: () => string | undefined
  readonly busy: () => string | undefined
  readonly onToggle: (assetId: string) => void
  readonly onRetry: () => void
}

const 头 =
  "flex h-7 w-full shrink-0 cursor-pointer items-center gap-1.5 px-2 text-[13px] text-v2-text-text-muted hover:bg-v2-overlay-simple-overlay-hover hover:text-v2-text-text-base"
const 贴底头 = "border-t border-v2-border-border-muted " + 头
const 条目框 = "rounded-[6px] border border-v2-border-border-muted px-2 py-1.5"
const 小钮 = "rounded-[4px] px-1.5 text-[12px] text-v2-text-text-base hover:bg-v2-overlay-simple-overlay-hover"

export function CapabilityPanel(props: CapabilityPanelProps): JSX.Element {
  const [展开, set展开] = createSignal<能力组>("skill")

  const 条目 = () => 合并能力({ enabled: props.enabled(), skills: props.skills() })

  const 点组 = (组: 能力组) => {
    // 点已展开的组 ⇒ 回技能（设计 §3 规则 3）。点别的组 ⇒ 换它展开。
    set展开(组 === 展开() ? "skill" : 组)
  }

  const 组头 = (组: 能力组, 贴底: boolean) => {
    const 开着 = () => 展开() === 组
    return (
      <button
        type="button"
        data-slot={
          组 === "skill" ? "capability-group-skill" : 组 === "knowledge" ? "capability-group-knowledge" : "capability-group-mcp"
        }
        aria-expanded={开着()}
        class={贴底 ? 贴底头 : 头}
        onClick={() => 点组(组)}
      >
        <span aria-hidden="true">◆</span>
        <span class="font-semibold">{组名[组]}</span>
        <Show when={组 === "skill"}>
          <span data-slot="capability-count-skill" class="text-v2-text-text-muted">
            · {计数(条目() ?? []).toString()} 个
          </span>
        </Show>
        <Show when={组 === "mcp"}>
          <span data-slot="capability-count-mcp" class="text-v2-text-text-muted">
            · 0 个
          </span>
        </Show>
        <Show when={组 === "knowledge"}>
          <span class="text-v2-text-text-muted">即将支持</span>
        </Show>
        <span class="ml-auto" aria-hidden="true">
          {开着() ? "⌄" : "›"}
        </span>
      </button>
    )
  }

  const 一条 = (条: 能力条目) => {
    if (条.kind === "local")
      return (
        <div data-slot="capability-item-local" class={条目框}>
          <div class="flex items-center gap-1.5">
            <span aria-hidden="true">●</span>
            <span class="font-semibold">{条.name}</span>
            <span class="ml-auto text-[11px] text-v2-text-text-muted">📝 本项目自带</span>
          </div>
          <Show when={条.description}>
            {(说) => <p class="pl-3 text-[11px] text-v2-text-text-muted">{说()}</p>}
          </Show>
        </div>
      )

    return (
      <div data-slot={条.loaded ? "capability-item-loaded" : "capability-item-stale"} class={条目框}>
        <div class="flex items-center gap-1.5">
          <span aria-hidden="true">{条.loaded ? "●" : "▲"}</span>
          <span class="font-semibold">{条.name}</span>
          <span class="ml-auto text-[11px] text-v2-text-text-muted">📦 资产库</span>
        </div>
        <Show
          when={条.loaded}
          fallback={<p class="pl-3 text-[11px]">已启用，但 AI 没加载到——项目里的快捷方式可能丢了</p>}
        >
          <Show when={条.description}>{(说) => <p class="pl-3 text-[11px] text-v2-text-text-muted">{说()}</p>}</Show>
        </Show>
        <div class="pl-3 pt-1">
          <button
            type="button"
            data-slot="capability-disable"
            disabled={props.busy() === 条.assetId}
            class={小钮}
            onClick={() => props.onToggle(条.assetId)}
          >
            停用
          </button>
        </div>
      </div>
    )
  }

  return (
    <div data-component="capability-panel" class="flex min-h-0 w-full flex-1 flex-col">
      <Show when={props.hasProject()} fallback={<p data-slot="capability-no-project" class="px-2 py-3 text-[13px] text-v2-text-text-muted">先选一个项目</p>}>
        <Show when={props.error()}>
          {(说) => (
            <div data-slot="capability-error" class="px-2 py-1 text-[12px] text-v2-state-fg-danger">
              {说()}
            </div>
          )}
        </Show>

        {/* 展开组之前 / 本身 / 之后，按组序一次算出来 */}
        <For each={组序.filter((组) => 组序.indexOf(组) < 组序.indexOf(展开()))}>{(组) => 组头(组, false)}</For>

        {组头(展开(), false)}

        <div class="min-h-0 flex-1 overflow-y-auto px-1 py-1">
          <Show when={展开() === "skill"}>
            <Show
              when={条目() !== undefined}
              fallback={
                <div data-slot="capability-unreadable" class="px-1 py-2 text-[12px] text-v2-text-text-muted">
                  暂时读不到这个项目的技能。
                  <button type="button" class={小钮} onClick={() => props.onRetry()}>
                    重试
                  </button>
                </div>
              }
            >
              <Show
                when={条目()!.length > 0}
                fallback={<p class="px-1 py-2 text-[12px] text-v2-text-text-muted">这个项目还没有装任何技能。</p>}
              >
                <div class="flex flex-col gap-1.5">
                  <For each={条目()}>{(条) => 一条(条)}</For>
                </div>
              </Show>
            </Show>
          </Show>

          <Show when={展开() === "knowledge"}>
            <p class="px-1 py-2 text-[12px] text-v2-text-text-muted">知识库还没有接入，接入后会出现在这里。</p>
          </Show>

          <Show when={展开() === "mcp"}>
            <p class="px-1 py-2 text-[12px] text-v2-text-text-muted">
              这个项目还没有连任何 MCP。MCP 要由管理员在治理后台登记、授权给你，才会出现在这里。
            </p>
          </Show>
        </div>

        <For each={组序.filter((组) => 组序.indexOf(组) > 组序.indexOf(展开()))}>{(组) => 组头(组, true)}</For>
      </Show>
    </div>
  )
}
