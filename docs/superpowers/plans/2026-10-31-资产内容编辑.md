# 资产内容编辑与悬停文案 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给资产加上「功能描述」和「注意事项」两个可编辑字段，把左栏技能卡片的悬停文案改读功能描述（中文），并让作者能在详情页就地编辑介绍 / 功能描述 / 注意事项。

**Architecture:** 加两列（`summary` / `precautions`）走迁移回填；写入单一出口 `updateAssetContent()` 只碰内容列；HTTP 侧新增 `PATCH /openhive/assets/:id`，权限用既有的 `effectivePermissions().has("write")`；悬停取值在服务端完成（`映射内核清单` 优先取库里的 `summary`，空则回落 SKILL.md）；前端按 Tab 就地编辑。

**Tech Stack:** Bun 1.4.2 monorepo、Effect 4、Drizzle（原始 SQL 为主）、PostgreSQL、SolidJS 1.9。

**Spec:** `docs/superpowers/specs/2026-10-31-资产内容编辑-design.md`

```ts
/**
 * 单个内容字段的校验。
 *
 * - 不是字符串 ⇒ 报错（HTTP 来的 body 什么都可能是）；
 * - `description` 空串 ⇒ 报错（「介绍」是必填，卡片与详情都靠它）；`summary` / `precautions`
 *   允许空串（= 未填）；
 * - 超长 ⇒ 报错，消息里带上限值，前端能直接把这句话显示出来。
 */
export function validateContentField(field: AssetContentField, value: unknown): MetadataIssue | undefined {
  if (typeof value !== "string") return { field, message: "必须是字符串" }
  if (field === "description" && value.trim() === "") return { field, message: "不能为空" }
  const limit = CONTENT_LIMITS[field]
  if (value.length > limit) return { field, message: `不能超过 ${limit} 字` }
  return undefined
}
```

在 `asset.ts` 的 `updateAssetLifecycle` 之后追加：

```ts
/** 可编辑的内容字段（2026-10-31）。只这三列——生命周期列归 `updateAssetLifecycle`。 */
export type AssetContentPatch = Partial<Record<AssetContentField, string>>

/**
 * 写内容字段：`description` / `summary` / `precautions` 三个里带哪个改哪个。
 *
 * **只碰内容列 + `updated_at`**，一个字都不动 `status` / `version` / 可见范围 / 计数。
 * 反过来 `updateAssetLifecycle()` 也只碰生命周期列——两条写入路径互不覆盖（`asset.ts` 上方
 * 「名称 / 描述 / 标签等内容字段由编辑路径负责」那句，在这里兑现）。
 *
 * 用 `coalesce(占位符, 列)` 一次写完三列，而不是按字段拼动态 SQL：值来自 HTTP，
 * 拼列名要把字段名做成标识符，这里干脆三个字段全写、没带的传 `null` 让 `coalesce` 保留原值。
 * 空串是**值**不是 `null`，所以「清空功能描述」能落库。
 *
 * 校验在碰库之前（`AssetMetadataError`），不合法时一行都不写。
 */
export async function updateAssetContent(
  db: AssetTarget,
  id: string,
  patch: AssetContentPatch,
  now: number,
): Promise<AssetMetadata | undefined> {
  const issues: MetadataIssue[] = []
  for (const field of ["description", "summary", "precautions"] as const) {
    if (!(field in patch)) continue
    const 问题 = validateContentField(field, patch[field])
    if (问题 !== undefined) issues.push(问题)
  }
  if (issues.length > 0) throw new AssetMetadataError(issues)

  const rows = rowsOf(await db.execute(sql`
    update auth.asset set
      description = coalesce(${patch.description ?? null}, description),
      summary = coalesce(${patch.summary ?? null}, summary),
      precautions = coalesce(${patch.precautions ?? null}, precautions),
      updated_at = ${now}
    where id = ${id}
    returning ${COLUMNS}
  `))
  const [row] = rows
  return row === undefined ? undefined : toMetadata(row)
}
```

const FORBIDDEN = 403
const NOT_FOUND = 404

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value)
}

function 读取体(request: HttpServerRequest.HttpServerRequest) {
  return request.json.pipe(
    Effect.match({ onFailure: () => undefined, onSuccess: (value: unknown) => value }),
  )
}

/**
 * `PATCH /openhive/assets/:id`（2026-10-31）：改介绍 / 功能描述 / 注意事项。
 *
 * 次序是「身份 → 存在 → 授权 → 校验 → 写」：
 * - 不存在回 404（存在性问题，与 `favorites.ts:140` 同一口径）；
 * - **有资产但无 write 一律 403，先于长度校验的 400**——否则可以靠 400/403 的差别
 *   试探「合法长度是多少」，也等于把校验规则暴露给无权者（`#004-09`）。
 *
 * 回包是**读完库之后的完整元数据**，不是「我送了什么」（与收藏出口同一取向）。
 */
function handleContent(request: HttpServerRequest.HttpServerRequest, deps: Deps) {
  return Effect.gen(function* () {
    const user = yield* 身份
    if (Option.isNone(user)) return unauthorized()
    const userId = user.value.id

    const 参数 = yield* HttpRouter.params
    const assetId = 参数.id ?? ""
    const body = yield* 读取体(request)
    if (!isRecord(body)) return fail(BAD_REQUEST, "请求体不是 JSON 对象")

    const asset = yield* Effect.promise(() => assetById(deps.pg(), assetId))
    if (asset === undefined) return fail(NOT_FOUND, "资产不存在")

    const department = yield* 取部门(deps, userId)
    const grants = yield* Effect.promise(() => grantsFor(deps.pg(), userId))
    if (!effectivePermissions({ userId, department }, asset, grants).has("write")) {
      return fail(FORBIDDEN, "无权修改该资产")
    }

    const patch: Record<string, string> = {}
    for (const 字段 of ["description", "summary", "precautions"] as const) {
      const 值 = body[字段]
      if (值 === undefined) continue
      if (typeof 值 !== "string") return fail(BAD_REQUEST, `${字段} 必须是字符串`)
      patch[字段] = 值
    }
    if (Object.keys(patch).length === 0) return fail(BAD_REQUEST, "没有可修改的字段")

    const 结果 = yield* Effect.result(
      Effect.tryPromise({
        try: () => updateAssetContent(deps.pg(), assetId, patch, nowSeconds()),
        catch: (error) => error,
      }),
    )
    if (Result.isFailure(结果)) {
      const error = 结果.failure
      if (error instanceof AssetMetadataError) return fail(BAD_REQUEST, error.message)
      throw error
    }
    if (结果.success === undefined) return fail(NOT_FOUND, "资产不存在")
    return ok(结果.success)
  })
}

/** 与内核 `OpenhiveAssets.PATH.content` 逐字对应（刻意不共享，前端不 import 内核源码）。 */
export const PATH = { base: "/openhive/assets" } as const
export const TIMEOUT_MS = 10_000

export interface 内容补丁 {
  readonly description?: string
  readonly summary?: string
  readonly precautions?: string
}

export type ContentOutcome =
  | { readonly kind: "ok"; readonly value: Record<string, unknown> }
  | { readonly kind: "rejected"; readonly message: string }
  | { readonly kind: "failed"; readonly message: string }

export type ContentFetch = (input: string, init?: RequestInit) => Promise<Response>

export interface ContentApi {
  save(assetId: string, patch: 内容补丁): Promise<ContentOutcome>
}

async function 默认发送(input: string, init?: RequestInit): Promise<Response> {
  const 控制器 = new AbortController()
  const 定时器 = setTimeout(() => 控制器.abort(), TIMEOUT_MS)
  try {
    return await fetch(input, { ...init, signal: 控制器.signal })
  } finally {
    clearTimeout(定时器)
  }
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value)
}

function 消息(body: unknown, response: Response): string {
  if (isRecord(body) && typeof body["error"] === "string" && body["error"].length > 0) return body["error"]
  return `保存失败（HTTP ${response.status}）`
}

export function createContentApi(send: ContentFetch = 默认发送): ContentApi {
  return {
    save: async (assetId, patch) => {
      let response: Response
      try {
        response = await send(`${PATH.base}/${encodeURIComponent(assetId)}`, {
          method: "PATCH",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(patch),
        })
      } catch {
        return { kind: "failed", message: "保存失败，请检查网络后重试" }
      }

      let body: unknown
      try {
        body = await response.json()
      } catch {
        body = undefined
      }

      if (response.status === 401) return { kind: "rejected", message: "登录状态已失效，请重新登录" }
      if (response.status >= 500) return { kind: "failed", message: 消息(body, response) }
      if (!response.ok) return { kind: "rejected", message: 消息(body, response) }
      if (!isRecord(body)) return { kind: "failed", message: "服务返回的数据格式不对" }
      return { kind: "ok", value: body }
    },
  }
}
