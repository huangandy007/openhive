> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让表格来源的资金流水在入库时按受控词表判出用途，写进 `case_event.purpose` + `purpose_l1` + `purpose_basis`，并让 `query_events` 能按用途查，三份真文件的分类计数可复跑对账。

**Tech Stack:** Bun + TypeScript + Effect/Hono 栈内既有模块；PG（Drizzle）迁移走 `packages/auth/src/migrations/`；表格解析用既有 `case-sheet.ts`（SheetJS 0.20.3，能读 OLE2 `.xls`）。

**Spec:** `docs/superpowers/specs/2026-11-09-事实卡片抽象/01-事实卡片与数据翻译-设计.md` §3.11（用途完整正文）、§11 A20/A21、§12.1 P4；`过程文档/06-备注字段与交易语义-存储与关联.md` §6（判据、优先级、词表、基线）。两处冲突时**以 `01` §3.11 为准**（`06` §6.0 已声明）。

- Consumes: 无（不 import 任何 AI / DB / 时钟）

**语义定义（实现与测试都按这份）**：
- 规则按 `优先级` 升序**全表扫描**，first-match wins；`优先级` 必须是 10 的倍数且全表唯一（留插入空位）。
- `任一: []` 表示「只看前置」——前置满足即命中（用于「交易类型=支付账户提现/转账至银行卡」这类**字段事实**判定的粗类）。
- `字段` 三选一：`"备注原文"`（整串）、`"备注主体"`（第一个 `##` 段再按 `;` 取第一段）、`"备注#<键>"`（`##` 段里 `键：值` 的键）。字段取不到时该规则不命中。
- **`任一` 内部按「越具体越靠前」排**：依据记录的是数组里**第一个**命中的词，所以 `["P云","ETCP","停车场","停车"]` 让「P云停车平台缴费」记成 `P云` 而不是泛词 `停车`。同类内部顺序也进回归——改顺序会改 `purpose_basis`。
- 大类顺序**照抄 `01` §3.11.3 的有序表**，尤其：提现/账户充值排在全部消费二级**之前**（否则提现行的备注提到商户就被判成消费），转账排在消费二级**之后**。
- `排除` 一律对**备注原文**判定：任一排除词出现即否决（`01` §3.11.4.1「有误伤风险的词必填」）。
- 依据里 `命中` 记录**具体命中的词**；无词的粗类规则 `命中=null`、`字段=null`。
- 兜底只有**一条**显式规则 `兜底.未分类`（表末尾、无条件命中），不是代码里的 `if`。「未识别转账」不是兜底，而是 `未识别转账.填充备注`（145）——它按 `06` §6.5 认填充符 `-` / `#|#` / `工资` / `货款`，用**全等**匹配（包含匹配会让 `ETCP-临时停车缴费` 被吞）。
- `字段` 第四种取值 `"备注原文全等"`：备注（trim 后）**等于** `任一` 里的某个词才算命中。只给填充符这种「整串就是一个无信息标记」的词用。

```ts
/**
 * 备注分段解析（06 §2.1：`##` 分段，段内 `键：值`）。
 *
 * 这是**机械还原**不是解释：切段、取键值，不改一个字符（01 §6.7「预处理只做格式对齐」）。
 * 主体 = 第一个 `##` 段再按 `;` 取第一段——「零钱通作后缀」的判据要靠它（06 §6.3 修正 4）。
 */
export function 解析备注段(备注: string): { 主体: string; 段: Record<string, string> } {
  const 段: Record<string, string> = {}
  const 首段 = 备注.split("##")[0] ?? ""
  for (const 一 of 备注.split("##")) {
    const 文本 = 一.trim()
    // 冒号必须在行首之后：`键` 为空或整段就是冒号时不算键值。
    const 冒号 = 文本.search(/[：:]/)
    if (冒号 <= 0) continue
    const 键 = 文本.slice(0, 冒号).trim()
    const 值 = 文本.slice(冒号 + 1).trim()
    // 同键重复取第一个：后出现的多半是拼接噪音，覆盖会让结果依赖拼接顺序。
    if (键 !== "" && !Object.hasOwn(段, 键)) 段[键] = 值
  }
  return { 主体: (首段.split(";")[0] ?? "").trim(), 段 }
}

function 字段取值(字段: string, 备注: string, 段: Record<string, string>, 主体: string): string | null {
  if (字段 === "备注原文" || 字段 === "备注原文全等") return 备注
  if (字段 === "备注主体") return 主体
  if (字段.startsWith("备注#")) return 段[字段.slice("备注#".length)] ?? null
  return null
}

/**
 * 判定一张卡片（一行流水）的用途。
 *
 * 规则表必须**已按优先级升序**排好；顺序错、id 重复、类名越界由 `校验规则表` 在测试里挡，
 * 运行时不再兜（生产环境不该为「数据表被写坏」付每行一次校验的代价）。
 */
export function 判定用途(输入: 判定输入, 规则们: readonly 用途规则[] = 用途规则们): 用途 {
  const 备注 = (输入.备注 ?? "").trim()
  const 交易类型 = (输入.交易类型 ?? "").trim()
  const { 主体, 段 } = 解析备注段(备注)

  for (const 规则 of 规则们) {
    // 前置只取①层字段事实：拿不到就不猜（规则不命中，往下走）。
    if (规则.前置?.交易类型 !== undefined) {
      if (交易类型 === "" || !规则.前置.交易类型.includes(交易类型)) continue
    }
    if (规则.前置?.方向 !== undefined && 输入.方向 !== 规则.前置.方向) continue

    const 值 = 字段取值(规则.字段, 备注, 段, 主体)
    if (值 === null && 规则.任一.length > 0) continue
    // 两种匹配语义：`备注原文全等` 只给填充符（`-` / `#|#`），其余一律包含匹配。
    const 命中 =
      规则.任一.length === 0
        ? null
        : (规则.任一.find((词) =>
            规则.字段 === "备注原文全等" ? 值 === 词 : (值 ?? "").includes(词),
          ) ?? null)
    if (规则.任一.length > 0 && 命中 === null) continue
    // 排除一律看原文：同一句备注里出现「便利店」就否决「加油」，哪怕命中词在别的段。
    if (规则.排除 !== undefined && 规则.排除.some((词) => 备注.includes(词))) continue

    return {
      一级: 规则.输出.一级,
      二级: 规则.输出.二级 ?? null,
      依据: { 规则: 规则.id, 命中, 字段: 命中 === null ? null : 规则.字段, 词表: 词表版本 },
    }
  }

  // 兜底规则在表末尾无条件命中，走到这里说明规则表被改坏（例如删掉了兜底）。
  // 不抛错：入库路径上抛错会让整份材料失败，宁可交一张「未分类」给人工。
  // 依据写 `兜底.缺失`（不是 `兜底.未分类`）：审计要能区分「兜底真命中」与「表坏了」。
  return {
    一级: "未分类",
    二级: null,
    依据: { 规则: "兜底.缺失", 命中: null, 字段: null, 词表: 词表版本 },
  }
}

/**
 * 规则表准入校验（01 §3.11.4.5 四条纪律的机器化）。
 *
 * 返回问题清单，空数组 = 合规。只在测试里跑，不在热路径上跑。
 */
export function 校验规则表(规则们: readonly 用途规则[]): string[] {
  const 问题: string[] = []
  const 已见id = new Set<string>()
  const 已见优先级 = new Set<number>()
  let 上次 = -Infinity

  for (const 规则 of 规则们) {
    if (已见id.has(规则.id)) 问题.push(`规则 id 重复：${规则.id}`)
    已见id.add(规则.id)
    if (已见优先级.has(规则.优先级)) 问题.push(`优先级重复：${规则.优先级}（${规则.id}）`)
    已见优先级.add(规则.优先级)
    if (规则.优先级 <= 上次) 问题.push(`规则表未按优先级升序：${规则.id}`)
    if (规则.优先级 % 10 !== 0) 问题.push(`优先级不是 10 的倍数：${规则.id}`)
    // `适用源` 声明了就必须有值：引擎本期不按来源过滤，但空数组等于「永不适用」，
    // 会让 T5 加词后以为规则生效、实际从未命中（Minor：声明了却无人读的字段）。
    if (规则.适用源.length === 0) 问题.push(`适用源为空：${规则.id}`)
    上次 = 规则.优先级

    // 一级 12 类已有「未分类」「未识别转账」两个出口，类名写错必须当场发现。
    if (!(一级类们 as readonly string[]).includes(规则.输出.一级))
      问题.push(`一级类不在词表内：${规则.id} → ${规则.输出.一级}`)
    if (规则.输出.二级 !== undefined && !(二级类们 as readonly string[]).includes(规则.输出.二级))
      问题.push(`二级类不在词表内：${规则.id} → ${规则.输出.二级}`)
    // 非消费不能有二级：词表只给消费生长（01 §3.11.4）。
    if (规则.输出.二级 !== undefined && 规则.输出.一级 !== "消费")
      问题.push(`非消费类挂了二级：${规则.id}`)

    for (const 词 of [...规则.任一, ...(规则.排除 ?? [])]) {
      // 空词永远禁入；「单字」禁的是单个汉字（「店」「费」这类通用字），
      // 填充符 `-` / `#|#` 是精确标记、不是泛词，按单字规则拦掉会让填充符规则写不出来。
      if (词.trim() === "") 问题.push(`单字/空词禁入：${规则.id} 的空词`)
      else if (词.trim().length < 2 && /[一-鿿]/.test(词))
        问题.push(`单字/空词禁入：${规则.id} 的「${词}」`)
    }
    if (
      规则.字段 !== "备注原文" &&
      规则.字段 !== "备注原文全等" &&
      规则.字段 !== "备注主体" &&
      !(规则.字段.startsWith("备注#") && 规则.字段.length > "备注#".length)
    )
      问题.push(`未知字段：${规则.id} → ${规则.字段}`)
  }
  return 问题
}
```

```ts
  文件: string
  行数: number
  sha256: string
}

  文件们: 清单项[]
}

  console.error('用法：bun run script/openhive-case-purpose-baseline.ts "<目录>" ["<699 标准副本.xlsx>"]')
  process.exit(1)
}

  readFileSync(join(import.meta.dir, "openhive-case-purpose-baseline.json"), "utf8"),
) as 清单

const 一级计: Record<string, number> = {}
const 二级计: Record<string, number> = {}
let 总行 = 0

for (const 项 of 清单.文件们) {
  const bytes = readFileSync(join(目录, 项.文件))
  const sha = createHash("sha256").update(bytes).digest("hex")
  if (sha !== 项.sha256) {
    console.error(`文件指纹不符：${项.文件}\n  期望 ${项.sha256}\n  实得 ${sha}`)
    process.exit(1)

  const 表 = readSheets(new Uint8Array(bytes)).sheets[0]
  if (表 === undefined) throw new Error(`没有工作表：${项.文件}`)
  // `header` 是 `RawCell[]` 不是字符串数组：按**显示文本**找列（case-sheet.ts:66）。
  const 表头 = 表.header.map((c) => c.text ?? "")

  const 类型列 = 表头.indexOf("交易类型")
  if (备注列 < 0) throw new Error(`列「备注」不在表头：${项.文件}`)

    console.error(`行数不符：${项.文件}\n  期望 ${项.行数}\n  实得 ${表.rows.length}`)
    process.exit(1)
  }
```
