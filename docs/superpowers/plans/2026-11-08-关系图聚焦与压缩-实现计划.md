# 关系图聚焦与压缩 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 关系数据一多，图上不再长成一团毛线球：边先按端点聚合、按强度切片、枢纽单独标出、二跳按父节点分簇，下钻换一张干净的图。

**Architecture:** 后端口径先修（`case-graph.ts` 的 SQL 从「一条关系一行」改成「一对端点一条聚合边」，并新增强度档、真实度数、二跳父节点）；前端只消费这些字段：两档模式是三个查看偏好的派生状态，不做第二套状态机；布局只增加一个「按 `经由` 排序」开关与环形弧。

**Tech Stack:** Bun 1.4.3 monorepo、TypeScript 5.8、Drizzle ORM（`sql` 模板）、PGlite 测试库、SolidJS 1.9.10、内联 SVG（不引 d3 / graphology）。

**Spec:** `docs/superpowers/specs/2026-11-08-关系图聚焦与压缩-design.md`

把 `取一圈`（`:104-165`）整体换成下面三块。`边行` 只换字段，`from_value` 等实体字段不动：

```ts
interface 聚合入参 {
  readonly projectId: string
  readonly 节点们: readonly string[]
  readonly 类型?: readonly string[]
  readonly 强度: 关系图强度
}

/**
 * 一组端点一条边：`(from_id, to_id, type)` 相同的关系先合成一行**再**去抢名额。
 * 不聚合的话，几百条流水只带出 1 个新节点，别的关系全被挤出上限（spec §4.1.2）。
 */
function 聚合边(input: 聚合入参): SQL {
  const 节点表 = sql.join(input.节点们.map((id) => sql`${id}`), sql`, `)
  const 类型条件 =
    input.类型 === undefined || input.类型.length === 0
      ? sql``
      : sql`and r.type in (${sql.join(input.类型.map((条) => sql`${条}`), sql`, `)})`
  const 阈 = 强度阈值[input.强度]
  const 金额阈 = 阈.金额 === null ? sql`null::numeric` : sql`${阈.金额}::numeric`
  return sql`
    select g.from_id, g.to_id, g.type, g.次数, g.金额数, g.金额合计, g.明细数, g.信任态,
           coalesce(g.次数 >= ${阈.次数} or (${金额阈} is not null and g.金额数 >= ${金额阈}), false) as 达标
      from (
        select r.from_id, r.to_id, r.type,
               sum(coalesce(r.times, 1))::int as 次数,
               sum(r.amount) as 金额数,
               sum(r.amount)::text as 金额合计,
               count(*)::int as 明细数,
               case when min(r.trust_state) = max(r.trust_state) then min(r.trust_state) else '混合' end as 信任态
          from auth.case_relation r
          join auth.case_entity 起 on 起.project_id = r.project_id and 起.id = r.from_id
          join auth.case_entity 终 on 终.project_id = r.project_id and 终.id = r.to_id
         where r.project_id = ${input.projectId}
           and r.trust_state <> '已排除'
           and 起.trust_state <> '已排除' and 终.trust_state <> '已排除'
           and (r.from_id in (${节点表}) or r.to_id in (${节点表}))
           ${类型条件}
         group by r.from_id, r.to_id, r.type
      ) g
  `
}

interface 一圈 {
  readonly 边们: 边行[]
  /** 本跳**达标**的全部组数（`count(*) over()`，不受 `limit` 影响）。 */
  readonly 总数: number
}

async function 取一圈(
  db: Queryable,
  input: {
    readonly projectId: string
    readonly 节点们: readonly string[]
    readonly 类型?: readonly string[]
    readonly 剩余: number
    readonly 强度: 关系图强度
  },
): Promise<一圈> {
  if (input.节点们.length === 0) return { 边们: [], 总数: 0 }
  const 行们 = rowsOf(
    await db.execute(sql`
      select h.from_id, h.to_id, h.type, h.次数, h.金额合计 as amount, h.信任态 as trust_state, h.明细数,
             起.canonical_value as from_value, 起.type as from_type, 起.trust_state as from_trust,
             起.attrs as from_attrs,
             终.canonical_value as to_value, 终.type as to_type, 终.trust_state as to_trust,
             终.attrs as to_attrs,
             (count(*) over())::int as 总数
        from (${聚合边({ projectId: input.projectId, 节点们: input.节点们, 类型: input.类型, 强度: input.强度 })}) h
        join auth.case_entity 起 on 起.project_id = ${input.projectId} and 起.id = h.from_id
        join auth.case_entity 终 on 终.project_id = ${input.projectId} and 终.id = h.to_id
       where h.达标
       order by h.次数 desc, h.金额数 desc nulls last, h.明细数 desc,
                h.from_id collate "C" asc, h.to_id collate "C" asc, h.type collate "C" asc
       limit ${input.剩余 + 1}
    `),
  )
  return {
    边们: 行们.map((行) => ({
      id: `rel|${行.from_id}|${行.to_id}|${行.type}`,
      from_id: String(行.from_id),
      to_id: String(行.to_id),
      type: String(行.type),
      times: Number(行.次数),
      amount: (行.amount as string | null) ?? null,
      trust_state: String(行.trust_state),
      明细数: Number(行.明细数),
      from_value: String(行.from_value),
      from_type: String(行.from_type),
      from_trust: String(行.from_trust),
      from_attrs: 行.from_attrs,
      to_value: String(行.to_value),
      to_type: String(行.to_type),
      to_trust: String(行.to_trust),
      to_attrs: 行.to_attrs,
    })),
    总数: 行们.length === 0 ? 0 : Number(行们[0].总数),
  }
}

/**
 * 范围里**全部**组数 vs **达标**组数（不受 `limit` 影响，永远回一行）。给 `强度缺口` 用。
 * 为什么不像 spec §4.1.6 那样把两个窗口塞进取一圈：那条 SQL 带 `where h.达标`，
 * 一旦一条都不达标，窗口值随行一起被筛掉，`全部组数` 就再也拿不到；
 * 聚合函数不带 group by 时恒回一行，才稳。
 */
async function 计强度(
  db: Queryable,
  input: 聚合入参,
): Promise<{ readonly 全部: number; readonly 达标: number }> {
  if (input.节点们.length === 0) return { 全部: 0, 达标: 0 }
  const 行 = rowsOf(
    await db.execute(sql`
      select (count(*))::int as 全部, (count(*) filter (where h.达标))::int as 达标
        from (${聚合边(input)}) h
    `),
  )[0]
  return { 全部: Number(行.全部), 达标: Number(行.达标) }
}
```

`边行` 接口同步换：删掉数据库 `id`，`times: number`，加 `明细数: number`。

export function 环半径(尺寸: 画布尺寸 = 画布) {
  const 半宽 = Math.max(60, 尺寸.宽 / 2 - 70)
  const 半高 = Math.max(60, 尺寸.高 / 2 - 55)
  return { 一: { rx: 半宽 * 0.55, ry: 半高 * 0.55 }, 二: { rx: 半宽, ry: 半高 } } as const
}

  const 中 = { x: 尺寸.宽 / 2, y: 尺寸.高 / 2 }
  const 环 = 环半径(尺寸)
  const 排 = (点们: readonly 图节点[], 半径: { readonly rx: number; readonly ry: number }, 按经由 = false) => {
    // 排序键必须与 按经由分簇 同源：父不在图上 ⇒ 归「其他」（""），否则两个不同的
    // 孤儿 经由 会散开，「其他」组不连续、弧会跨过别人的组（2026-11-08 复审 S2）。
    const 已知 = new Set(图.节点.map((点) => 点.id))
    const 分键 = (点: 图节点) => (点.经由 !== undefined && 已知.has(点.经由) ? 点.经由 : "")
    const 有序 = [...点们].sort((甲, 乙) =>
      按经由
        ? 分键(甲).localeCompare(分键(乙)) || 甲.type.localeCompare(乙.type) || 甲.id.localeCompare(乙.id)
        : 甲.type.localeCompare(乙.type) || 甲.id.localeCompare(乙.id),
    )
    const 偏 = 有序.length >= 2 ? Math.PI / 有序.length : 0
    return 有序.map((点, 序号): 位点 => {
      const 角 = (Math.PI * 2 * 序号) / Math.max(有序.length, 1) - Math.PI / 2 + 偏
      return { ...点, x: 中.x + Math.cos(角) * 半径.rx, y: 中.y + Math.sin(角) * 半径.ry }
    })
  }
  return {
    中心: 图.节点.filter((点) => 点.中心).map((点): 位点 => ({ ...点, x: 中.x, y: 中.y })),
    一层: 排(图.节点.filter((点) => !点.中心 && 点.跳数 < 2), 环.一),
    二层: 排(图.节点.filter((点) => !点.中心 && 点.跳数 >= 2), 环.二, 分簇),
  }
}

```ts
export interface 簇 {
  /** 经由的父节点 id；`""` = 其他（父不在图上 / 没有经由）。 */
  readonly 键: string
  readonly 标签: string
  readonly 点们: 位点[]
}

/**
 * 二跳点按 `经由` 分组。入参顺序必须是**布局顺序**（`全部点()` 里 2 跳那一段），
 * 否则同父的点不连续，弧会跨过别人的组。
 */
export function 按经由分簇(二层: readonly 位点[], 图: 关系图结果): 簇[] {
  const 值 = new Map(图.节点.map((点) => [点.id, 点.value]))
  const 组 = new Map<string, 位点[]>()
  for (const 点 of 二层) {
    const 键 = 点.经由 !== undefined && 值.has(点.经由) ? 点.经由 : ""
    组.set(键, [...(组.get(键) ?? []), 点])
  }
  return [...组.entries()]
    .map(([键, 点们]) => ({ 键, 标签: 键 === "" ? "其他" : `经由 ${值.get(键)}`, 点们 }))
    .sort((甲, 乙) => 乙.点们.length - 甲.点们.length || 甲.键.localeCompare(乙.键))
}
```
